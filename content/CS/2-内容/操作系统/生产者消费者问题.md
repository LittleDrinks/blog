生产者—消费者问题是并发[[进程]] 共享有界缓冲池的经典同步互斥模型。
一组生产者进程向容量为 $n$ 的环形缓冲区写入数据，一组消费者进程从缓冲区读取数据。

> [!faq]- 【14-15 秋期末二·7】生产者-消费者问题是（　）。</br>A. 工厂生产调度问题</br>B. 使用缓冲区时的同步互斥问题</br>C. 资源分配问题</br>D. 进程调度问题
> 选 B。
> 
> 生产者-消费者问题的本质是并发进程在共享有界缓冲区时的同步与互斥问题。

> [!faq]- 【第 2 章作业题 1·(1)】请用[[记录型信号量]] 解决生产者-消费者问题，并写出相应的伪代码。
> ```c
> int in = 0, out = 0;
> item buffer[n];
> semaphore mutex = 1, empty = n, full = 0;  // 有n个空，有0个满
> 
> void producer() {
>     do {
>         produce an item in nextp;
>         wait(empty);      // 先申请空缓冲（同步信号量）
>         wait(mutex);      // 再申请缓冲池互斥锁
>         buffer[in] = nextp;
>         in = (in + 1) % n;
>         signal(mutex);
>         signal(full);
>     } while (TRUE);
> }
> 
> void consumer() {
>     do {
>         wait(full);       // 先申请满缓冲（同步信号量）
>         wait(mutex);      // 再申请缓冲池互斥锁
>         nextc = buffer[out];
>         out = (out + 1) % n;
>         signal(mutex);
>         signal(empty);
>         consume the item in nextc;
>     } while (TRUE);
> }
> 
> void main() {
>     cobegin producer(); consumer(); coend
> }
> ```

> [!faq]- 【第 2 章作业题 1·(2)】请用 [[AND型信号量]] 解决生产者-消费者问题，并写出相应的伪代码。
> ```c
> int in = 0, out = 0;
> item buffer[n];
> semaphore mutex = 1, empty = n, full = 0;
> 
> void producer() {
>     do {
>         produce an item in nextp;
>         Swait(empty, mutex); // 原子性同时申请空缓冲与互斥锁
>         buffer[in] = nextp;
>         in = (in + 1) % n;
>         Ssignal(mutex, full);
>     } while (TRUE);
> }
> 
> void consumer() {
>     do {
>         Swait(full, mutex);  // 原子性同时申请满缓冲与互斥锁
>         nextc = buffer[out];
>         out = (out + 1) % n;
>         Ssignal(mutex, empty);
>         consume the item in nextc;
>     } while (TRUE);
> }
> ```

用[[管程]]解决：
```cpp
monitor ProducerConsumer {
    condition full, empty;  // 分别表示 not_full 和 not_empty
    int count=0;
    void insert(Item item) {
        if (count == N) full.wait();
        count++;
        insert_item(item);
        if (count == 1) empty.signal();
    }
    void remove() {
        if (count == 0) empty.wait();
        count--;
        if (count == N) signal(full);
        return remove_item();
    }
}
end monitor;

Producer() { while(1) { item=produce(); ProducerConsumer.insert(item); } }
Consumer() { while(1) { item=ProducerConsumer.remove(); consume(item); } }
```
