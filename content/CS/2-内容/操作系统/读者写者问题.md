多个并发 [[进程]] 共享同一个文件或数据区：允许多个读者同时并发读取，但严禁写者与任何其他读者或写者同时访问。

> [!faq]- 【第2章作业题1·(5)】请用 [[记录型信号量]] 解决读者-写者问题，并写出相应的伪代码。
> ```c
> semaphore rmutex = 1, wmutex = 1;
> semaphore w = 1; // 用于读写公平排队，防止写者饥饿
> int readcount = 0;
> 
> void reader() {
>     do {
>         wait(w);
>         wait(rmutex);
>         if (readcount == 0) wait(wmutex); // 第 1 个读者负责锁写者
>         readcount++;
>         signal(rmutex);
>         signal(w);
>         // perform read operation
>         wait(rmutex);
>         readcount--;
>         if (readcount == 0) signal(wmutex); // 最后 1 个读者负责释放写锁
>         signal(rmutex);
>     } while (TRUE);
> }
> 
> void writer() {
>     do {
>         wait(w);
>         wait(wmutex);
>         // perform write operation
>         signal(wmutex);
>         signal(w);
>     } while (TRUE);
> }
> 
> void main() {
>     cobegin reader(); writer(); coend
> }
> ```

> [!faq]- 【第2章作业题1·(6)】请用 [[信号量集]] 机制解决读者-写者问题（最多允许 $RN$ 个读者同时读），并写出相应的伪代码。
> ```c
> int RN;
> semaphore L = RN, mx = 1;
> 
> void reader() {
>     do {
>         Swait(L, 1, 1);
>         Swait(mx, 1, 0); // 当 mx >= 1（无写者）时放行，不消耗 mx
>         // perform read operation
>         Ssignal(L, 1);
>     } while (TRUE);
> }
> 
> void writer() {
>     do {
>         Swait(mx, 1, 1, L, RN, 0); // 既无 writer 在写(mx=1)、又无 reader 在读(L=RN)才能写
>         // perform write operation
>         Ssignal(mx, 1);
>     } while (TRUE);
> }
> 
> void main() {
>     cobegin reader(); writer(); coend
> }
> ```
