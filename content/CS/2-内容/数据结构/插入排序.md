```cpp
/*
顺序表上的直接插入
*/
template <class ElemType>
void StraightInsertSort(ElemType elem[], int n){
	int i, j;				
	for (i=1; i < n; i++)	{	     // 第i趟直接插入排序
		ElemType e=elem[i];     // 暂存elem[i]
		for (j=i - 1; j >= 0 && e < elem[j]; j--)	
			elem[j + 1]=elem[j];
		elem[j + 1]=e;	     // j+1为插入位置
	}
}
```

```cpp
/*
链表上的直接插入排序
*/
template <class ElemType>
void LinkInsertSort(Node<ElemType> elem[])  {
    int p, q, pre, r;
    p=elem[0].next;    elem[0].next=-1;
    while (p != -1) {
         q=elem[0].next;         pre=0;
         while (q != -1 && elem[q].data <= elem[p].data) {
               pre=q;               q=elem[q].next;
         }
         r=elem[p].next;         elem[p].next=q;
         elem[pre].next=p;     p=r;
	}
}

```
比较次数：$\sum_{i=1}^{n-1}i=n(n-1) / 2$
移动次数：$\sum_{i=1}^{n-1}(i+2)=(n+4)(n-1)/ 2$

> [!faq]- 对一组记录 $(54,38,106,21,15,72,60,45,83)$ 进行直接插入排序，当把元素 $60$ 插入到有序表时，为寻找插入位置需比较多少次？
> $3$ 次
