```cpp
template <class ElemType>
void BubbleSort(ElemType elem[], int n)  {
    bool finish=false;
    int i=1;
    while (i < n && !finish)	{
         finish=true;  
         for (int j=0; j < n - i; j++)
	    if (elem[j] > elem[j + 1]) {
		Swap(elem[j], elem[j + 1]);
		finish=false;
               }
          i++;
     }
}

```

比较次数 $\sum_{i=1}^{n-1}(n-i)=\frac{n(n-1)}{2}$
移动次数 $\sum_{i=1}^{n-1}3(n-i)=\frac{3n(n-1)}{2}$
