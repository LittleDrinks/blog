```cpp
template <class ElemType>
void ShellSort(ElemType elem[], int n)   {
     int i, j, d=n / 2;            // 初始步长取n/2 
     ElemType e;
     while (d > 0) {
            for (i=d ; i < n; i++)	{
	        e=elem[i];	  j=i - d; 
                   while (j >= 0 && e < elem[j]) {
		     elem[j + d]=elem[j];	j -= d;
                   }	
	        elem[j + d]=e;
             }
             d=d / 2;                   // 缩短步长 
       }
}
```
