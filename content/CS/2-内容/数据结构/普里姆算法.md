// ==========================================
// 最小生成树 Prim 算法（完整代码）
// ==========================================

// 1. 结构体定义
template <class ElemType, class WeightType>
struct CloseArcType {
    WeightType lowweight; // 记录当前顶点到生成树集合的最小权值
    int nearV;            // 记录对应的最小权值边在生成树集合中的顶点下标
};

// 2. 算法主函数
template <class ElemType, class WeightType>
void MiniSpanTreePrim(const AdjMatrixUndirNetwork<ElemType, WeightType>& G, int u0) 
{
    WeightType min;
    ElemType v1, v2;
    int vexnum = G.GetVexNum();
    CloseArcType<ElemType, WeightType>* closearc;
    
    // 顶点合法性检查
    if (u0 < 0 || u0 >= vexnum) throw Error("顶点 u0 不存在!"); 
    
    int u, v, k;
    closearc = new CloseArcType<ElemType, WeightType>[vexnum];
    
    // 初始化辅助数组 closearc
    for (v = 0; v < vexnum; v++) {
        closearc[v].nearV = u0;
        closearc[v].lowweight = G.GetWeight(u0, v);
    }
    
    // 将 u0 标记为已加入生成树集合 (lowweight = 0 代表已入树)
    closearc[u0].nearV = -1; 
    closearc[u0].lowweight = 0; 
    
    // 循环执行 vexnum - 1 次，找出剩余的顶点
    for (k = 1; k < vexnum; k++) {
        min = G.GetInfinity();
        v = u0;
        
        // 从所有未加入生成树的顶点中，选出与当前树集合距离最近的顶点 v
        for (u = 0; u < vexnum; u++) {
            if (closearc[u].lowweight != 0 && closearc[u].lowweight < min) {
                v = u;
                min = closearc[u].lowweight; // (5) 更新最小权值
            }
        }
        
        if (v != u0) {
            G.GetElem(closearc[v].nearV, v1);
            G.GetElem(v, v2);
            cout << "边( " << v1 << ", " << v2 << " ) 权:" << min << endl;
            
            closearc[v].lowweight = 0; // (6) 将顶点 v 加入生成树集合
            
            // 遍历新加入顶点 v 的所有邻接点，更新其到生成树集合的最近距离
            // *注：原图代码中使用的是 net，实际参数名为 G，这里做了修正*
            for (u = G.FirstAdjVex(v); u != -1 ; u = G.NextAdjVex(v, u)) {
                if (closearc[u].lowweight != 0 && (G.GetWeight(v, u) < closearc[u].lowweight)) {
                    closearc[u].lowweight = G.GetWeight(v, u);
                    closearc[u].nearV = v; // (7) 更新最临近顶点为新加入的 v
                }
            }
        }
    }
    
    delete []closearc;
}
