/*
顺序表上的简单选择排序
*/
template <class ElemType>
void SimpleSelectionSort(ElemType elem[], int n)  {
      int k; 
      for (int i=0; i < n - 1; i++)	{
	k=i;
	for (int j=i + 1; j < n; j++)
	     if (elem[j] < elem[k])   k=j;
	 if (k != i)
	     Swap(elem[i], elem[k]);
       }
}

/*

*/
template <class ElemType>
void LinkSelectionSort(Node<ElemType> elem[])  {
    int p, q, pp, pq, h ;
    h=p=elem[0].next;      elem[0].next=-1;   
    while ( p != -1 ) {
        q=pq=pp=p;    p=elem[p].next;
        while (p != -1) { 
            if ( elem[p].data >= elem[q].data) {
                q=p ;       pq=pp ; 
            }
            pp=p ;         p=elem[p].next;
       } 
       if ( h == q ) {
           h=elem[q].next;   elem[q].next=elem[0].next;  elem[0].next=q;
       } 
       else  {
           elem[pq].next=elem[q].next;elem[q].next=elem[0].nextelem[0].next=q;
       }
       p=h ; 
    }
}
