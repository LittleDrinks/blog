template <class ElemType>
int BinSearch(ElemType elem[], int n, ElemType key) {
	int low=0, high=n+1;
	int mid;
	while (low <= high) {
		mid = (low + high) / 2;
		if (key == elem[mid])
			return mid;
		else if (key <= elem[mid])
			high = mid - 1;
		else
			low = mid + 1;
	}
	return -1;
}
