## [QOJ18365. 高阶情景口语](https://qoj.ac/contest/3784/problem/18365)

设初始序列排序后为 $a_{(1)}\le a_{(2)}\le \cdots \le a_{(n)}$，一个数能活到最后，当且仅当它在排序后的数 $[2,n-1]$ 位
由于只有唯一最小值和唯一最大值可能被排除，所以可能分数个数是：
不妨设一共存在 $s$ 种数，每种数出现的次数为 $d_i$，则可以存活到最后的数的个数为
$$s-[d_1=1]-[d_s=1]$$
因此可及格分数集合大小只可能来自三种情况：$s=k,k+1,k+2$
先特判边界，当 $k>m$ 或 $k>n-2$ 时无解

---

定义 $F(N,T)$ 表示 $N$ 个数字，每个数字可以从 $T$ 种数中选择一个填入，每种数字都至少出现一次的方案数，这是经典的“满射计数”，可以通过容斥计算
$$
F(N,T)=\sum_{j=0}^{T}(-1)^{T-j}\binom{T}{j}j^N  
$$
接下来分三种情况计数：

情况一：$s=k$
$$
A=F(n,k) - \underbrace{2nF(n-1,k-1)}_{去掉最值只出现一次的情况} + \underbrace{n(n-1)F(n-2,k-2)}_{容斥，加回最大最小值都只出现一次的情况}
$$
情况二：$s=k+1$
$$
B=2\left(\underbrace{nF(n-1,k)}_{最值只出现一次的情况}-\underbrace{n(n-1)F(n-2,k-1)}_{去掉最大最小值都只出现一次的情况}\right)
$$
情况三：$s=k+2$
$$
C=\underbrace{n(n-1)F(n-2,k)}_{最大最小值都只出现一次}
$$
最终答案为
$$
{m\choose k}A+{m\choose k+1}B+{m\choose k+2}C
$$

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int MOD = 998244353, N = 1e6 + 5;
template<class...T> ll add(T...t){ll r=0;(((r+=t%MOD)>=MOD?r-=MOD:0),...);return(r%MOD+MOD)%MOD;}
template<class...T> ll mul(T...t){ll r=1;((r=r*(t%MOD)%MOD),...);return(r%MOD+MOD)%MOD;}
ll del(ll x,ll y){return add(x,MOD-y);}
ll qpow(ll a,ll b=MOD-2){ll r=1;for(;b;b>>=1,a=mul(a,a))if(b&1)r=mul(r,a);return r;}
ll fac[N + 5], Inv[N + 5];
void preprocess() {
    fac[0] = 1;
    for (int i = 1; i <= N; ++i) fac[i] = mul(fac[i-1], i);
    Inv[N] = qpow(fac[N]);
    for (int i = N - 1; ~i; --i) Inv[i] = mul(Inv[i + 1], i + 1);
}
ll Csmall(ll n, ll m) { return mul(fac[n], Inv[n-m], Inv[m]); }
ll Cbig(ll n, ll m) {
    if (m < 0 || m > n) return 0;
    ll res = 1;
    for (ll x = n; x >= n-m+1; --x) res = mul(res, x);
    return mul(res, Inv[m]);
}

ll f(ll N, ll T) {
    ll res = 0;
    for (int j = 0; j <= T; ++j) {
        if ((T - j) % 2 == 0) {
            res = add(res, mul(Csmall(T, j), qpow(j, N)));
        } else {
            res = del(res, mul(Csmall(T, j), qpow(j, N)));
        }
    }
    return res;
}
 
int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    preprocess();
    int n, m, k;
    cin >> n >> m >> k;
    if (k > n-2 || k > m) {
        cout << "0\n"; return 0;
    }

    ll ans1 = f(n, k);
    if (k-1 >= 0) ans1 = del(ans1, mul(2, n, f(n-1, k-1)));
    if (k-2 >= 0) ans1 = add(ans1, mul(n, n-1, f(n-2, k-2)));

    ll ans2 = mul(2, n, f(n-1, k));
    if (k-1 >= 0) ans2 = del(ans2, mul(2, n, n-1, f(n-2, k-1)));

    ll ans3 = mul(n, n-1, f(n-2, k));
    
    // cerr << mul(C(m, k), ans1) << " " <<  mul(C(m, k+1), ans2) << " " << mul(C(m, k+2), ans3) << "\n";

    cout << add(mul(Cbig(m, k), ans1), mul(Cbig(m, k+1), ans2), mul(Cbig(m, k+2), ans3)) << "\n";
}
