## [QOJ18472. 交换余生](https://qoj.ac/contest/3789/problem/18472)

先把所有数同时除以 $\gcd$，这不会影响答案。
显然不能两侧 $\gcd$ 都为 $1$，先把至少出现两次 $1$ 的情况判掉。
考虑构造，我们需要为前后缀选择两个不同的 $\gcd$，最简单的做法就是选两个不同的质数，让前后缀 $\gcd$ 分别是这两个质数的倍数。显然这和原来的条件是等价的。
设 $P,Q$ 为两个不同的质数，$(z)$ 表示 $z$ 可有可无，不难发现最终的答案序列一定形如
$$
\begin{pmatrix}
i_1P & i_2P & \cdots & iP & (z) & j_1Q & j_2Q & \cdots & jQ
\end{pmatrix}
$$
原序列中，$a_1$ 必须被分配到三部分中的一个，于是我们可以对 $a_1$ 分解质因数，枚举质因子作为 $P$，这部分的复杂度是 $O(\sqrt A)$。
取出剩下数字中不包含质因子 $P$ 的部分，记为集合 $W$，由于 $\gcd(a)=1$，所以 $|W|\ge1$
当 $|W|=1$ 时，把那个数字作为 $z$ 放到中间，其他任意放，总可以满足条件
当 $|W|>1$ 时，需要选择一个质因子 $Q$，使得 $W$ 中至多只有一个不包含质因子 $Q$，把那个数作为 $z$ 构造即可。这步 check 可以通过预处理 $W$ 的前后缀 $\gcd$ 实现，时间复杂度 $O(n\log n)$
如果 check 失败，说明 $a_1$ 就是 $z$，再对 $a_2$ 做一遍相同的 check 即可。
于是总时间复杂度为 $O(T\sqrt A+\omega(A)\sum n\log A)$。另解参考官方题解。

[***AC 代码***](https://qoj.ac/submission/2521072)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

vector<int> sieve(int n) {
	vector<bool> vis(n+1);
    vector<int> mP(n+1,1);
	vector<int> prime;
	vis[1] = 1;
	for (int i = 2; i <= n; ++i) {
		if (!vis[i]) {
            prime.push_back(i); 
            mP[i] = i;
        }
		for (auto p: prime) {
			if (i*p > n) { break; }
			vis[i*p] = 1;
            mP[i*p] = p;
			if (i%p == 0) { break; }  // 线性筛关键优化
		}
	}
	return prime;
}

const int N = 1e6 + 5;
const vector<int> P = sieve(N);

void solve() {
    int n;
    cin >> n;
    vector<ll> a(n + 1);
    ll g = 0;
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
        g = gcd(g, a[i]);
    }
    for (int i = 1; i <= n; ++i) {
        a[i] /= g;
    }

    if (count(a.begin() + 1, a.end(), 1) > 1) {
        cout << "NO" << "\n";
        return;
    }

    auto check_q = [&](vector<ll> r) -> bool {
        int siz = r.size();
        if (siz <= 1) return true;
        r.insert(r.begin(), 0);
        r.push_back(0);
        vector<ll> pre(r), suf(r);
        for (int i = 1; i <= siz; ++i) {
            pre[i] = gcd(pre[i], pre[i-1]);
        }
        for (int i = siz; i; --i) {
            suf[i] = gcd(suf[i], suf[i + 1]);
        }
        for (int i = 1; i <= siz; ++i) {
            if (gcd(pre[i-1], suf[i+1]) > 1) return true;
        }
        return false;
    };

    auto check_p = [&](int id) -> bool {
        ll x = a[id];
        bool ok = false;
        vector<ll> pm;
        for (auto y: P) {
            if (x % y == 0) {
                while (x % y == 0) x /= y;
                pm.push_back(y);
            }
        }
        if (x > 1) pm.push_back(x);
        for (auto y: pm) {
            vector<ll> r;
            for (int i = 1; i <= n; ++i) {
                if (i == id) continue;
                if (a[i] % y != 0) {
                    r.push_back(a[i]);
                }
            }
            ok |= check_q(r);
        }
        return ok;
    };

    bool ok = check_p(1) || check_p(2);
    cout << (ok ? "YES": "NO") << "\n";
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
