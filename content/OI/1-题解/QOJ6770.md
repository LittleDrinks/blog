## [QOJ6770. Ants](https://qoj.ac/contest/1292/problem/6770)

一个经典 trick：两只蚂蚁相撞后弹回，相当于两只蚂蚁互相穿过。
可以直接计算左右侧墙壁倒塌的准确时间，然后每一只蚂蚁在墙倒塌后下一次来到这里就会掉下去。
也可以直接跳过 $\frac{\min(A,B)}{n}$ 轮，然后往后模拟至多两轮。

[***AC 代码***](https://qoj.ac/submission/2198683)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

int main()
{
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    ll n, A, B, base=1e9+1;
    cin >> n >> A >> B;
    vector<ll> a(n), d(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    for (int i = 0; i < n; ++i) {
        cin >> d[i];
    }
    vector<ll> tl(n), tr(n);
    for (int i = 0; i < n; ++i) {
        if (d[i] == 0) {
            tl[i] = a[i];
            tr[i] = a[i] + base;
        } else {
            tr[i] = base - a[i];
            tl[i] = tr[i] + base;
        }
    }
    sort(tl.begin(), tl.end());
    sort(tr.begin(), tr.end());
    ll ta = 0, tb = 0;
    if (A % n == 0) {
        ta = 2 * base * (A / n - 1) + tl.back();
    } else {
        ta = 2 * base * (A / n) + tl[A % n - 1];
    }
    if (B % n == 0) {
        tb = 2 * base * (B / n - 1) + tr.back();
    } else {
        tb = 2 * base * (B / n) + tr[B % n - 1];
    }
    auto calc = [&](ll b, ll fst) -> ll {
        ll v = (b - fst + 2 * base - 1) / (2 * base);
        return v * 2 * base + fst;
    };
    ll ans = 0;
    for (int i = 0; i < n; ++i) {
        if (d[i] == 0) {
            gmax(ans, min(calc(ta+1, a[i]), calc(tb+1, a[i]+base)));
        } else {
            gmax(ans, min(calc(ta+1, 2*base-a[i]), calc(tb+1, base-a[i])));
        }
    }
    cout << ans << "\n";
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

int main()
{
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    ll n, A, B, base=1e9+1;
    cin >> n >> A >> B;
    vector<ll> a(n), d(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    for (int i = 0; i < n; ++i) {
        cin >> d[i];
    }
    ll ans = 0;
    ll t = min(A, B) / n;
    A -= t * n;
    B -= t * n;
    ans += 2 * t * base;
    vector<array<ll, 3>> q;
    for (int i = 0; i < n; ++i) {
        if (d[i] == 0) {
            q.push_back({ans+a[i], i, 0});
            q.push_back({ans+a[i]+base, i, 1});
            q.push_back({ans+a[i]+2*base, i, 0});
            q.push_back({ans+a[i]+3*base, i, 1});
        } else {
            q.push_back({ans+base-a[i], i, 1});
            q.push_back({ans+2*base-a[i], i, 0});
            q.push_back({ans+3*base-a[i], i, 1});
            q.push_back({ans+4*base-a[i], i, 0});
        }
    }
    sort(q.begin(), q.end());
    vector<int> drop(n);
    for (auto [tm, id, b]: q) {
        if (drop[id]) {
            continue;
        }
        gmax(ans, tm);
        ll &cnt = (b == 0 ? A : B);
        if (cnt == 0) {
            drop[id] = 1;
        } else {
            --cnt;
        }
    }
    cout << ans << "\n";
}
```
