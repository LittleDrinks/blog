## [CF2276D](https://codeforces.com/contest/2276/problem/D)

- 整体连通
    - 菊花图（有 $n-1$ 个叶子）时答案是 $2$
    - 否则答案是 `Many`
- 有不止一个连通块
    - 如果有孤点，答案是 `Brain Melt`
    - 否则答案是 `Many`

[***AC 代码***](https://codeforces.com/contest/2276/submission/396074546)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

struct DSU {
    int n;
    vector<int> f, e;
    DSU(int n): n(n), f(n+1), e(n+1) {
        iota(f.begin(), f.end(), 0);
    }
    int find(int x) {
        return f[x]==x? f[x]: f[x]=find(f[x]);
    }
    void merge(int x, int y) {
        x = find(x);
        y = find(y);
        e[x]++;
        if (x == y) return;
        e[y] += e[x];
        e[x] = 0;
        f[x] = y;
    }
    bool same(int x, int y) {
        return find(x) == find(y);
    }
};

void solve() {
    int n, m;
    cin >> n >> m;
    DSU D(n);
    int zero = 0;
    int many = 0;
    vector<int> deg(n+1);
    int connected = n;
    for (int i = 1; i <= m; ++i) {
        int u, v;
        cin >> u >> v;
        ++deg[u];
        ++deg[v];
        if (!D.same(u, v)) {
            D.merge(u, v);
            --connected;
        }
    }

    if (connected == 1) {
        int cnt = 0;
        for (int i = 1; i <= n; ++i) {
            cnt += (deg[i] == 1);
        }
        if (cnt == n-1) {
            cout << "2\n";
        } else {
            cout << "Many\n";
        }
    } else {
        vector<int> vis(n+1);
        int zero = 0;
        for (int i = 1; i <= n; ++i) {
            int x = D.find(i);
            if (!vis[x]) {
                vis[x] = 1;
                if (D.e[x] == 0) zero=1;
            }
        }
        if (zero) cout<<"Brain Melt\n";
        else cout << "Many\n";
    }
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
