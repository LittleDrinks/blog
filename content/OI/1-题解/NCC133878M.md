## [NCC133878M]()

最直接的思路是，设 $f(u\to v)$ 表示上一步在 $u$，现在在 $v$，走到 $t$ 的期望时间。根据定义 $f(\cdot,t)=0$。特别地，我们规定 $f(t,\cdot)>0$。一共 $O(n)$ 个状态，直接高斯消元可以做到 $O(n^3)$
接下来考虑根据树的结构直接求解方程。

> [!tip] Observation
> 以 $t$ 为根，如果我选择向下走，一定是走到底再回头

据此我们可以得到，如果 $L$ 是叶子，那么 $f(p,L)=\underbrace{1}_{向下走一步}+\underbrace{f(L,p)}_{走回来}$
尝试推广这个结论，我们发现可以构建出一个节点附近的关系：
$$
f(u,v)=E(v)+f(v,u)
$$
其中 $E(v)$ 表示从父节点走入这棵子树，当前在 $v$，最终走到了父节点 $u$ 的期望步数。
直接求解比较困难。

> [!tip] 
> 我们尝试利用随机游走中的对称性

对于 $u$ 来说，往每个方向走的概率都是一样的。从子树中走回 $u$ 后，决策都是：以 ${1}/{(\deg(u)-1)}$ 的概率走到父亲，以相同的概率走向每一个儿子。
我们转而计算“从 $u$ 向子树走的次数”，发现其满足 $p=\frac{1}{(\deg(u)-1)}$ 的几何分布，期望次数为 $\deg(u)-1$，又因为随机游走规定了每个儿子被选择的概率都是一样的，于是 $\deg(u)-1$ 被均分到每个儿子身上，期望意义下平均每个子节点访问一次。
$$
E(u)=1+\sum_i [1+E(v_i)]
$$
对叶子 $L$ 满足 $E(L)=1$。可以进一步解得 $E(u)=2siz(u)-1$，但这是不必要的。
至此我们构建了 $f(u\to v)$ 和 $f(v\to u)$ 的关系，接下来利用这个关系求解方程

> [!tip] Observation
> $f(\cdot,t)=0$，以此为起点向下递推即可

我们考虑在 $f(u\to p)$ 已知的情况下推导 $f(v_i\to u)$，也就是维护走向父节点之后的期望步数
容易写出
$$
f(v_i\to u)=\underbrace{1}_{先走到u}+\frac{1}{k}\Bigg(\underbrace{\sum_{j\ne i}[f(u\to v_j)]}_{随机走向子树内}+\underbrace{f(u\to p)}_{走出子树后}\Bigg)
$$
将 $f(u\to v_j)$ 展开得到
$$
f(v_i\to u)=1+\frac{1}{k}\sum_{j\ne i}[f(v_j\to u)+E(v_j)]+\frac{1}{k}f(u\to p)
$$
记 $S=\sum_i [f(v_i\to u)]$，有
$$
f(p\to u)=1+\frac{S+\sum_i E(v_i)}{k}
$$
两边加上一个 $\frac{f(v_i\to u)}{k}$ 得到
$$
\begin{align}
\frac{k+1}{k}f(v_i\to u)=\Bigg(1+\frac{S}{k}+\frac{1}{k}\sum_iE(v_i)\Bigg)\underbrace{-\frac{1}{k}\sum_iE(v_i)}_{补一项凑f(u\to p)}+\frac{1}{k}\sum_{j\ne i}[E(v_j)]+\frac{1}{k}f(u\to p) \\
=\frac{k+1}{k}f(u\to p)-\frac{1}{k}\sum_iE(v_i)+E(u)+\frac{1}{k}\sum_{j\ne i}[E(v_j)]
\end{align}
$$
整理得到
$$
f(v_i\to u)=f(u\to p)+\frac{kE(u)-E(v_i)}{k+1}
$$
从上往下递推即可得到答案

------

最终答案需要再讨论一下，在起点 $s$ 以 $\frac{1}{\deg(s)}$ 的概率往外走，期望步数是
$$
\underbrace{1}_{先走一步}+\underbrace{f(s\to p)}_{肯定要往根走}+\underbrace{\frac{k+1}{k}[E(s)-1]}_{有概率往子树走}
$$

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int MOD = 998244353;
ll add(ll x, ll y) { return (x + y) % MOD; }
ll del(ll x, ll y) { return (x - y + MOD) % MOD; }
ll mul(ll x, ll y) { return (x * y) % MOD; }
ll qpow(ll a, ll b = MOD - 2) {
    ll res = 1;
    for (; b; b >>= 1, a = mul(a, a)) {
        if (b & 1) {
            res = mul(res, a);
        }
    }
    return res;
}

void solve() {
    int n, s, t;
    cin >> n >> s >> t;
    vector G(n + 1, vector<int>());
    for (int i = 2; i <= n; ++i) {
        int p;
        cin >> p;
        G[i].push_back(p);
        G[p].push_back(i);
    }

    vector<int> siz(n + 1, 1);
    vector<int> p(n + 1);
    vector<ll> e(n + 1, 1);
    auto dfs1 = [&](auto &&dfs1, int u, int fa) -> void {
        p[u] = fa;
        for (auto v: G[u]) {
            if (v == fa) continue;
            dfs1(dfs1, v, u);
            siz[u] += siz[v];
            e[u] = add(e[u], 1+e[v]);
        }
    };
    dfs1(dfs1, t, 0);

    vector<ll> f(n + 1);
    auto dfs2 = [&](auto &&dfs2, int u, int fa) -> void {
        int k = G[u].size()-1;
        for (auto v: G[u]) {
            if (v == fa) continue;
            f[v] = add(f[u], mul(del(mul(k,e[u]), e[v]), qpow(k+1)));
            dfs2(dfs2, v, u);
        }
    };
    for (auto v: G[t]) {
        f[v] = 0;
        dfs2(dfs2, v, t);
    }

    if (s == t) {
        cout << "0\n";
    } else {
        ll ans = add(1, f[s]);
        int k = G[s].size() - 1;
        ans = add(ans, mul(mul(k, del(e[s], 1)), qpow(k+1)));
        cout << ans << "\n";
    }

}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
