## [NCC133881I]()

看到 popcount，联想不到什么好的性质，于是考虑拆位
不妨设此时 $i$ 低位放了 $A$ 个 $1$，$i+d$ 低位有 $B$ 个 $1$，枚举这一位放的是数字 $r$，对应 $i+d$ 是 $p$，则
$$
ANS=(A+r)(B+p)=AB+rB+Ap+rp
$$
枚举低位的所有情况，则有：
$$
\sum A_i\cdot B_i=\sum A_{i-1}\cdot B_{i-1}+\sum B_{i-1}+\sum A_{i-1}+\sum 1_{i-1}
$$
也就是说我们可以通过维护 $i$ 的 popcount 和、$i+d$ 的 popcount 和、以及有多少符合条件的数字，来递推得到 $f(i)\cdot f(i+d)$ 
时间复杂度 $O(T\log n)$
实现时需要按照是否顶满、是否进位额外对状态划分。这里记录一些对状态设计的理解：
- 采用记忆化搜索时，$limit$ 表示高位是否顶满，$carry$ 表示强制低位有/无进位
- 采用递推时，$limit$ 表示是否允许高位顶满，$carry$ 表示是否产生进位
- 也可以将 carry 立即为高位是否会向低位借位，这样就可以去掉 $limit$ 了

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int MOD = 998244353;
ll add(ll x, ll y) { return (x + y) % MOD; }
ll mul(ll x, ll y) { return (x * y) % MOD; }

const int I = 60;
ll n, d;
int a[I+5], b[I+5];
bool vis[I+5][2][2];
ll num[I+5][2][2], suma[I+5][2][2], sumb[I+5][2][2], prod[I+5][2][2];

void dfs(int u, int c, int up) {
    if (vis[u][c][up]) {
        return;
    }
    vis[u][c][up] = 1;
    if (u == 0) {
        num[u][c][up] = (c == 0);
        return;
    }
    for (int x = 0; x <= (up ? a[u]: 1); ++x) {
        int y = x + b[u];
        for (int pc = 0; pc <= 1; ++pc) {
            if (c != (y + pc >= 2)) {
                continue;
            }
            dfs(u-1, pc, (up&&(a[u]==x)));
            num[u][c][up] = add(num[u][c][up], num[u-1][pc][up&&(a[u]==x)]);
            suma[u][c][up] = add(suma[u][c][up], suma[u-1][pc][up&&(a[u]==x)]);
            sumb[u][c][up] = add(sumb[u][c][up], sumb[u-1][pc][up&&(a[u]==x)]);
            prod[u][c][up] = add(prod[u][c][up], prod[u-1][pc][up&&(a[u]==x)]);
            if (x == 1) {
                suma[u][c][up] = add(suma[u][c][up], num[u-1][pc][up&&(a[u]==x)]);
                prod[u][c][up] = add(prod[u][c][up], sumb[u-1][pc][up&&(a[u]==x)]);
            }
            if ((y+pc) & 1) {
                sumb[u][c][up] = add(sumb[u][c][up], num[u-1][pc][up&&(a[u]==x)]);
                prod[u][c][up] = add(prod[u][c][up], suma[u-1][pc][up&&(a[u]==x)]);
            }
            if ((x == 1) && ((y+pc)&1)) {
                prod[u][c][up] = add(prod[u][c][up], num[u-1][pc][up&&(a[u]==x)]);
            }
        }
    }
}

void solve() {
    memset(vis, 0, sizeof(vis));
    memset(num, 0, sizeof(num));
    memset(suma, 0, sizeof(suma));
    memset(sumb, 0, sizeof(sumb));
    memset(prod, 0, sizeof(prod));
    cin >> n >> d;
    for (int i = 0; i < I; ++i) {
        a[i+1] = n>>i&1;
        b[i+1] = d>>i&1;
    }
    dfs(I+1, 0, 1);
    cout << prod[I+1][0][1] << "\n";
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
