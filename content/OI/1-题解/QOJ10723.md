## [QOJ10723. Outer LIS](https://qoj.ac/contest/2021/problem/10723)

设 $f(i)$ 表示以 $i$ 结尾的 LIS 最大权值，$g(i)$ 表示以 $i$ 开头的 LIS 最大权值。
对前缀和后缀分别做一次【带权LIS】，就可以处理单次询问。
使用分块处理多测。记 $bk(i)$ 表示下标为 $i$ 的位置所属的块，$B$ 为块数。特别地，我们记 $0,B+1$ 为两个空块，以处理一些边界情况。
对整块，我们处理：

- $pre(i,v)$ 表示整块 $[0,i]$，最后一位 $a\le v$ 时的最大权值
- $suf(i,v)$ 表示整块 $[i,B+1]$，最后一位 $a\ge v$ 的最大权值。

答案分为两类：

- 整块 $[0,bk(l))$ 对整块 $(bk(r),B+1]$
- 至少有一个元素在 $l,r$ 所在的散块内

我们可以预处理出所有第一类答案。记 $ans(i,j)$ 表示整块 $[0,i]$ 中的元素和整块 $[j,B+1]$ 中的元素拼接所得到的最大权值，我们枚举 $i$，然后从后往前遍历每一个位置 $pos$，用 $pre(i,a_{pos})+w_{pos}$ 更新 $ans(i,bk(pos))$，时间复杂度 $O(Bn)=O(B^3)$
每次询问，我们实时求出第二类答案。将所有散块内的位置拉出来，按照 $a$ 从小到大排序，让 $a$ 大的 $r$ 散块匹配 $a$ 小的 $l$ 散块，并且同时更新散块对整块的答案，时间复杂度 $O(qB\log B)$

[***AC 代码***](https://qoj.ac/submission/2209419)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ y = min(x, y); }

struct BIT {
    int n;
    vector<int> t;
    BIT(int n): n(n), t(n + 1) { }
    int lowbit(int x) { return x & -x; }
    void modify(int x, int d) {
        for (; x <= n; x += lowbit(x)) { gmax(t[x], d); }
    }
    int query(int x) {
        int res = 0;
        for (; x; x -= lowbit(x)) { gmax(res, t[x]); }
        return res;
    }
};

const int N = 1e5 + 5, S = 500;
int n, q, siz, B;
int a[N], w[N], f[N], g[N], pre[S][N], suf[S][N], ans[S][S];

// arr 1-index <-> bk 1-index
inline int bk(int i) { return i / siz + 1; }
inline int bkl(int b) { return max(1, (b - 1) * siz); }
inline int bkr(int b) { return min(b * siz - 1, n); }

int main()
{
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    cin >> n >> q;
    siz = sqrt(n) + 1;
    B = bk(n);
    for (int i = 1; i <= n; ++i) {
        cin >> a[i] >> w[i];
    }

    // 预处理前缀的答案
    {
        BIT t(n);
        for (int i = 1; i <= n; ++i) {
            f[i] = t.query(a[i]) + w[i];
            t.modify(a[i], f[i]);
        }
        for (int b = 1; b <= B; ++b) {
            for (int i = 1; i <= n; ++i) {
                pre[b][i] = pre[b - 1][i];
            }
            for (int i = bkl(b); i <= bkr(b); ++i) {
                gmax(pre[b][a[i]], f[i]);
            }
            for (int i = 1; i <= n; ++i) {
                gmax(pre[b][i], pre[b][i - 1]);
            }
        }
    }

    // 预处理后缀的答案
    {
        BIT t(n);
        for (int i = n; i; --i) {
            g[i] = t.query(n-a[i]+1) + w[i];
            t.modify(n-a[i]+1, g[i]);
        }
        for (int b = B; b; --b) {
            for (int i = 1; i <= n; ++i) {
                suf[b][i] = suf[b + 1][i];
            }
            for (int i = bkl(b); i <= bkr(b); ++i) {
                gmax(suf[b][a[i]], g[i]);
            }
            for (int i = n; i; --i) {
                gmax(suf[b][i], suf[b][i + 1]);
            }
        }
    }

    // 预处理块对块的答案
    for (int i = 1; i <= B; ++i) {
        ans[0][i] = suf[i][1];
        ans[i][B+1] = pre[i][n];
    }
    for (int i = 1; i <= B; ++i) {
        for (int j = B; j >= i; --j) {
            ans[i][j] = ans[i][j+1];
            for (int p = bkl(j); p <= bkr(j); ++p) {
                gmax(ans[i][j], pre[i][a[p]] + g[p]);
            }
        }
    }

    while (q--) {
        int l, r;
        cin >> l >> r;
        int res = 0;
        gmax(res, ans[bk(l)-1][bk(r)+1]);
        vector<pair<int,int>> v;
        v.reserve(2*B);
        for (int i = bkl(bk(l)); i < l; ++i) {
            v.emplace_back(a[i], i);
        }
        for (int i = r + 1; i <= bkr(bk(r)); ++i) {
            v.emplace_back(a[i], i);
        }
        sort(v.begin(), v.end());
        int mx = 0;
        for (auto &[val, id]: v) {
            if (id <= l) {
                gmax(mx, f[id]);
                gmax(res, f[id] + suf[bk(r)+1][val]);
            } else {
                gmax(res, mx + g[id]);
                gmax(res, pre[bk(l)-1][val] + g[id]);
            }
        }
        cout << res << "\n";
    }
}
```
