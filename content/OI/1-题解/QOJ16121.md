## [QOJ16121. Intervals from Triplets](https://qoj.ac/contest/2828/problem/16121)

观察一些 corner case：

- 如果 $b$ 落在别的区间内部一定不合法
- 如果有两个 $b$ 相等，一定是一左一右
- 如果三个 $b$ 相等，一定不合法


这启发我们按照 $b$ 排序，这样一来最后的区间一定是按照 $b$ 的顺序来的。当然，相邻两个相等的 $b$ 是可以互换位置的。
DP，设 $dp(i, 0/1)$ 表示考虑到第 $i$ 条线段，右端点记号为 $0/1$ 时的最长长度。$dp(0,0/1)=0$。
如果 $b_{i-1}\ne b_i$，那么 $dp(i,0/1)$ 表示第 $i$ 个区间的右端点选为 $b/c$。
如果 $b_{i-1}=b_i$，那么他们的顺序是可以交换的，此时设 $dp(i,j)$ 表示第 $i$ 个区间的右端点选为 $c_{i-1+j}$
转移时根据状态，找到上一条线段的右端点，判断是否在当前左端点左侧，如果是，就加上现在这段的长度即可。

[***AC 代码***](https://qoj.ac/submission/2370477)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int inf = 1e9;

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int n;
    cin >> n;
    vector<array<int, 3>> seg(n + 1);
    for (int i = 1; i <= n; ++i) {
        auto &[a, b, c] = seg[i];
        cin >> a >> b >> c;
    }
    sort(seg.begin(), seg.end(), [](auto x, auto y){
        return x[1] < y[1];
    });
    vector dp(n + 1, vector<int>(2, -inf));
    dp[0][0] = dp[0][1] = 0;
    for (int i = 1; i <= n; ++i) {
        auto [a, b, c] = seg[i];
        if (i >= 3 && b == seg[i-1][1] && b == seg[i-2][1]) {
            cout << "-1\n";
            return 0;
        }

        if (i + 1 <= n && b == seg[i + 1][1]) {
            continue;
        }

        for (int j: {0, 1}) {
            // b=b
            if (seg[i-1][1] == b) {
                int pre;
                if (i >= 3 && seg[i - 2][1] == seg[i - 3][1]) {
                    pre = seg[i - 3 + j][2];
                } else {
                    pre = seg[i - 2][j + 1];
                } 
                if (pre <= seg[i - 1][0]) {
                    gmax(dp[i][1], dp[i-2][j] + seg[i][2] - seg[i-1][0]);
                }
                if (pre <= seg[i][0]) {
                    gmax(dp[i][0], dp[i-2][j] + seg[i-1][2] - seg[i][0]);
                }
            } else {
                int pre;
                if (i >= 2 && seg[i-1][1] == seg[i-2][1]) {
                    pre = seg[i - 2 + j][2];
                } else {
                    pre = seg[i - 1][j + 1];
                }
                // ab, ac
                if (pre <= a) {
                    gmax(dp[i][0], dp[i-1][j] + b - a);
                    gmax(dp[i][1], dp[i-1][j] + c - a);
                }
                // bc
                if (pre <= b) {
                    gmax(dp[i][1], dp[i-1][j] + c - b);
                }
            }
        }
    }

    int ans = max(dp[n][0], dp[n][1]);
    cout << (ans < 0 ? -1: ans) << "\n";
}
```
