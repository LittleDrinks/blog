## [NCC133879F. 23 Subsequences](https://ac.nowcoder.com/acm/contest/133879/F)

单次询问的 DP 是倒着做，$dp(i)=LEN$ 表示当前位置向后的最长合法序列长度。
$$
dp(i) = 1 + \max_{i+1\le j\le n, 2a_i\le a_j\le 3a_i} \{dp(j)\}
$$
这道题的痛点在于区间不具有可加性，没有办法直接上数据结构莽过去，于是我们从 DP 入手加以改进
注意到权值每次至少扩大两倍，因此序列长度不超过 $60$
反表示状态，把答案塞到状态里，设 $dp(i,L)=pos$ 表示 $i$ 位置往后长度为 $L$ 的合法序列，最后一个元素下标最早可以在哪里
转移变为：
$$
\begin{cases}
dp(i,1)=i \\
dp(i,L)=\min_{i+1\le j\le n,2a_i\le a_j\le 3a_i} dp(j,L-1)
\end{cases}
$$
询问时，我们从大到小枚举答案 $L$，判断 $\min_{i=l}^r dp(i,L)\le r$ 是否成立即可。
状态转移可以在 $O(n\log n\log V)$ 的时间内用权值线段树维护，求出 $dp$ 后取后缀 min 即可 $O(q\log V)$ 直接处理询问。

[***AC 代码***](https://ac.nowcoder.com/acm/contest/view-submission?submissionId=84274065)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

const int N = 6e5 + 5, I = 65;
const int inf = 1e9;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
ostream& operator<<(ostream& os, const array<int,I+5>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }


#define m ((l+r)>>1)
#define lson (p << 1)
#define rson (p << 1 | 1)
array<int,I+5> a[N<<2];
void pushup(int p, int l, int r) {
    for (int i = 0; i < I; ++i) {
        a[p][i] = min(a[lson][i], a[rson][i]);
    }
}
void build(int p, int l, int r) {
    if (l == r - 1) {
        fill(a[p].begin(), a[p].end(), inf);
        return;
    }
    build(lson, l, m);
    build(rson, m, r);
    pushup(p, l, r);
}
void modify(int p, int l, int r, int x, array<int,I+5> v) {
    if (r <= x || x+1 <= l) {
        return;
    }

    if (x <= l && r <= x+1) {
        a[p] = v;
        return;
    }

    modify(lson, l, m, x, v);
    modify(rson, m, r, x, v);
    pushup(p, l, r);
}
array<int,I+5> query(int p, int l, int r, int x, int y) {
    if (y <= l || r <= x) {
        array<int,I+5> nd;
        fill(nd.begin(), nd.end(), inf);
        return nd;
    }

    if (x <= l && r <= y) {
        return a[p];
    }

    array<int,I+5> LA = query(lson, l, m, x, y);
    array<int,I+5> RA = query(rson, m, r, x, y);
    array<int,I+5> res;
    for (int i = 0; i < I; ++i) {
        res[i] = min(LA[i], RA[i]);
    }
    pushup(p, l, r);
    return res;
}
#undef m
#undef lson
#undef rson

array<int,I+5> dp[N];

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int n, q;
    cin >> n >> q;
    
    vector<ll> a(n + 1);
    vector<ll> P;
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
        P.push_back(a[i]);
        P.push_back(2*a[i]);
        P.push_back(3*a[i]);
    }

    sort(P.begin(), P.end());
    P.erase(unique(P.begin(), P.end()), P.end());
    
    auto get_id = [&](ll v) {
        return lower_bound(P.begin(), P.end(), v) - P.begin();
    };

    int V = P.size();

    build(1, 0, V);

    for (int i = 1; i <= n; ++i) {
        fill(dp[i].begin(), dp[i].end(), inf);
    }

    dp[n][1] = n;
    modify(1, 0, V, get_id(a[n]), dp[n]);

    for (int i = n-1; i; --i) {
        int vl = get_id(2 * a[i]);
        int vr = get_id(3 * a[i]);
        auto res = query(1, 0, V, vl, vr+1);
        dp[i][1] = i;
        for (int j = 2; j < I; ++j) {
            dp[i][j] = res[j-1];
        }
        int vp = get_id(a[i]);
        modify(1, 0, V, vp, dp[i]);
    }

    for (int i = n-1; i; --i) {
        for (int j = 0; j < I; ++j) {
            gmin(dp[i][j], dp[i+1][j]);
        }
    }

    while (q--) {
        int l, r;
        cin >> l >> r;
        int ans;
        for (ans = I-1; ans; --ans) {
            if (dp[l][ans] <= r) {
                break;
            }
        }
        cout << ans << "\n";
    }
}
```
