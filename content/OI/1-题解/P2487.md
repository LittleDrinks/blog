## [P2487 \[SDOI2011\] 拦截导弹](https://www.luogu.com.cn/problem/P2487)

根据 Dilworth 定理，只需要求二维最长不升子序列长度，就可以解决第一问
设 $f(i)$ 表示以 $i$ 结尾的 LDS 长度，我们在 DP 的过程中顺手维护出对应的方案数 $fc(i)$。总的拦截方案数为：
$$
T = \sum_{i,f(i)=\max f} fc(i)
$$
正反各做一遍，设为 $f,fc,g,gc$，对于位置 $i$，如果
$$
f(i)+g(i)-1=\max f
$$
说明可以被拦截，强制拦截该枚导弹的方案数为
$$
fc(i)\times gc(i)
$$
相乘求出概率即可。

---

二维的 LDS，相当于同时包含 $i<j$，$h_i\ge h_j$，$v_i\ge v_j$ 三个条件，这是一个经典的三位偏序 DP，用 cdq 分治加树状数组维护一下即可。
时间复杂度 $O(n\log^2n)$

[***AC 代码***](https://www.luogu.com.cn/record/295189797)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int inf = 1e9;

template<typename T>
struct BIT {
    int n;
    vector<T> t;
    BIT(int n): n(n), t(n+1, T()) { }
    int lowbit(int x) { 
        return x & -x;
    }
    void modify(int x, T d) {
        for (; x <= n; x += lowbit(x)) t[x] = t[x] + d;
    }
    void reset(int x) {
        for (; x <= n; x += lowbit(x)) t[x] = T();
    }
    T query(int x) {
        T res = T();
        for (; x; x -= lowbit(x)) res = res + t[x];
        return res;
    }
};

struct node {
    int mx; db cnt;
    node(int mx=-inf, db cnt=0): mx(mx), cnt(cnt) { }
};
node operator+(const node &a, const node &b) {
    if (a.mx != b.mx) return a.mx > b.mx ? a: b;
    return node(a.mx, a.cnt+b.cnt);
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int n;
    cin >> n;
    vector<array<int,3>> a(n);
    for (int i = 0; i < n; ++i) {
        auto &[id, h, v] = a[i];
        id = i + 1;
        cin >> h >> v;
        h = -h;
        v = -v;
    }
    
    auto work = [&]() -> vector<node> {
        vector<int> val;
        for (int i = 0; i < n; ++i) {
            val.push_back(a[i][2]);
        }
        sort(val.begin(), val.end());
        val.erase(unique(val.begin(), val.end()), val.end());
        auto gid = [&](int x) {
            auto it = lower_bound(val.begin(), val.end(), x);
            assert(*it == x);
            return it - val.begin() + 1;
        };
        vector<node> f(n+1, node(1,1));
        f[1] = node(1, 1);BIT<node> t(val.size()+1);
        auto cdq = [&](auto &&cdq, int l, int r) -> void {
            if (l == r - 1) {
                return;
            }
            int m = (l + r) >> 1;
            cdq(cdq, l, m);
            sort(a.begin()+l, a.begin()+m, [&](auto x, auto y){ return x[1]<y[1]; });
            sort(a.begin()+m, a.begin()+r, [&](auto x, auto y){ return x[1]<y[1]; });
            int i=l, j=m;
            for (; j < r; ++j) {
                for (; i < m && a[i][1] <= a[j][1]; ++i) {
                    t.modify(gid(a[i][2]), f[a[i][0]]);
                }
                auto info = t.query(gid(a[j][2]));
                f[a[j][0]] = f[a[j][0]] + node(info.mx+1, info.cnt);
            }
            for (int k = l; k < i; ++k) {
                t.reset(gid(a[k][2]));
            }
            sort(a.begin()+m, a.begin()+r);
            cdq(cdq, m, r);
        };
        cdq(cdq, 0, n);
        return f;
    };
    
    sort(a.begin(), a.end());
    auto f = work();
    for (int i = 0; i < n; ++i) {
        a[i] = {n-a[i][0]+1, -a[i][1], -a[i][2]};
    }
    sort(a.begin(), a.end());
    auto g = work();
    reverse(g.begin()+1, g.end());
    
    node lis;
    for (int i = 1; i <= n; ++i) {
        lis = lis + f[i];
    }
    cout << lis.mx << "\n";
    for (int i = 1; i <= n; ++i) {
        db p = 0;
        if (f[i].mx + g[i].mx - 1 == lis.mx) {
            p = f[i].cnt * g[i].cnt / lis.cnt;
        }
        cout << fixed << setprecision(5) << p << " \n"[i == n];
    }
}
```
