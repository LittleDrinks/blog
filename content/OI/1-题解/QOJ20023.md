## [QOJ20023. The First Problem](https://qoj.ac/contest/4071/problem/20023)

对 $T$ 建后缀连接树

> [!tip] 性质
> 后缀连接树上沿着 link 往下走，相当于在前面加一个字符

设字符串在后缀连接树上对应的节点是 $tr(\cdot)$
如果 $S$ 在 $T[x-|S|+1, x]$ 出现了，也就是 $pref(T,x)$ 由 $S$ 结尾，或者说 $x\in endpos(S)$
那么根据上述性质，$tr(pref(T,x))$ 这棵子树内的所有节点都只是在前面加若干个字符，于是都会以 $S$ 结尾。
更进一步，$pref(T,x)$ 以 $S$ 结尾，当且仅当 $tr(pref(T,x))\in subtree(tr(S))$。因为从 $tr(S)$ 往上走就要从 $S$ 里删字符了

> [!tip] 如何还原出所有的 $endpos(S)$？

在后缀连接树上的每个节点，都记上他是某个前缀 $pref(T,i)$ 对应的节点
对于 $tr(S)$，收集其子树内的所有 $i$，就可以还原出 $endpos$
反过来，想要判断 $i$ 是 $endpos(S)$ 之一，只需要判断 $tr(pref(S,i))\in subtree(tr(S))$ 即可

> [!faq] 题面转化
> 如何求出 $occur(S,pref(T,i))$？

对于某一个字符串 $S$，如果他以 $x$ 作为一个结束位置，那么它会对 $pref(T,x),\ldots pref(T,n)$ 的 $occur(S,\cdot)$ 都会产生贡献。
于是可以将 $\sum occur(S,pref(T,i))$ 转化为：
$$
\sum_{i=1}^{r} occur(S,pref(T,i))=\sum_{x\in endpos(S), x\le r} (r-x+1)
$$
可以进一步转化为
$$
cnt*(r+1)-sum
$$
其中 $cnt$ 表示出现次数，$sum$ 表示出现位置之和
至此，通过对 $S$ 的每一个前缀计算，我们可以求出 $\sum score$ 并进一步二分求解，接下来考虑如何计算信息。

> [!faq] 题面转化
> $x\in endpos(S)$ 可以转化为后缀连接树上的子树信息
> $x\le r$ 是另外一重限制
> 如何在这两个限制条件下维护区间 $cnt$ 和 $sum$？

这本质上是一个偏序问题
$x\le r$ 可以开线段树，线段树下标对应 $T$ 的下标，统计出现次数 $cnt$ 和出现位置之和 $sum$
$x\in endpos(S)$ 可以通过 dfs 序解决，也就是 $in(tr(S))\le in(x)\le out(tr(S))$，这可以通过历史版本维护
所以我们开一棵主席树，对一个前缀 $pref(T,i)$，在 $dfn(tr(pref(T,i)))$ 这个版本，在 $i$ 位置上更新出现次数和出现位置之和。
最后主席树上整体二分即可。对于 $\{S\}$ 的所有前缀 $s$，可以 $O(\sum|S|)$ 算出所有 $[in(tr(s)),out(tr(s))]$ 版本、$[l,r)$ 上的 $cnt$ 和 $sum$，如果大于 $b$ 就所有区间递归到左儿子 $[l,m)$，否则递归到右儿子 $[m,r)$

/**   后缀自动机（SAM 新版）
 *    2023-05-27: https://cf.dianhsu.com/gym/104353/submission/207318083
 *    2023-09-25: https://qoj.ac/submission/188106
 *    2024-04-09: https://qoj.ac/submission/384423 【模板】

struct SAM {
    static constexpr int ALPHABET_SIZE = 26;
    struct Node {
        int len;
        int link;
        std::array<int, ALPHABET_SIZE> next;
        Node() : len{}, link{}, next{} {}
    };
    std::vector<Node> t;
    SAM() {
        init();
    }
    void init() {
        t.assign(2, Node());
        t[0].next.fill(1);
        t[0].len = -1;
    }
    int newNode() {
        t.emplace_back();
        return t.size() - 1;
    }
    int extend(int p, int c) {
        if (t[p].next[c]) {
            int q = t[p].next[c];
            if (t[q].len == t[p].len + 1) {
                return q;
            }
            int r = newNode();
            t[r].len = t[p].len + 1;
            t[r].link = t[q].link;
            t[r].next = t[q].next;
            t[q].link = r;
            while (t[p].next[c] == q) {
                t[p].next[c] = r;
                p = t[p].link;
            }
            return r;
        }
        int cur = newNode();
        t[cur].len = t[p].len + 1;
        while (!t[p].next[c]) {
            t[p].next[c] = cur;
            p = t[p].link;
        }
        t[cur].link = extend(p, c);
        return cur;
    }
    int extend(int p, char c, char offset = 'a') {
        return extend(p, c - offset);
    }
    
    int next(int p, int x) {
        return t[p].next[x];
    }
    
    int next(int p, char c, char offset = 'a') {
        return next(p, c - 'a');
    }
    
    int link(int p) {
        return t[p].link;
    }
    
    int len(int p) {
        return t[p].len;
    }
    
    int size() {
        return t.size();
    }
};

#define m ((l+r)>>1)
struct segmentTree {
    struct node {
        ll cnt, sum;
    };
    vector<node> a;
    vector<int> lson, rson;
    segmentTree(int cap): a(1), lson(1), rson(1) {
        a.reserve(cap);
        lson.reserve(cap);
        rson.reserve(cap);
    }
    int newNode(int p=0) {
        a.push_back(a[p]);
        lson.push_back(lson[p]);
        rson.push_back(rson[p]);
        return a.size()-1;
    }
    void pushup(int p) {
        a[p].cnt = a[lson[p]].cnt + a[rson[p]].cnt;
        a[p].sum = a[lson[p]].sum + a[rson[p]].sum;
    }
    int update(int p, int l, int r, int x) {
        p = newNode(p);
        if (x+1 <= l || r <= x) return p;
        if (l == r-1) { a[p].cnt++; a[p].sum+=x; return p; }
        lson[p] = update(lson[p], l, m, x);
        rson[p] = update(rson[p], m, r, x);
        pushup(p);
        return p;
    }
    int first(vector<pair<int,int>> &rgs, ll b, int l, int r) {
        ll cnt=0, sum=0;
        for (auto [x, y]: rgs) {
            cnt += a[y].cnt-a[x].cnt;
            sum += a[y].sum-a[x].sum;
        }
        if (r*cnt-sum < b) {
            return -1;
        }
        ll precnt=0, presum=0;
        while (l+1 < r) {
            ll c=0, s=0;
            for (auto [x, y]: rgs) {
                c += a[lson[y]].cnt - a[lson[x]].cnt;
                s += a[lson[y]].sum - a[lson[x]].sum;
            }
            if (m*(precnt+c)-(presum+s) >= b) {
                for (auto &[x, y]: rgs) {
                    x = lson[x];
                    y = lson[y];
                }
                r = m;
            } else {
                precnt += c;
                presum += s;
                for (auto &[x, y]: rgs) {
                    x = rson[x];
                    y = rson[y];
                }
                l = m;
            }
        }
        return l;
    }
};
#undef m

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    string T;
    cin >> T;
    SAM sam;
    int n = T.size();
    vector<int> get_tr_pos(n+1);
    for (int i = 1, cur = 1; i <= n; ++i) {
        cur = sam.extend(cur, T[i-1]);
        get_tr_pos[i] = cur;
    }
    int m = sam.size();

    vector G(m, vector<int>());
    for (int i = 2; i < m; ++i) {
        int p = sam.link(i);
        G[p].push_back(i);
    }

    int totdfn=0;
    vector<int> in(m), out(m);
    auto dfs = [&](auto &&dfs, int u) -> void {
        in[u] = ++totdfn;
        for (auto v: G[u]) {
            dfs(dfs, v);
        }
        out[u] = totdfn;
    };
    dfs(dfs, 1);

    vector<int> dfn_to_st_pos(m);
    for (int i = 1; i <= n; ++i) {
        dfn_to_st_pos[in[get_tr_pos[i]]] = i;
    }

    segmentTree t(n*20);
    vector<int> root(m);
    vector<pair<int, int>> rgs;
    for (int i = 1; i < m; ++i) {
        root[i] = root[i-1];
        if (dfn_to_st_pos[i]) {
            root[i] = t.update(root[i], 1, n+1, dfn_to_st_pos[i]);
        }
    }

    int q;
    cin >> q;
    while (q--) {
        int a;
        ll b;
        cin >> a >> b;
        vector<pair<int, int>> rgs;
        for (int i = 1; i <= a; ++i) {
            string s;
            cin >> s;
            int cur = 1;
            for (auto c: s) {
                cur = sam.next(cur, c);
                if (!cur) break;
                rgs.push_back({root[in[cur]-1], root[out[cur]]});
            }
        }
        cout << t.first(rgs, b, 1, n+1) << "\n";
    }
}
