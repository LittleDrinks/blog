## [QOJ16117. XOR-Excluding Sets](https://qoj.ac/contest/2828/problem/16117)

前置题：[[QOJ7679]]，异或和为零的子集数量为 $2^{n-r}$
反过来计数。坏子集可以先选一个数，然后在剩下的 $n-1$ 个数内选择一个异或和为 $0$ 的子集。
分类，一种是删数后秩下降，另一种是删点后秩不下降。
我们可以维护一个状态 $msk$，表示这个数是由哪几个基底组成的。如果插入一个数字，秩没有变化，那么这个 $msk$ 对应的那些基底就都是可以被代替的，删除他们，秩不会下降。
我们称那些删除后秩不会下降的数字数量为 $c$
如果删数后秩没有下降，那么相当于有一个大小为 $r$ 的基底和剩下 $n-1$ 个数字，方案数为
$$
c\cdot 2^{n-1-r}
$$
如果删数后秩下降了，那么相当于有一个大小为 $r-1$ 的基底和剩下 $n-1$ 个数字，方案数为
$$
(n-c)\cdot2^{n-1-(r-1)}=(n-c)\cdot2^{n-r}
$$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int MOD = 1e9 + 7;
ll add(ll x, ll y) { return (x + y) % MOD; }
ll del(ll x, ll y) { return (x - y + MOD) % MOD; }
ll mul(ll x, ll y) { return (x * y) % MOD; }
ll qpow(ll a, ll b = MOD - 2) {
    if (b < 0) return 0;
    ll res = 1;
    for (; b; b >>= 1, a = mul(a, a)) {
        if (b & 1) {
            res = mul(res, a);
        }
    }
    return res;
}

const int I = 64;
ull p[I], msk[I], cu;
int ord;
void insert(ull x) {
    ull cmsk = 0;
    for (int i = I - 1; i >= 0; --i) {
        if (x >> i & 1) {
            if (p[i] != 0) {
                x ^= p[i];
                cmsk ^= msk[i];
            } else {
                p[i] = x;
                msk[i] = cmsk ^ (1ULL << i);
                cu |= 1ULL << i;
                ++ord;
                return;
            }
        }
    }
    cu &= ~cmsk;
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int n;
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        ull x;
        cin >> x;
        insert(x);
        int ncu = popcount(cu);
        ll ans = mul(ncu, qpow(2, i-ord));
        ans = add(ans, mul(i-ncu, qpow(2, i-1-ord)));
        cout << del(qpow(2, i), ans) << "\n";
    }
}
```
