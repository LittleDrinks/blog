## [QOJ15324. No more regrets](https://qoj.ac/contest/2908/problem/Q/statement/zh_cn)

分块，散块直接暴力修改，整块维护 `tag` 和 `add` 两个懒标记
每次带着前缀的 $(pmx,pmn)$ 进入，先二分找到第一个大于 $pmx$ 的点 $x$ 和第一个小于 $pmn$ 的点 $y$，根据 $x,y$ 的相对关系讨论，分为三段累加
时间复杂度 $O(\sqrt{n}\log n)$

[***AC 代码***](https://qoj.ac/submission/3163292)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;
using ull = unsigned long long;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

int n, B, siz;
inline int bk(int i) { return i / siz + 1; }
inline int bkl(int b) { return max(1, (b - 1) * siz); }
inline int bkr(int b) { return min(b * siz - 1, n); }

const ll inf = 2e9;

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int q;
    cin >> n >> q;
    siz = sqrt(n) + 1;
    B = bk(n);
    
    vector<ll> a(n+1);
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }

    vector<ll> add(B+1), tag(B+1,inf);
    auto pushdown = [&](int b) {
        for (int i = bkl(b); i <= bkr(b); ++i) {
            if (tag[b]!=inf) a[i] = tag[b];
            if (add[b]) a[i] += add[b];
        }
        add[b] = 0;
        tag[b] = inf;
    };

    vector mx(B+1, vector<ll>(siz+1, 0)), mn(B+1, vector<ll>(siz+1, inf));
    vector sx(B+1, vector<ull>(siz+1)), sn(B+1, vector<ull>(siz+1)), su(B+1, vector<ull>(siz+1));
    auto build = [&](int b) {
        pushdown(b);
        mx[b][1] = mn[b][1] = a[bkl(b)];
        sx[b][1] = sn[b][1] = a[bkl(b)];
        su[b][1] = a[bkl(b)] * a[bkl(b)];
        for (int i = bkl(b)+1; i <= bkr(b); ++i) {
            int j = i - bkl(b) + 1;
            mx[b][j] = max(mx[b][j-1], a[i]);
            mn[b][j] = min(mn[b][j-1], a[i]);
            sx[b][j] = sx[b][j-1] + mx[b][j];
            sn[b][j] = sn[b][j-1] + mn[b][j];
            su[b][j] = su[b][j-1] + mx[b][j]*mn[b][j];
        }
    };
    for (int i = 1; i <= B; ++i) build(i);

    auto query = [&](int b, ll &pmx, ll &pmn) -> ull {
        int r = bkr(b)-bkl(b)+1;
        if (tag[b] != inf) {
            ll v = tag[b]+add[b];
            gmax(pmx, v);
            gmin(pmn, v);
            return (ull)r*pmx*pmn;
        }
        int x = upper_bound(mx[b].begin()+1, mx[b].begin()+r+1, pmx-add[b])-mx[b].begin();
        int y = upper_bound(mn[b].begin()+1, mn[b].begin()+r+1, pmn-add[b], greater<>())-mn[b].begin();
        ll res = 0;
        if (x <= y) {
            // [1, x): pmx, pmn
            res += pmx*pmn*(x-1);
            // [x, y): ..., pmn
            res += ((sx[b][y-1]-sx[b][x-1])+add[b]*(y-x)) * pmn;
            // [y, r]: ..., ...
            res += (su[b][r]-su[b][y-1]) + add[b]*(sx[b][r]-sx[b][y-1]) + add[b]*(sn[b][r]-sn[b][y-1]) + add[b]*add[b]*(r-y+1);
        } else {
            // [l, y): pmx, pmn
            res += pmx*pmn*(y-1);
            // [y, x): pmx, ...
            res += pmx * (sn[b][x-1]-sn[b][y-1]+add[b]*(x-y));
            // [x, r]: ..., ...
            res += (su[b][r]-su[b][x-1]) + add[b]*(sx[b][r]-sx[b][x-1]) + add[b]*(sn[b][r]-sn[b][x-1]) + add[b]*add[b]*(r-x+1);
        }
        gmax(pmx, mx[b][r]+add[b]);
        gmin(pmn, mn[b][r]+add[b]);
        return res;
    };

    while (q--) {
        int o;
        cin >> o;
        if (o == 1) {
            int l, r, v;
            cin >> l >> r >> v;
            for (int i = l; i <= r; ++i) {
                int b = bk(i);
                if (i == bkl(b) && bkr(b) <= r) {
                    add[b] += v;
                    i = bkr(b);
                } else {
                    if (add[b] || tag[b]!=inf) pushdown(b);
                    a[i] += v;
                }
            }
            build(bk(l));
            build(bk(r));
        } else if (o == 2) {
            int l, r, v;
            cin >> l >> r >> v;
            for (int i = l; i <= r; ++i) {
                int b = bk(i);
                if (i == bkl(b) && bkr(b) <= r) {
                    tag[b] = v;
                    add[b] = 0;
                    i = bkr(b);
                } else {
                    if (add[b] || tag[b]!=inf) pushdown(b);
                    a[i] = v;
                }
            }
            build(bk(l));
            build(bk(r));
        } else {
            int l, r;
            cin >> l >> r;
            ll pmx=0, pmn=inf;
            ull ans=0;
            for (int i = l; i <= r; ++i) {
                int b = bk(i);
                if (i == bkl(b) && bkr(b) <= r) {
                    ll c = query(b, pmx, pmn);
                    ans += c;
                    i = bkr(b);
                } else {
                    if (add[b] || tag[b]!=inf) build(b);
                    gmax(pmx, a[i]);
                    gmin(pmn, a[i]);
                    ans += pmx*pmn;
                }
            }
            cout << ans << "\n";
        }
    }
}
```
