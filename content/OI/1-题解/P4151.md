## [P4151 \[WC2011\] 最大 XOR 和路径](https://www.luogu.com.cn/problem/P4151)

任意一条路径的贡献，都可以被拆分为：【生成树上的路径异或和，加若干个环的异或和】
dfs，记沿着 dfs 树上路径走的异或值为 `path[u]`
遍历到已有的点时，将当前的异或值异或上 `path[u]`，就能得到这个环的异或值
把环插入线性基，`queryMax(path[n])` 即可

[***AC 代码***](https://www.luogu.com.cn/record/307628485)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

template<class T, int B>
struct LinearBasis {
    // T 使用有符号数, B < digits 避免 corner case
    // LinearBasis<ll, 62>

    array<T, B> a{};
    int tot = 0;
    bool dependent = false;

    void clear() {
        a.fill(0);
        tot = 0;
        dependent = false;
    }
    bool insert(T x) {
        assert(x >= 0 && x < (T(1)<<B));
        for (int i = B-1; i >= 0; --i) {
            if (!(x>>i&1)) continue;
            if (a[i]) {
                x ^= a[i];
                continue;
            }
            for (int j = i-1; j >= 0; --j) {
                if (a[j] && (x>>j&1)) x ^= a[j];
            }
            for (int j = i+1; j < B; ++j) {
                if (a[j]>>i&1) a[j] ^= x;
            }
            a[i] = x;
            ++tot;
            return true;
        }
        dependent = true;
        return false;
    }
    void merge(const LinearBasis &b) {
        bool dep = b.dependent;
        for (auto x: b.a) {
            if (x) insert(x);
        }
        dependent |= dep;
    }
    // 是否能由非空子集异或得到 x
    bool testNonEmpty(T x) const {
        T y = x;
        for (int i = B-1; i >= 0; --i) {
            if (y>>i&1) y ^= a[i];
        }
        return y == 0 && (x != 0 || dependent);
    }
    T queryMax(T x = 0) const {
        assert(x >= 0);
        for (int i = B-1; i >= 0; --i) {
            x = max(x, x^a[i]);
        }
        return x;
    }
    // 去重，包含 0，0-based 第 k 小
    T queryKthMin(T k) const {
        if (k < 0 || k >= (T(1)<<tot)) return -1;
        T res = 0;
        int j = 0;
        for (int i = 0; i < B; ++i) {
            if (!a[i]) continue;
            if (k>>j&1) res ^= a[i];
            ++j;
        }
        return res;
    }
    T queryKthMax(T k) const {
        T cnt = T(1)<<tot;
        if (k < 0 || k >= cnt) return -1;
        return queryKthMin(cnt-k-1);
    }
};

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int n, m;
    cin >> n >> m;
    vector G(n+1, vector<pair<int,ll>>());
    for (int i = 1; i <= m; ++i) {
        int u, v; ll w;
        cin >> u >> v >> w;
        G[u].emplace_back(v, w);
        G[v].emplace_back(u, w);
    }

    LinearBasis<ll,61> L;
    vector<ll> path(n+1, -1);
    auto dfs = [&](auto &&dfs, int u, int fa, ll X) -> void {
        path[u] = X;
        for (auto [v, w]: G[u]) {
            if (v == fa) continue;
            if (path[v] == -1) {
                dfs(dfs, v, u, X^w);
            } else {
                L.insert(X^w^path[v]);
            }
        }
    };

    dfs(dfs, 1, 0, 0);
    cout << L.queryMax(path[n]) << "\n";
}
```
