## [QOJ17345. Qenerals](https://qoj.ac/contest/3729/problem/17345)

我们称一开始的堡垒为“占领第 0 座堡垒”。
一定是按 $a$ 从小到大占领堡垒。
排序后枚举占领了多少个堡垒，确定占领的数量之后，最优的策略一定是尽早占领堡垒。
设 $t_i$ 表示最早什么时候占领第 $i$ 座堡垒， $s_i$ 表示占领 $i$ 座堡垒总共需要多少士兵，$p_i$ 表示尽早占领第 $i$ 座堡垒的前提下累计生产了几个士兵。
对于每一个 $i$，占领 $a$ 第 $i$ 小的堡垒需要等待
$$t_i=t_{i-1}+\max\left( 0,\lceil\frac{(s_i-p_{i-1})}{i}\rceil \right)$$
此时累计生产的士兵数量为
$$p_i=p_{i-1}+i\times(t_{i}-t_{i-1})$$
此时的答案为
$$
(p_i-s_i)+(i+1)\times(m-t_i)
$$
时间复杂度 $O(n)$

[***AC 代码***](https://qoj.ac/submission/2355761)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

ll ceilDiv(ll x, ll y) {
    return (x + y - 1) / y;
}

void solve() {
    int n, m;
    cin >> n >> m;
    vector<ll> a(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }
    sort(a.begin() + 1, a.end());
    partial_sum(a.begin(), a.end(), a.begin());

    ll ans = m;
    vector<ll> t(n + 1), p(n + 1);
    for (int i = 1; i <= n; ++i) {
        ll cur = (p[i-1] - a[i-1]);
        t[i] = t[i-1] + max(0LL, ceilDiv(a[i] - a[i-1] - cur, i));
        p[i] = p[i - 1] + (t[i] - t[i-1]) * i;
        if (t[i] > m) {
            break;
        }
        gmax(ans, (p[i]-a[i]) + (m-t[i]) * (i+1));
    }
    cout << ans << "\n";
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
