## [QOJ20029. Red Sequence](https://qoj.ac/contest/4071/problem/20029)

显然是优化 DP
$$
f(i)=\max_{sum_r[j+1,i]\ge\max(sum_y[j+1,i],sum_b[j+1,i])}\Bigg( f(i-1),f(j)+i-j\Bigg)
$$

### 解法一：cdq 分治

把限制条件写作：
$$
\begin{cases}
j\le i \\
sum_r(j)-sum_y(j)\le sum_r(i)-sum_y(i) \\
sum_r(j)-sum_b(j)\le sum_r(i)-sum_b(j)
\end{cases}
$$
三维偏序，直接 cdq 分治优化 DP，可以做到 $O(n\log^2n)$
需要大量卡常，比如 cdq 分治本身可以做到单 log，加上快读之类的

[***AC 代码***](https://qoj.ac/submission/2893395)

```cpp
#include <bits/stdc++.h>
using namespace std;

namespace IO {
    const int S = 1 << 20;
    char buf[S];
    int p = 0, len = 0;

    inline char gc() {
        if (p == len) {
            len = fread(buf, 1, S, stdin);
            p = 0;
            if (!len) return 0;
        }
        return buf[p++];
    }

    inline int read() {
        char c = gc();
        while (c < '0' || c > '9')
            c = gc();

        int x = 0;
        while (c >= '0' && c <= '9') {
            x = x * 10 + c - '0';
            c = gc();
        }
        return x;
    }
}
using IO::read;

const int inf = 1e9;

struct BIT {
    int n;
    vector<int> t, stk;
    BIT(int n): n(n), t(n+1, -inf) {
        stk.reserve(n+1);
    }
    int lowbit(int x) {
        return x&-x;
    }
    void modify(int x, int d) {
        for (; x <= n; x += lowbit(x)) {
            if (t[x] == -inf) 
                stk.push_back(x);
            t[x] = max(t[x], d);
        }
    }
    void reset() {
        for (auto x: stk) t[x]=-inf;
        stk.clear();
    }
    int query(int x) {
        int res = -inf;
        for (; x; x -= lowbit(x)) res = max(res, t[x]);
        return res; 
    }
};

const int N = 1e6+7;
int n, t[N], sr[N], sy[N], sb[N];

int main() {
    n = read();
    vector<int> v;
    for (int i = 1; i <= n; ++i) {
        sr[i] = read();
        sy[i] = read();
        sb[i] = read();
        sr[i] += sr[i-1];
        sy[i] += sy[i-1];
        sb[i] += sb[i-1];
    }

    auto cmp = [&](int x, int y) {
        return make_pair(sr[x]-sy[x], sr[x]-sb[x]) < make_pair(sr[y]-sy[y], sr[y]-sb[y]);
    };

    vector<int> id(n+1);
    iota(id.begin(), id.end(), 0);
    sort(id.begin(), id.end(), cmp);

    vector<int> f(n+1);
    BIT t(2*n+5);

    auto cdq = [&](auto &&cdq, int l, int r, vector<int>&ord, vector<int>&tmp) -> void {
        if (l == r-1) {
            if (l) {
                f[l] = max(f[l], f[l-1]);
            }
            return;
        }
        int m = ((l + r) >> 1);
        const int nl = m-l;
        const int nr = r-m;
        auto lo = tmp.begin()+l;
        auto hi = tmp.begin()+m;
        partition_copy(ord.begin()+l, ord.begin()+r, lo, hi, [m](int x){ return x<m; });
        cdq(cdq, l, m, tmp, ord);
        int i=0;
        for (int j = 0; j < nr; ++j) {
            while (i < nl && sr[lo[i]]-sy[lo[i]] <= sr[hi[j]]-sy[hi[j]]) {
                t.modify(sr[lo[i]]-sb[lo[i]]+n+1, f[lo[i]]-lo[i]);
                ++i;
            }
            int fr = t.query(sr[hi[j]]-sb[hi[j]]+n+1);
            if (fr > -inf) {
                f[hi[j]] = max(f[hi[j]], fr + hi[j]);
            }
        }
        t.reset();
        cdq(cdq, m, r, tmp, ord);
        merge(lo, lo+nl, hi, hi+nr, ord.begin()+l, cmp);
    };

    vector<int> tmp(n+1);
    cdq(cdq, 0, n+1, id, tmp);
    cout << f[n] << "\n"; 
}
```
