## [P4169 \[Violet\] 天使玩偶/SJY 摆棋子](https://www.luogu.com.cn/problem/P4169)

只考虑 $x_1\le x_2$ 且 $y_1\le y_2$ 的情况，距离等于 $(x_2-y_2)-(x_1+y_1)$
离线所有询问，这是一个经典的三维偏序问题，考虑 cdq 分治
先按每个点出现/询问的时间排序，每次合并 $[l,m)$ 和 $[m,r)$ 时按照 $x$ 排序，对每一个右侧的 $x_j$，双指针维护出左侧 $x_i\le x_j$ 的区间，开下标为 $y$ 的树状数组，在上面维护 $x+y$ 的前缀最小值即可
其余三种情况令 $x\to V-x+1$，$y\to V-y+1$，跑四遍 cdq 分治

[***AC 代码***](https://www.luogu.com.cn/record/295072874)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int inf = 1e9;

struct BIT {
    int n;
    vector<int> t;
    BIT(int n): n(n), t(n+1, -inf) {}
    int lowbit(int x) {
        return x & -x;
    }
    void modify(int x, int d) {
        for (; x <= n; x += lowbit(x)) gmax(t[x], d);
    }
    void reset(int x) {
        for (; x <= n; x += lowbit(x)) t[x] = -inf;
    }
    int query(int x) {
        int res = -inf;
        for (; x; x -= lowbit(x)) gmax(res, t[x]);
        return res;
    }
};

const int A = 1e6+5;

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int n, m;
    cin >> n >> m;

    vector<array<int,4>> p;
    for (int i = 0; i < n; ++i) {
        int x, y;
        cin >> x >> y;
        ++x; ++y;
        p.push_back({0,x,y,1});
    }
    vector<bool> need_answer(m+1);
    for (int i = 1; i <= m; ++i) {
        int t, x, y;
        cin >> t >> x >> y;
        ++x; ++y;
        p.push_back({i,x,y,t});
        if (t==2) need_answer[i] = 1;
    }
    sort(p.begin(), p.end());

    vector<int> ans(m+1, inf);
    BIT t(A);

    auto work = [&t=t, &ans=ans](vector<array<int,4>> p) {
        auto cdq = [&](auto &&cdq, int l, int r) -> void {
            if (l == r-1) {
                return;
            }
            int m = (l + r) >> 1;
            cdq(cdq, l, m);
            cdq(cdq, m, r);
            int i=l, j=m;
            while (j < r) {
                while (i < m && p[i][1] <= p[j][1]) {
                    if (p[i][3] == 1) {
                        t.modify(p[i][2], p[i][1] + p[i][2]);
                    }
                    ++i;
                }
                if (p[j][3] == 2) {
                    gmin(ans[p[j][0]], p[j][1]+p[j][2]-t.query(p[j][2]));
                }
                ++j;
            }
            for (int k = l; k < i; ++k) {
                if (p[k][3] == 1) {
                    t.reset(p[k][2]);
                }
            }
            inplace_merge(p.begin()+l, p.begin()+m, p.begin()+r, [&](auto x, auto y){
                return array<int,3>{x[1],x[2],x[3]} < array<int,3>{y[1],y[2],y[3]};
            });
        };
        cdq(cdq, 0, p.size());
    };

    for (int x = 0; x < 4; ++x) {
        auto q(p);
        for (int i = 0; i < int(q.size()); ++i) {
            if (x&1) q[i][1] = A-q[i][1]+1;
            if (x>1) q[i][2] = A-q[i][2]+1;
        }
        work(q);
    }
    for (int i = 1; i <= m; ++i) {
        if (need_answer[i]) {
            cout << ans[i] << "\n";
        }
    }
}
```
