## [QOJ18478. 博德之跃 3](https://qoj.ac/contest/3789/problem/18478)

不是所有的字符串都需要算法的，有的时候要敢贪。
注意到操作三中，将 $A$ 分成一个字母一个字母取一定不劣。
本质上操作三就是允许在前后缀相等的情况下删掉某一部分，而这一部分对字典序会有怎样的影响，则取决于中间的部分。
先特判全相等的情况，一直进行操作三直到字符串长度 $\le2$，再把剩下的两个字母加进去即可。
然后分类讨论，记此时 $s_l=s_r=c$，分别找到 $l$ 右侧和 $r$ 左侧第一个不等于 $c$ 的字母 $x,y$，此时左右侧连续 $c$ 的数量分别为 $A,B$，则：

- 若 $x<c$ 且 $y<c$，需要尽可能快地消耗 $c$，应进行 $\min(A,B)$ 次操作三
- 若 $x<c$ 但 $y>c$，肯定需要将左侧的若干个 $c$ 先取出来，而右侧的 $y>c$，应当予以保留，于是进行 $A$ 次操作一
- 若 $x>c$ 且 $y<c$，同上，进行 $B$ 次操作二
- 若 $x>c$ 且 $y>c$，则应当全部保留，即进行 $A$ 次操作一和 $B$ 次操作二

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

void solve() {
    string s;
    cin >> s;
    int l=0, r=s.size()-1;
    
    vector<int> cnt(26);
    for (int i = 0; i < int(s.size()); ++i) {
        cnt[s[i]-'a']++;
    }

    string ans;
    auto op1 = [&]() {
        ans += s[l];
        cnt[s[l]-'a']--;
        l++;
    };
    auto op2 = [&]() {
        ans += s[r];
        cnt[s[r]-'a']--;
        r--;
    };
    auto op3 = [&]() {
        assert(s[l] == s[r]);
        ans += s[l];
        cnt[s[l]-'a']--;
        cnt[s[r]-'a']--;
        l++;
        r--;
    };

    while (l <= r) {
        if (cnt[s[l]-'a'] == r-l+1) {
            while (r-l+1 > 2) {
                op3();
            }
            ans += s.substr(l, r-l+1);
            break;
        } else if (s[l] == s[r]) {
            int nl=l, nr=r;
            while (s[nl+1] == s[l]) ++nl;
            while (s[nr-1] == s[r]) --nr;
            assert(nl+1 <= nr-1);
            int A = nl-l+1;
            int B = r-nr+1;
            if (s[nl+1] < s[l] && s[nr-1] < s[r]) {
                for (int i = 1; i <= min(A,B); ++i) op3();
            } else if (s[nl+1] < s[l]) {
                for (int i = 1; i <= A; ++i) op1();
            } else if (s[nr-1] < s[r]) {
                for (int i = 1; i <= B; ++i) op2();
            } else {
                for (int i = 1; i <= A; ++i) op1();
                for (int i = 1; i <= B; ++i) op2();
            }
        } else if (s[l] < s[r]) {
            op1();
        } else {
            op2();
        }
    }
    cout << ans << "\n";
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
