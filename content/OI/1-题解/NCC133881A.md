## [NCC133881A](https://ac.nowcoder.com/acm/contest/133881/A)

Observation 1：交换前后下标的奇偶性不变，相当于把奇数位置当桥交换偶数位置，或者反过来
Observation 2：一个数可以出现的位置存在包含关系，想象 $2,3,1,5,4,...$，交换后 $1$ 能够到达 $4$ 的所有位置，而 $4$ 会被 $3$ 拦住
从大到小放数字，使用并查集维护可以到达的位置，初始时独立成组。
数字 $v$ 在组内有 $siz[v]$ 个可以放的位置，每次调一个位置放下，答案乘 $siz$，然后减一。
此时 $v$ 可以当作桥连通两边，只要两边都有比 $v$ 小的数，即 $siz[pos[v]-1]>0$ 且 $siz[pos[v]+1]>0$ 时，把两边合并，表示剩余的数字可以在这个大组内任意选位置。

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

struct DSU {
    int n;
    vector<int> f, siz;
    DSU(int n): n(n), f(n + 1), siz(n + 1, 1) {
        iota(f.begin(), f.end(), 0);
    }
    int find(int x) {
        return x == f[x] ? f[x]: f[x]=find(f[x]);
    }
    bool same(int x, int y) {
        return find(x) == find(y);
    }
    void merge(int x, int y) {
        x = find(x);
        y = find(y);
        if (x == y) return;
        f[x] = y;
        siz[y] += siz[x];
    }
    int& gsiz(int x) {
        return siz[find(x)];
    }
};

const int MOD = 998244353;
ll add(ll x, ll y) { return (x + y) % MOD; }
ll del(ll x, ll y) { return (x - y + MOD) % MOD; }
ll mul(ll x, ll y) { return (x * y) % MOD; }

void solve() {
    int n;
    cin >> n;
    vector<int> p(n + 2), pos(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> p[i];
        pos[p[i]] = i;
    }
    DSU D(n);
    ll ans = 1;
    for (int v = n; v; --v) {
        int &siz = D.gsiz(pos[v]);
        ans = mul(ans, siz--);
        if (2 <= pos[v] && pos[v] <= n-1 && D.gsiz(pos[v]-1) && D.gsiz(pos[v]+1)) {
            D.merge(pos[v]-1, pos[v]+1);
        }
    }
    cout << ans << "\n";
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
