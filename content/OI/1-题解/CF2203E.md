## [CF2203E. Probabilistic Card Game](https://codeforces.com/contest/2203/problem/E)

题目的意思就是只有 $b$ 在 $a,c$ 中间时得分。
把序列按照从小到大排序，记 $rank(x)$ 表示 $x$ 从左数的排位，$val_i$ 表示排序后从左数第 $i$ 个数。显然 $b$ 肯定贴着 $a$ 选，并且只有一边是可以得分的，具体而言分成如下两类：

- 如果 $rank(b)=rank(a)-1$，那么答案为 $\displaystyle (rank(a)-1)\times b-\sum_{i=1}^{rank(a)-1}(val_i)$，记为 $A$
- 如果 $rank(b)=rank(a) + 1$，那么答案为 $\displaystyle\sum_{i=rank(a)+1}^{n}(val_i)-(n-rank(a))\times b$，记为 $B$

维护 $rank$、区间和、$\sum\max(0,a_i-x)$，这是树状数组的经典运用，开两个权值树状数组，一个记录数量，使用树状数组上二分的技巧维护 $rank$，另一个维护区间和。我们可以在 $O(\log n)$ 的时间内维护和查询出 $A,B$。
Bob 的目标是让 $\max(A,B)$ 最大，而 Alice 则是让 $max(A,B)$ 最小。显然 $rank(a)$ 应该取在 $A\approx B$ 的位置，形象地说叫“重心”，二分一下然后在这个附近小区间枚举即可。
最终答案为 $\dfrac{\min\{\max(A,B)\}}{n-2}$。时间复杂度 $O(n\log n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int MOD = 998244353, N = 1e6 + 5;
template<class...T> ll add(T...t){ll r=0;(((r+=t%MOD)>=MOD?r-=MOD:0),...);return(r%MOD+MOD)%MOD;}
template<class...T> ll mul(T...t){ll r=1;((r=r*(t%MOD)%MOD),...);return(r%MOD+MOD)%MOD;}
ll del(ll x,ll y){return add(x,MOD-y);}
ll qpow(ll a,ll b=MOD-2){ll r=1;for(;b;b>>=1,a=mul(a,a))if(b&1)r=mul(r,a);return r;}
ll fac[N + 5], Inv[N + 5];
void preprocess() {
    fac[0] = 1;
    for (int i = 1; i <= N; ++i) fac[i] = mul(fac[i-1], i);
    Inv[N] = qpow(fac[N]);
    for (int i = N - 1; ~i; --i) Inv[i] = mul(Inv[i + 1], i + 1);
}
ll C(ll n, ll m) { return m > n ? 0: mul(fac[n], Inv[m], Inv[n-m]); }
ll P(ll n, ll m) { return m > n ? 0: mul(fac[n], Inv[n-m]); }

struct BIT {
	int n;
    vector<ll> t;
    BIT(int n): n(n), t(n+1) { }  // 注意值域树状数组中 n=tot
    int lowbit(int x) { return x&-x; }
    void modify(int x, ll d) {
        for (; x <= n; x += lowbit(x)) { t[x] += d; }
    }
	// 前缀和
    ll query(ll x) {
        ll res = 0;
        for (; x; x -= lowbit(x)) { res += t[x]; }
        return res;
    }
    // 求最小的前缀和等于x的位置
	int select(int x) {
	    int ans = 0, sum = 0;
	    for (int i = __lg(n); i >= 0; --i) {
	        if (ans+(1<<i) <= n && sum+t[ans+(1<<i)] < x) { 
	            ans += 1<<i;
	            sum += t[ans];
	        }
	    }
	    return ans + 1;
	}
};

void solve()
{
    int n;
    cin >> n;
    vector<ll> a(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }
    vector<ll> v(a.begin() + 1, a.end());
    sort(v.begin(), v.end());
    v.erase(unique(v.begin(), v.end()), v.end());
    int V = v.size();

    BIT tcnt(V + 1), tsum(V + 1);

    auto calcL = [&](int k) -> ll {
        if (k <= 0) {
            return 0;
        }
        int idx = tcnt.select(k);
        ll val = v[idx - 1];
        ll s = tsum.query(idx - 1);
        return (k - 1) * val - s;
    };

    auto calcR = [&](int k, int i) -> ll {
        if (k > i) {
            return 0;
        }
        int idx = tcnt.select(k);
        ll val = v[idx - 1];
        ll s = tsum.query(V + 1) - tsum.query(idx);
        return s - (i - k) * val;
    };

    auto calc = [&](int k, int i) -> ll {
        ll A = calcL(k - 1);
        ll B = calcR(k + 1, i);
        return max(A, B);
    };

    for (int i = 1; i <= n; ++i) {
        int pos = lower_bound(v.begin(), v.end(), a[i]) - v.begin() + 1;
        tcnt.modify(pos, 1);
        tsum.modify(pos, a[i]);
        if (i < 3) {
            continue;
        }
        int l = 1, r = i;
        while (l + 1 < r) {
            int m = (l + r) >> 1;
            if (calcL(m - 1) >= calcR(m + 1, i)) {
                r = m;
            } else {
                l = m;
            }
        }
        ll ans = calc(l, i);
        if (l > 1) gmin(ans, calc(l-1, i));
        if (l < i) gmin(ans, calc(l+1, i));
        ans = mul(ans, qpow(i - 2));
        cout << ans << "\n";
    }
}

int main()
{
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    preprocess();
    int T = 1;
    // cin >> T;
    while (T--) {
        solve();
    }
}
```
