## [NCC133877F. Fabulous Tree](https://ac.nowcoder.com/acm/contest/133877/F)

注意到题目限制了 $\sum nW$，显然是根据这个 DP
记 $W$ 是树上的最大边权，显然答案有一个宽松的上界 $2W$，固定根的权值 $X$ 然后考虑从根开始往下搜索，每次必然会落在 $[X-W,X+W]$，并且总能选择 $\pm w$ 让孩子也落在这个区间里。
一次性控制两边的边界太过困难，DP 的思路是固定下界，求出对应的最小上界是什么。
具体而言，设 $dp(u,i)$ 表示固定当前 $u$ 子树树根的值，子树内的最小权值是 $V-i$ 时，子树内的最大权值是什么。取 $0\le i\le 2W$ 即可，共 $O(nW)$ 个状态
转移时，对根 $u$ 考虑其若干儿子 $v$，每个儿子连向父亲的边权为 $w$
对应单个儿子，固定 $u$ 的权值为 $X$，则 $v$ 的权值有 $X-w$ 和 $X+w$ 两种情况，前者相当于原树根权值和下界减小 $w$，后者相当于原树根权值下界增大 $w$。对于权值下界 $V-i$，我们应该在这两种方案中取一个上界最小的，即
$$
\min\bigg(\underbrace{\max(0,dp(v,i-w)-w)}_{a_v=X-w},\underbrace{dp(v,i+w)+w}_{a_v=X+w}\bigg)
$$
对于若干个不同的儿子，每个权值下界 $V-i$ 应当取这些“最优方案”的最大值，即
$$
dp(u,i)=\max_{v\in son(u)}\Bigg\{ \min\bigg(\underbrace{\max(0,dp(v,i-w)-w)}_{a_v=X-w},\underbrace{dp(v,i+w)+w}_{a_v=X+w}\bigg) \Bigg\}
$$
每个状态只需要常数时间就可以完成转移，最终每个子树 $u$ 的答案为 $\min_i\{dp(u,i)+i\}$
时间复杂度 $O(nW)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int inf = 1e9;

void solve() {
    int n;
    cin >> n;

    vector G(n + 1, vector<pair<int, int>>());
    int W = 0;
    for (int i = 1; i < n; ++i) {
        int u, v, w;
        cin >> u >> v >> w;
        gmax(W, w);
        G[u].emplace_back(v, w);
        G[v].emplace_back(u, w);
    }

    vector dp(n + 1, vector<int>(2*W));
    auto dfs = [&](auto &&dfs, int u, int fa) -> void {
        for (auto [v, w]: G[u]) {
            if (v == fa) {
                continue;
            }

            dfs(dfs, v, u);
            vector<int> tmp(2*W, inf);
            for (int j = 0; j < 2*W; ++j) {
                if (0 <= j + w && j + w < 2 * W) gmin(tmp[j+w], max(0, dp[v][j]-w));
                if (0 <= j - w && j - w < 2 * W) gmin(tmp[j-w], dp[v][j]+w);
            }
            for (int j = 0; j < 2*W; ++j) {
                gmax(dp[u][j], tmp[j]);
            }
        }
        for (int j = 1; j < 2*W; ++j) {
            gmin(dp[u][j], dp[u][j-1]);
        }
    };

    dfs(dfs, 1, 0);
    for (int i = 1; i <= n; ++i) {
        int ans = inf;
        for (int j = 0; j < 2*W; ++j) {
            gmin(ans, dp[i][j]+j);
        }
        cout << ans << " \n"[i == n];
    }
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }    
}
```
