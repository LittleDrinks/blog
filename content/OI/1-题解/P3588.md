## [P3588 \[POI 2015 R2\] 沙漠 Desert](https://www.luogu.com.cn/problem/P3588)

对每个点维护 `least` 和 `most` 表示可以接受的区间，一开始 `least=1, most=1e9`
输入的 $s$ 个位置，令 `least=most=a`
建图，如果 $u$ 比 $v$ 大，则建边 $u\to v$，边权为 $-1$，表示 `most` 必须至少减一，在这张图上跑拓扑排序即可维护出所有位置的 `most`。
无解的条件是：成环（边权非正，所以成环一定无解）或某个位置 `least>most`。
但是这个建边的复杂度是难以接受的，达到了 $O(kmn)$，所以考虑用数据结构优化建图
每一条限制 `l r k`，建立一个虚拟点 $X$，`k` 个点向 $X$ 连 $-1$ 的边，$X$ 向剩下 `r-l+1-k` 个点连 $0$ 的边。后者可以看作向 $k+1$ 个区间连边，于是可以用线段树进一步优化。
建图的时间复杂度是 $O(n+\sum k\log k)$，总点数/拓扑排序复杂度 $O(n+m)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int inf = 1e9;

#define m ((l+r)>>1)
#define lson (p << 1)
#define rson (p << 1 | 1)
const int N=1e5+5, M=2e5+5, E=(N<<2)+M;
int tot, pos[N], dwId[N<<2], deg[E], most[E], least[E];
vector<pair<int,int>> G[E];
void add_edge(int u, int v, int w) {
    G[u].emplace_back(v, w);
    // cerr << u << " -> " << v << " :: " << w << "\n";
    ++deg[v];
}
void build(int p, int l, int r) {
    if (l == r-1) {
        pos[l] = dwId[p] = ++tot;
        return;
    }
    build(lson, l, m);
    build(rson, m, r);
    dwId[p] = ++tot;
    add_edge(dwId[p], dwId[lson], 0);
    add_edge(dwId[p], dwId[rson], 0);
}
void work(int p, int l, int r, int x, int y, auto &&F) {
    if (x <= l && r <= y) { F(p); return; }
    if (y <= l || r <= x) { return; }
    work(lson, l, m, x, y, F);
    work(rson, m, r, x, y, F);
}
#undef m
#undef lson
#undef rson

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int n, s, m;
    cin >> n >> s >> m;
    build(1, 0, n);
    fill(most, most+E, 1e9);
    fill(least, least+E, 1);
    for (int i = 1; i <= s; ++i) {
        int p;
        cin >> p;
        --p;
        cin >> most[pos[p]];
        least[pos[p]] = most[pos[p]];
    }
    for (int i = 1; i <= m; ++i) {
        int l, r, k;
        cin >> l >> r >> k;
        --l;
        vector<int> id(k, r);
        int u = ++tot;
        for (int j = 0; j < k; ++j) {
            cin >> id[j];
            --id[j];
            add_edge(pos[id[j]], u, -1);
        }
        for (int las=l, j=0; j < k; ++j) {
            // cerr << "***" << pair{las,pos[id[j]]} << "\n";
            work(1, 0, n, las, id[j], [&](int p){ add_edge(u, dwId[p], 0); });
            las = id[j]+1;
        }
        work(1, 0, n, id[k-1]+1, r, [&](int p){ add_edge(u, dwId[p], 0); });
    }

    for (int i = 1; i <= tot; ++i) {
        // cerr << pair{least[i], most[i]} << "\n";
    }
    int cnt = 0;
    queue<int> q;
    for (int i = 1; i <= tot; ++i) {
        if (deg[i] == 0) {
            q.push(i);        
        }
    }
    while (!q.empty()) {
        int u = q.front(); q.pop();
        ++cnt;
        for (auto [v, w]: G[u]) {
            gmin(most[v], most[u]+w);
            if (--deg[v] == 0) {
                q.push(v);
            }
        }
    }
    bool ok = (cnt == tot);
    for (int i = 1; i <= tot; ++i) {
        ok &= (least[i] <= most[i]);
        // cerr << pair{least[i], most[i]} << "\n";
    }
    if (!ok) {
        cout << "NIE\n";
    } else {
        cout << "TAK\n";
        for (int i = 0; i < n; ++i) {
            cout << most[pos[i]] << " \n"[i == n-1];
        }
    }
}
```
