## [QOJ24851. 凸包序列](https://qoj.ac/contest/4685/problem/24851/statement/zh_cn)

如果已知 $p,q$ 是凸包上的两个点，那么 $[-p, -q]$ 这个极角区间内只要有点，这个凸包就能把 $(0,0)$ 包进去。
二分区间长度（或者双指针），从左到右遍历长度为该值的区间 $[l,r]$，每次钦定 $l$ 是合法区间的左端点，拿一个 set 维护区间内的点
分类讨论：

1. 必要条件是必须有三个方向的点
2. 如果 `-p[l]` 在 set 里，加上“三个方向”直接就符合条件了
3. 否则，二分找到 `-P[l]` 方向极角序最接近的两个点，判断这三个点是不是把 $(0,0)$ 包进去即可

[***AC 代码***](https://qoj.ac/submission/3459446)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

int sgn(ll x) { return (x>0)-(x<0); }

using T = long long;
struct Point {
    T x, y;
    Point(): x(0), y(0) { }
    Point(T x, T y): x(x), y(y) { }
    friend istream& operator>> (istream& is, Point &p) { return is >> p.x >> p.y; }
    friend ostream& operator<< (ostream& os, const Point &p) { return os << "{" << p.x<<","<<p.y<<"}"; }
    Point operator+ (const Point &p) const { return Point(x+p.x, y+p.y); }
    Point operator- (const Point &p) const { return Point(x-p.x, y-p.y); }
    T operator* (const Point &p) const { return x*p.x+y*p.y; }
    T operator^ (const Point &p) const { return x*p.y-y*p.x; }
    bool operator< (const Point &p) const { return pair{x,y} < pair{p.x,p.y}; }
};

struct argcmp {
    bool operator()(const Point&a, const Point &b) const {
        auto quad = [](const Point &a) {
            if (a.y < 0)  { return 1; }
            if (a.y > 0)  { return 3; }
            if (a.x >= 0) { return 2; }
            if (a.x < 0)  { return 4; }
            return -1;
        };
        int qa = quad(a), qb = quad(b);
        if (qa == qb) {
            ll cross = a ^ b;
            return cross > 0;
        }
        return qa < qb;
    }
};

void solve() {
    int n;
    cin >> n;
    vector<Point> P(n);
    for (int i = 0; i < n; ++i) {
        cin >> P[i];
    }
    auto check = [&](int m) {
        argcmp cmp;
        map<Point,int,argcmp> cnt{cmp};
        bool ok = false;
        for (int i = 0; i < m-1; ++i) {
            cnt[P[i]]++;
        }
        for (int i = 0, j = m-1; j < n; ++i, ++j) {
            cnt[P[j]]++;
            if (cnt.size() >= 3) {
                Point t(-P[i].x, -P[i].y);
                auto it = cnt.lower_bound(t);
                if (it != cnt.end() && !cmp(it->first, t) && !cmp(t, it->first)) {
                    return true;
                }
                auto nxt = it==cnt.end()? cnt.begin(): it;
                auto pre = it==cnt.begin()? prev(cnt.end()): prev(it);
                ok |= sgn((pre->first)^(nxt->first)) >= 0;
            }
            if (--cnt[P[i]] == 0) {
                cnt.erase(P[i]);
            }
        }
        return ok;
    };
    if (!check(n)) {
        cout << "-1\n";
        return;
    }
    int l=0, r=n;
    while (l + 1 < r) {
        int m = ((l+r)>>1);
        bool ok = check(m);
        if (ok) {
            r = m;
        } else {
            l = m;
        }
    }
    cout << r << "\n";
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
