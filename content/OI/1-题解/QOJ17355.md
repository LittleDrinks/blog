## [QOJ17355. Zebra Crossing](https://qoj.ac/contest/3729/problem/17355)

有点奇怪为什么这题赛时没做出来，赛后没细想直接看题解了，这里记录一下我反推出来的思考路径。
首先要想到一个转化，一次走到 $dis\le k$ 的点，相当于初始时有 $k$ 点体力，每走一步体力减少 $1$，走到白色节点体力恢复，走到黑色节点且体力归零时答案加一。这样就可以把每次走 $k$ 步的断裂路径变成一条连续的路径。这个是“恰好走 $k$ 步”常用的 trick，核心思想是拆点、分层图。
显然我们要尽量用白色节点作为中转，要么就是从白色节点走到子树，要么就是从一个白色节点走到父节点再走到其他子树。
这里的难点在于【走到其他子树】比较难操作，看上去是一个换根的操作，但其实【白色节点往回，走到其他子树】的操作可以变为【从当前节点 $u$ 往下找一个最近的白色节点，走一个来回，再走到子树内】，于是这又变回了一个树上问题，我们可以在父结点处理答案。

然后从上到下模拟，记录当前答案 $ans$  和体力 $x$。走到其他子树可以描述为：当 $x\ge d_u$ 时，令
$$
x\leftarrow\max(x,k-d_u)
$$
模拟全过程即可得到答案。最后需要特判一下“终点是黑点且答案未计入”的情况

[***AC 代码***](https://qoj.ac/submission/2353092)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int inf = 1e9;

void solve() {
    int n, k;
    cin >> n >> k;

    string is_white;
    cin >> is_white;

    vector G(n + 1, vector<int>());
    for (int i = 1; i < n; ++i) {
        int u, v;
        cin >> u >> v;
        G[u].push_back(v);
        G[v].push_back(u);
    }

    vector<int> d(n + 1, inf);
    auto dfs1 = [&](auto &&dfs1, int u, int fa) -> void {
        if (is_white[u-1] == '1') {
            d[u] = 0;
        }
        for (auto v: G[u]) {
            if (v == fa) continue;
            dfs1(dfs1, v, u);
            gmin(d[u], d[v] + 1);
        }
    };
    dfs1(dfs1, 1, 0);
    
    vector<int> ans(n + 1);
    vector<int> stp(n + 1);
    ans[1] = 0;
    stp[1] = k;
    auto dfs2 = [&](auto &&dfs2, int u, int fa) -> void {
        if (u != 1) {
            ans[u] = ans[fa];
            if (is_white[u-1] == '1') {
                stp[u] = k;
            }
            else {
                stp[u] = stp[fa] - 1;
                if (stp[u] == 0) {
                    ++ans[u];
                    stp[u] = k;
                }
            }  
        }
        if (stp[u] >= d[u]) {
            gmax(stp[u], k-d[u]);
        }
        for (auto v: G[u]) {
            if (v == fa) continue;
            dfs2(dfs2, v, u);
        }
    };
    dfs2(dfs2, 1, 0);
    
    for (int i = 2; i <= n; ++i) {
        cout << (ans[i] + (is_white[i-1] == '0' && stp[i] != k)) << " \n"[i == n];
    }
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
