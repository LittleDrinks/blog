## [NCC133878I]()

设 $g(i)=|x_i-x_{i-1}|+|x_i-x_{i+1}|$
相邻的交换是很好做的，先 $O(n)$ 做掉
当交换的 $i,j$ 不相邻时，两个 $g$ 的计算是互不影响的，我们可以把这里的 $a_{i-1},a_{i+1},a_{j-1},a_{j+1}$ 当作常数去研究函数的性质
$|x-a|+|x-b|$，把绝对值拆掉是一个平底锅，也可以表示为：
$$
\max\begin{cases}
2x-a-b \\
\max(a,b)-\min(a,b) \\
-2x+a+b
\end{cases}
$$
为了兼容端点只有一个绝对值的情况，我们不妨将这个值看作是斜率在 $\{\pm2,\pm1,0\}$ 的五条直线的 $\max$，不可表示则截距 $b=-\infty$
这里的 max 是一个很重要的性质，这说明我们并不用在乎具体斜率的取值，或者说不用在乎被换过来的值会让这个函数的斜率变成什么样子，我们只需要枚举 $k$、计算增量、取 max 即可
记 $i<j$，且此时选取的直线斜率分别为 $k_i,k_j$，此时交换 $i,j$ 的增量可以表示为
$$
\begin{align}
k_i\times a_j+b_{i,k_i}-g(i)\\
k_j\times a_i+b_{j,k_{j}}-g(j)
\end{align}
$$
记 $best[k_1][k_2]$ 表示此时斜率的选择，分离变量，维护 $\max\{k_j\times a_i+b_{i,k_i}-g(i)\}$，然后在 $a_j$ 处计算剩余部分，即可在 $O(5\times5\times n)$ 的时间内 DP 解决。注意枚举右端点到 $j$ 的时候，$i\le j-2$

[***AC 代码***](https://ac.nowcoder.com/acm/contest/view-submission?submissionId=84260735)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int K = 5, BIAS=2;
const ll inf = 1e18;

void solve() {
    int n;
    cin >> n;
    vector<ll> a(n + 2);
    vector b(n + 2, vector<ll>(K + 1, -inf));
    ll ans = 0, base = 0;
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }
    
    for (int i = 1; i <= n; ++i) {
        if (i < n) {
            base += abs(a[i] - a[i + 1]);
        }
        if (i == 1) {
            b[i][BIAS-1] = a[i+1];
            b[i][BIAS+1] = -a[i+1];
        }
        else if (i == n) {
            b[i][BIAS-1] = a[i-1];
            b[i][BIAS+1] = -a[i-1];
        }
        else {
            b[i][BIAS] = max(a[i-1],a[i+1]) - min(a[i-1],a[i+1]);
            b[i][BIAS-2] = a[i-1]+a[i+1];
            b[i][BIAS+2] = -a[i-1]-a[i+1];
        }
    } 

    
    ans = base;
    
    for (int i = 1; i < n; ++i) {
        ll cur = base;
        if (i - 1 >= 1) {
            cur -= abs(a[i-1]-a[i]);
            cur += abs(a[i-1]-a[i+1]);
        }
        if (i + 2 <= n) {
            cur -= abs(a[i+1]-a[i+2]);
            cur += abs(a[i]-a[i+2]);
        }

        gmax(ans, cur);
    }
    
    vector best(K+1, vector<ll>(K+1, -inf));

    for (int i = 1; i <= n; ++i) {
        for (int k1 = -2; k1 <= 2; ++k1) {
            for (int k2 = -2; k2 <= 2; ++k2) {
                ll ori = 0;
                if (i-1 >= 1) ori += abs(a[i-1]-a[i]);
                if (i+1 <= n) ori += abs(a[i]-a[i+1]);
                gmax(ans, base+best[BIAS+k1][BIAS+k2]+b[i][BIAS+k2]+k1*a[i]-ori); 
            }
        }
        if (i == 1) {
            continue;
        }
        int id = i-1;
        for (int k1 = -2; k1 <= 2; ++k1) {
            for (int k2 = -2; k2 <= 2; ++k2) {
                ll ori = 0;
                if (id-1 >= 1) ori += abs(a[id-1]-a[id]);
                if (id+1 <= n) ori += abs(a[id]-a[id+1]);

                ll be = k2*a[id]+b[id][BIAS+k1]-ori;

                gmax(best[BIAS+k1][BIAS+k2], be);
            }
        }
    }

    cout << ans << "\n";
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
