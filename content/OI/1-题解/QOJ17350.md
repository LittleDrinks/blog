## [QOJ17350. VIP Coupon](https://qoj.ac/contest/3729/problem/17350)

显然应该丢弃 $b>c$ 的优惠券，纯亏本
转化一下题意，其实是要把优惠券分为 $n$ 组，每组若干张优惠券加恰好一件纪念品，使花费最小。
我们先考虑 $n=1$ 的情况。
手玩一些样例可以得到一个形象的结论：把优惠券看作区间 $[b,c]$，假设最后一件商品的价格是 $a$（可以是纪念品或优惠券），那么优惠的价格其实是 $a$ 左侧所有区间并的长度。
拓展到 $n$ 件纪念品的情况。假设我们已经分了 $n$ 组，每组右端点初始为 $0$。我们把这 $n$ 组按右端点排序，所有区间按左端点排序。
我们考察两组 $r_1<r_2$ 和两个区间 $l_1<l_2$，可以讨论一下，不难证明让 $(l_1,r_2), (l_2,r_1)$ 匹配是劣的。于是我们总可以微调，每次让左端点更小的区间追加到右端点最小的区间后面，获得更大的优惠。对于纪念品，我们可以将其视为区间 $[a,\infty)$，优雅地套用这套算法，顺便禁止往纪念品后追加商品。
时间复杂度 $O(n\log n)$

[***AC 代码***](https://qoj.ac/submission/2355701)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const ll INF = 1e18;

void solve() {
    int n, m;
    cin >> n >> m;
    vector<ll> a(n), b(m), c(m);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    for (int i = 0; i < m; ++i) {
        cin >> b[i];
    }
    for (int i = 0; i < m; ++i) {
        cin >> c[i];
    }

    vector<pair<ll,ll>> seg;
    for (int i = 0; i < n; ++i) {
        seg.emplace_back(a[i], INF);
    }
    for (int i = 0; i < m; ++i) {
        if (b[i] > c[i]) continue;
        seg.emplace_back(b[i], c[i]);
    }
    sort(seg.begin(), seg.end());
    
    priority_queue<array<ll,3>, vector<array<ll,3>>, greater<>> q;
    for (int i = 1; i <= n; ++i) {
        q.push({0,0});
    }
    for (auto &[x, y]: seg) {
        auto [r, l, save] = q.top();
        q.pop();

        if (y <= r) {
            q.push({r, l, save});
            continue;
        }
        
        save += min(r, x) - l;
        r = y;
        l = x;
        q.push({r, l, save});
    }
    
    ll ans = 0;
    while (!q.empty()) {
        auto [r, l, save] = q.top();
        q.pop();
        ans += l - save;
    }

    cout << ans << "\n";
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
