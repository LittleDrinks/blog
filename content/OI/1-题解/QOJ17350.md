## [QOJ17350. VIP Coupon](https://qoj.ac/contest/3729/problem/17350)



[***AC 代码***](https://qoj.ac/submission/2355701)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const ll INF = 1e18;

void solve() {
    int n, m;
    cin >> n >> m;
    vector<ll> a(n), b(m), c(m);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    for (int i = 0; i < m; ++i) {
        cin >> b[i];
    }
    for (int i = 0; i < m; ++i) {
        cin >> c[i];
    }

    vector<pair<ll,ll>> seg;
    for (int i = 0; i < n; ++i) {
        seg.emplace_back(a[i], INF);
    }
    for (int i = 0; i < m; ++i) {
        if (b[i] > c[i]) continue;
        seg.emplace_back(b[i], c[i]);
    }
    sort(seg.begin(), seg.end());
    
    priority_queue<array<ll,3>, vector<array<ll,3>>, greater<>> q;
    for (int i = 1; i <= n; ++i) {
        q.push({0,0});
    }
    for (auto &[x, y]: seg) {
        auto [r, l, save] = q.top();
        q.pop();

        if (y <= r) {
            q.push({r, l, save});
            continue;
        }
        
        save += min(r, x) - l;
        r = y;
        l = x;
        q.push({r, l, save});
    }
    
    ll ans = 0;
    while (!q.empty()) {
        auto [r, l, save] = q.top();
        q.pop();
        ans += l - save;
    }

    cout << ans << "\n";
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
