## [QOJ17469]()

> [!note] 结论：括号序列一定形如 $\texttt{()()()}$
> 序列中一定不存在 $\texttt{((}$，否则从右边一个左括号出发，序列会以一个左括号结尾，一定不合法。
> 为了平衡，左右括号数必然相等，而相邻两个左括号之间必须有一个右括号，于是只能是 $\texttt{()()()}$

只要找到一个凸包即可，显然最容易的找法是找一个只包含四个点的凸包，也就是找到两条直线，端点分别各自为 $\texttt{((}$ 和 $\texttt{))}$，使其相交。
按 $x$ 坐标从小到大遍历所有点 $p_2$，枚举其左侧同色的点 $p_1$，如果异色线段 $q_1q_2$ 能与线段 $p_1p_2$ 相交，那它一定能和易色点组成的凸包 $\{q\}$ 上一条边相交。
$O(n)$ 枚举右侧同色点 $p_2$，每次先 $O(n\log n)$ 求出左侧异色点的凸包，预处理凸包上所有点相对于 $p_2$ 的斜率，然后 $O(n)$ 枚举 $p_1$，在斜率区间上二分找到凸包上接近的若干条边，然后再判断即可。
时间复杂度 $O(n^2\log n)$。
$O(n\log n)$ 的做法：[O(nlogn) 解法](https://qoj.ac/problem/17469/discussion/1957)

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const db eps=1e-8;
const db PI=acos(-1.0);
const db inf = numeric_limits<db>::max();
int sgn(auto x) { return (x > eps) - (x < -eps); }

using T = ll;
struct Point {
	T x, y;
    int idx, col;
    Point(T x=0, T y=0): x(x), y(y) {}
	friend istream& operator>> (istream& is, Point& p) { return is>>p.x>>p.y; }
	friend ostream& operator<< (ostream& os, const Point& p) { return os<<p.x<<" "<<p.y; }
	db ang() { return atan2(y, x); }
	bool operator== (const Point &p) const { return sgn(x-p.x)==0 && sgn(y-p.y)==0; }
	bool operator< (const Point &p) const { return x<p.x || (x==p.x && y<p.y); }
	Point operator+ (const Point &p) const { return Point(x+p.x, y+p.y); }
	Point operator- (const Point &p) const { return Point(x-p.x, y-p.y); }
	Point operator* (const T &k) const { return Point(k*x, k*y); }
	Point operator/ (const T &k) const { return Point(x/k, y/k); }
	T operator* (const Point &p) const { return x*p.x + y*p.y; }
	T operator^ (const Point &p) const { return x*p.y - y*p.x; }  // 叉乘时打括号
	T len2() { return (*this)*(*this); }
	db len() { return sqrtl(len2()); }  // 等价于 hypotl(x, y)
	Point rot(db ang) { return Point(x*cos(ang)-y*sin(ang), x*sin(ang)+y*cos(ang)); }
	Point trunc(db l) { return (*this) * (l/len()); }
};

struct Line {
    Point a, b, v;
	Line(Point a, Point b): a(a), b(b) { v = b - a; }
	int toLeft(Point p) { return sgn(v^(p-a)); }
    // 判断点 p 是否在线段 AB 上
	bool onSegment(Point p) { return (toLeft(p)==0) && ((p-a)*(p-b)<=0); }
    // 点 p 到直线 AB 的距离
	db distToLine(Point p) { return fabs((v^(p-a))/v.len()); }
    // 点 p 到线段 AB 的距离
    db distToSeg(Point p) {
        if(a == b) return (p-a).len();
        Point v1=b-a, v2=p-a, v3=p-b;
        if(sgn(v1*v2) < 0) return v2.len();
        if(sgn(v1*v3) > 0) return v3.len();
        return distToLine(p);
    }
    // 点 p 在直线上的投影
	Point proj(Point p) { return a+v*((v*(p-a))/(v*v)); }
	// 用整数操作判断直线与点 C 的距离是否小于（等于）某个值 r
	// 常见于判断直线与圆的位置关系
	// 注意叉乘再相乘可能会爆 ll，所以这里用 __int128
	bool disLess(Point p, i128 r) {
	    i128 xmult = (v^(p-a));
	    return xmult*xmult <= r*r*v.len2();
	}
};

// 判断两条线段是否相交
// 先做两次跨立实验判断是否规范相交，再特判至少三点共线的情况
bool hasIntersection(Line L1, Line L2) {
	// 特判三点共线的情况
    if (L1.onSegment(L2.a)) { return true; }
    if (L1.onSegment(L2.b)) { return true; }
    if (L2.onSegment(L1.a)) { return true; }
    if (L2.onSegment(L1.b)) { return true; }
    // 跨立实验判断两直线是否规范相交
    // 线段端点分布在直线两侧，“线段与线段/线段与直线/直线与直线”是否相交对应修改即可
	function<bool(Line,Point,Point)> crossStanding = [](Line L, Point a, Point b){
        return L.toLeft(a) * L.toLeft(b) == -1;
    };
    return crossStanding(L1, L2.a, L2.b) && crossStanding(L2, L1.a, L1.b);
}

// 求平面点集的凸包，conv[0]是最左下角的点，逆时针顺序
// Andrew 算法，时间复杂度 O(nlogn)
vector<Point> toConvex(vector<Point> po) {
	if (po.empty()) { return {}; }
    if (po.size() <= 2) { return po; }
	sort(po.begin(), po.end());
    vector<Point> conv;
	auto check = [&](Point u){
		Point p1=conv.back(), p2=conv[conv.size()-2];
		return sgn((p1-p2)^(u-p1))<=0;
	};
	for (const Point &u: po) {
		while (conv.size()>1 && check(u)) { conv.pop_back(); }
		conv.push_back(u);
	}
	size_t k = conv.size();
	po.pop_back();
    reverse(po.begin(),po.end());
	for (const Point &u: po) {
		while (conv.size()>k && check(u)) { conv.pop_back(); }
		conv.push_back(u);
	}
	conv.pop_back();
    return conv;
}

bool argcmp(Point a, Point b) { return sgn(a^b) > 0; }

void solve() {
    int n;
    cin >> n;
    vector<Point> p(n);
    vector<Point> q[2];
    for (int i = 0; i < n; ++i) {
        cin >> p[i] >> p[i].col;
        p[i].idx = i + 1;
        q[p[i].col].push_back(p[i]);
    }
    sort(p.begin(), p.end());
    sort(q[0].rbegin(), q[0].rend());
    sort(q[1].rbegin(), q[1].rend());

    // cerr << "in\n";

    vector<Point> pL[2];
    for (int i = 0; i < n; ++i) {
        const auto &po = p[i];
        // cerr << "now " << i << " :: " << po << "\n";
        int c = po.col;
        while (!q[c^1].empty() && q[c^1].back().x <= po.x) {
            pL[c^1].push_back(q[c^1].back());
            q[c^1].pop_back();
        }

        auto poly = toConvex(pL[c^1]);
        // cerr << "\t" << " " << poly << "\n";
        int siz = poly.size();
        vector<pair<Point,int>> slope;

        for (int j = 0; j < siz; ++j) {
            slope.emplace_back(poly[j], j);
        }
        auto cmp = [&](auto x, auto y){
            return argcmp(x.first-po, y.first-po);
        };
        sort(slope.begin(), slope.end(), cmp);
        
        for (int j = 0; j < i && p[j].x < po.x; ++j) {
            if (p[j].col != c) continue;
            // cerr << "\tconsidering Line " << p[j] << " <-> " << p[i] << "\n";
            int idx = lower_bound(slope.begin(), slope.end(), make_pair(p[j]    , 0), cmp) - slope.begin();
            for (int k = idx-3; k <= idx+3; ++k) {
                if (k < 0 || k >= siz) continue;
                for (int bias = -3; bias <= 3; ++bias) {
                    int pos = (slope[k].second+bias+2*siz) % siz;
                    // cerr << "\t\t" << pos << ":" <<poly[pos] << ", " << (pos+1)%siz << ":" << poly[(pos+1)%siz] << '\n';
                    if (hasIntersection(
                        Line(poly[pos], poly[(pos+1)%siz]),
                        Line(p[j], po)
                    )) {
                        cout << "4\n";
                        cout << p[j].idx << " " << poly[pos].idx << " " << p[i].idx << " " << poly[(pos+1)%siz].idx << "\n";
                        return;
                    }
                }
            }
            
        }
    }

    cout << "-1\n";
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
