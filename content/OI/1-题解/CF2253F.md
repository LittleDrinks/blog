## [CF2253F. 4-beauty](https://codeforces.com/contest/2253/problem/F)

> [!tip] Observation
> $(1,2,3,4)$ 的 beauty 值是 4
> beauty 的最大值是 4，形如 $(a,2a,3a,4a)$ 的序列可以取到

假设四个数分别为 $(a+d,a+2d,a+3d,a+4d)$，根据辗转相除法，有 $a+id\mid (j-i)d$

- 当 $a>0$ 时，只有 $i=1$ 时可能成立，也就是只可能出现 $(a+d,a+jd)$ 至多三种情况
- 当 $a=0$ 时，取 $(d,2d),\ (d,3d),\ (d,4d),\ (2d,4d)$ 即可让 beauty 值为 4

相当于要求：如果 $a$ 在集合中，$2a,3a,4a$ 就不集合中
后半段的做法相当于[P3226 [HNOI 2012] 集合选数](https://www.luogu.com.cn/problem/P3226)的加强版
构造一个矩阵，第一行第一列的值是 $1$，往右边走就乘 $2$，往下面走就乘 $3$，形如：
```
1  2  4  8   16  32  ...
3  6  12 24  48  96  ...
9  18 36 72  ...
27 ...
```
相当于对于集合中的每一个┌ 都要有一个点能够覆盖他
反过来做，对于每个点，可以覆盖一个 ┘形状的区域，考虑轮廓线 DP，从下往上，从右往左选择每个位置是不是被选择，维护轮廓线
```
     1111
00000xxxx
xxxxxxxxx
```
如果一个位置可以延伸┌ ，并且在 ┘的右下角还未被选择，那么必须选择
对于每个边长为 $x<y$ 的矩阵，时间复杂度 $O(xy 2^x)$，加起来就是 O(能过)

[***AC 代码***](https://codeforces.com/contest/2253/submission/387119086)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const ll inf = 1e16;

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int n;
    cin >> n;
    vector<int> c(n + 1);
    c[0] = 1e18;
    for (int i = 1; i <= n; ++i) {
        cin >> c[i];
    }
    vector<bool> vis(n + 1);

    ll ans = 0;

    auto dp = [&](vector<vector<int>> G) {
        int n = G.size();
        int M = G[0].size();

        vector<ll> f(1<<M, inf);
        f[0] = 0;
        
        for (int i = n-1; i >= 0; --i) {
            int m = G[i].size();
            G[i].resize(M);
            for (int j = M-1; j >= 0; --j) {
                vector<ll> nf(1<<M, inf);
                for (int s = 0; s < (1<<M); ++s) {
                    {   // choose 
                        int ns = s|(1<<j);
                        if (j-1 >= 0) ns |= (1<<(j-1));
                        if (j-2 >= 0) ns |= (1<<(j-2));
                        gmin(nf[ns], f[s] + c[G[i][j]]);
                    }
                    if (i<n-1 && j<m-2 && (~s>>j&1)) {   // not choose
                    } else {
                        int ns = s&~(1<<j);
                        gmin(nf[ns], f[s]);
                    }
                }
                f = move(nf);
            }
        }
        return *min_element(f.begin(), f.end());
    };

    for (int i = 1; i <= n; ++i) {
        if (vis[i]) {
            continue;
        }
        vector<vector<int>> G;
        for (int x = i; x <= n; x *= 3) {
            G.push_back(vector<int>());
            for (int y = x; y <= n; y *= 2) {
                G.back().push_back(y);
                vis[y] = 1;
            }
        }
        ans += dp(G);
    }

    cout << ans << "\n";
}
```
