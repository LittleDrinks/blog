## [P15255 [USACO26JAN2] Moo Hunt B](https://www.luogu.com.cn/problem/P15255)

考虑容斥
把 $(x,y,z)$ 看作是 $2^x$ 的子集得一分、$2^x +2^y$ 和 $2^x+2^z$ 的子集减一分，$2^x+2^y+2^z$ 的子集加一分
高位前缀和即可

[***AC 代码***](https://www.luogu.com.cn/record/263189434)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int N = 21;
int n, k, f[1<<N];
int ans, anscnt;

int main()
{
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    cin >> n >> k;
    vector<tuple<int,int,int>> p;
    for (int i = 1; i <= k; ++i) {
        int x, y, z;
        cin >> x >> y >> z;
        --x; --y; --z;
        int bx = 1<<x;
        int by = 1<<y;
        int bz = 1<<z;
        f[bx]++;
        f[bx+by]--;
        f[bx+bz]--;
        f[bx+by+bz]++;
    }
    for (int k = 0; k < 19; ++k) {
        for (int i = 0; i < (1<<n); ++i) {
            if (i >> k & 1) {
                f[i] += f[i ^ (1<<k)];
            }
        }
    }
    for (int i = 0; i < (1<<n); ++i) {
        if (f[i] > ans) {
            ans = f[i];
            anscnt = 0;
        }
        if (f[i] == ans) {
            ++anscnt;
        }
    }
    cout << ans << " " << anscnt << "\n";
}
```
