## [P4093 \[HEOI2016/TJOI2016\] 序列](https://www.luogu.com.cn/problem/P4093)

修改 LIS 的定义之后，永远从最原始的式子出发
最原始的 LIS 的做法是：设 $f(i)$ 为以 $i$ 结尾的 LIS 的长度，则
$$
f(i) = \max_{j<i,a_j\le a_i} \bigg\{f(j)\bigg\}+1
$$
维护每个点能取到的最小值 $lv(i)$ 和 $rv(i)$
这道题中只用考虑“修改让 LIS 变劣”的情况，最多修改一次应该理解为每一处修改之后都不劣，也就是只有当 $i<j$，$a_i\le lv(j)$，$rv(i)\le a_j$ 同时满足时才能转移，否则在 $i$ 或 $j$ 修改就能让答案变劣。
相当于有三条偏序关系，我们可以采用中序 cdq 维护。

---

先按下标排序，从上到下递归时遇到区间 $[l,m)$ 和 $[m,r)$：

- 此时先递归左区间，表示左侧 DP 转移完成
- 让 $[l,m)$ 按照 $a_i$ 排序，让 $[m,r)$ 按照 $lv(j)$ 排序
    - 对于每个 $j$，双指针求出 $a_i\le lv(j)$ 的区间
    - 把这个区间上信息插到以 $rv(i)$ 为下标的树状数组上，值为 $f(i)$
    - 求前缀 max 再加一就可以得到 $f(j)$ 的最优值
- 让 $[m,r)$ 按照下标排序，递归右区间

时间复杂度 $O(n\log^2n)$

[***AC 代码***](https://www.luogu.com.cn/record/295083796)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int N = 1e5+5;
const int inf = 1e9;

struct BIT {
    int n;
    vector<int> t;
    BIT(int n): n(n), t(n+1, -inf) {}
    int lowbit(int x) {
        return x & -x;
    }
    void modify(int x, int d) {
        for (; x <= n; x += lowbit(x)) gmax(t[x], d);
    }
    void reset(int x) {
        for (; x <= n; x += lowbit(x)) t[x] = -inf;
    }
    int query(int x) {
        int res = -inf;
        for (; x; x -= lowbit(x)) gmax(res, t[x]);
        return res;
    }
};

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int n, m;
    cin >> n >> m;

    vector<array<int,5>> arr(n);
    for (int i = 0; i < n; ++i) {
        auto &[p, a, lv, rv, f] = arr[i];
        p = i+1;
        cin >> a;
        lv = rv = a;
        f = 1;
    }
    for (int i = 1; i <= m; ++i) {
        int p, x;
        cin >> p >> x;
        auto &[_, a, lv, rv, f] = arr[p-1];
        gmin(lv, x);
        gmax(rv, x);
    }
    BIT t(N);
    auto cdq = [&](auto &&cdq, int l, int r) -> void {
        if (l == r-1) {
            return;
        }
        int m = (l + r) >> 1;
        cdq(cdq, l, m);
        sort(arr.begin()+l, arr.begin()+m, [&](auto x, auto y){ return x[1]<y[1]; });
        sort(arr.begin()+m, arr.begin()+r, [&](auto x, auto y){ return x[2]<y[2]; });
        int i=l, j=m;
        while (j < r) {
            while (i < m && arr[i][1] <= arr[j][2]) {
                t.modify(arr[i][3], arr[i][4]);
                ++i;
            }
            gmax(arr[j][4], t.query(arr[j][1]) + 1);
            ++j;
        }
        for (int k = l; k < i; ++k) {
            t.reset(arr[k][3]);
        }
        sort(arr.begin()+m, arr.begin()+r);
        cdq(cdq, m, r);
    };
    cdq(cdq, 0, n);

    int ans = 0;
    for (int i = 0; i < n; ++i) {
        gmax(ans, arr[i][4]);
    }
    cout << ans << "\n";
}
```
