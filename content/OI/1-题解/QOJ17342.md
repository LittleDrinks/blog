## [QOJ17342. North and South](https://qoj.ac/contest/3729/problem/17342)

最后变成的那个值一定是 $tar=(\sum_{i=1}^na_i) / n$，如果这个数不是整数，答案一定是 `-1`
先只进行长度为 2 的操作。从左到右依次操作，如果 $a_i>tar$ 则答案为 `-1`，否则进行 $f(i)=tar-a_i$ 次操作。
显然，如果只进行长度为 2 的操作，这就是最少的操作步数
接下来将奇偶相同的位置的操作合并起来，每个位置的贡献是 $\max(0,f(i)-f(i-2))$

[***AC 代码***](https://qoj.ac/submission/2342093)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }


void solve() {
    int n;
    cin >> n;
    vector<ll> a(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }
    ll tar = accumulate(a.begin(), a.end(), 0LL);

    if (n == 1) {
        cout << "0\n";
        return;
    }
    
    if (tar % n != 0) {
        cout << "-1\n";
        return;
    }

    tar /= n;
    if (a[1] > tar) {
        cout << "-1\n";
        return;
    }

    vector<ll> f(n + 1);
    for (int i = 1; i <= n; ++i) {
        f[i] = tar - a[i];
        if (f[i] < 0) {
            cout << "-1\n";
            return;
        }
        if (i + 1 <= n) {
            a[i + 1] -= f[i];
        }
    }

    ll ans = f[1] + f[2];
    for (int i = 1; i + 2 <= n; i += 2) {
        ans += max(0LL, f[i + 2] - f[i]);
    }
    for (int i = 2; i + 2 <= n; i += 2) {
        ans += max(0LL, f[i + 2] - f[i]);
    }
    cout << ans << "\n";

}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
