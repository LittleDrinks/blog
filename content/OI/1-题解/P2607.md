## [P2607 \[ZJOI2008\] 骑士](https://www.luogu.com.cn/problem/P2607)

在基环树的环上删掉一条边 $(x,y)$，破环为链变成一棵树，然后就是 [[AcWing285|没有上司的舞会]]
对每个基环树，分别以 $x,y$ 为根跑两遍 DP，取 $f(x,0)$ 和 $f(y,0)$ 中较大的一个即可
小细节是，可能出现重边，所以“删边”的操作需要按边的序号来，而不能按边的端点来

[***AC 代码***](https://www.luogu.com.cn/record/296252336)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int n;
    cin >> n;
    vector<int> e;
    vector G(n+1, vector<int>());
    vector<ll> a(n+1);
    for (int i = 1; i <= n; ++i) {
        int p;
        cin >> a[i] >> p;
        G[i].push_back(e.size());
        e.push_back(p);
        G[p].push_back(e.size());
        e.push_back(i);
    }

    int x, y, skip = 0, totdfn = 0;
    vector<int> dfn(n+1);
    auto dfs1 = [&](auto &&dfs1, int u) -> void {
        dfn[u] = ++totdfn;
        for (auto id: G[u]) {
            int v = e[id];
            if (!dfn[v]) {
                dfs1(dfs1, v);
            } else if (dfn[u] < dfn[v]) {
                x = u;
                y = v;
                skip = id / 2;
            }
        }
    };

    vector<array<ll,2>> f(n+1);
    auto dfs2 = [&](auto &&dfs2, int u, int fa) -> void {
        f[u][1] = a[u];
        f[u][0] = 0;
        for (auto id: G[u]) {
            int v = e[id];
            if (v == fa || id / 2 == skip) continue;
            dfs2(dfs2, v, u);
            f[u][0] += max(f[v][0], f[v][1]);
            f[u][1] += f[v][0];
        }
    };

    ll ans = 0;
    for (int i = 1; i <= n; ++i) {
        if (!dfn[i]) {
            x = y = i;
            skip = 0;
            ll res = 0;
            dfs1(dfs1, i);
            dfs2(dfs2, x, 0);
            gmax(res, f[x][0]);
            swap(x, y);
            dfs2(dfs2, x, 0);
            gmax(res, f[x][0]);
            ans += res;
        }
    }

    cout << ans << "\n";
}
```
