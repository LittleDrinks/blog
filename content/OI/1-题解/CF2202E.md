## [CF2022E. Rigged Bracket Sequence](https://codeforces.com/contest/2202/problem/E)

括号序列的经典思路，把 $\texttt{(}$ 看成 $+1$，把 $\texttt{)}$ 看成 $-1$，观察右移操作对前缀和的影响。
手玩一些例子可以发现两个性质：

- 以 $\texttt($ 结尾的子序列一定合法，因为他移到开头会让最开头那段前缀和的变化量为 $0$ 或 $+2$；
- 以 $\texttt{)}$ 结尾的子序列不一定合法，其合法性取决于中间每一段 $\texttt{()}$ 结构。具体而言，不妨设 $\texttt{()}$ 结构中的左括号下标为 $i$，右括号下标为 $j$，右移之后 $[i,j)$ 这段区间上的前缀和会 $-2$。

我们只要数出以 $\texttt{)}$ 结尾的合法子序列数量即可，据此 DP。设 $f(i)$ 表示以 $i$ 结尾的合法子序列数量，这里我们只考虑 $s_i=\texttt{)}$ 的那些 $i$。$f(i)$ 有三种来源。

- 只选自己，这部分为 $1$
- 选择一个以 $)$ 结尾的子序列拼在后面，因为不构成 $\texttt{()}$，一定可以拼
- 选择一个以 $\texttt{(}$ 结尾的子序列拼在后面，从 $\texttt{(}$ 到这个 $\texttt{)}$ 这个左闭右开的区间前缀和都不小于 $2$

目测是一个区间累加问题，采用刷表法，用差分数组做。
我们先用 $O(n)$ 预处理出每个 $i$ 右侧第一个前缀和小于 $2$ 的位置 $R_i$
设我们此时枚举到了 $i$，维护两个数 $sum$ 和 $diff$。前者是 $s_i=\texttt{)}$ 那些位置 $f(i)$ 的和，表示以 $\texttt{)}$ 结尾的子序列。后者是差分数组前缀和，表示以 $\texttt($ 结尾且到这里依然合法的子序列，算法如下：

- 累加差分数组，更新 $diff$
- 按照之前说的三部分更新，$f(i)=1+sum+diff$
- $s_i=\texttt{)}$：$f(i)$ 保持原来的含义，直接把 $f(i)$ 累加入 $sum$ 和答案即可
- $s_i=\texttt{(}$：先直接把 $2^{i-1}$ 次方累加入答案。此时 $f(i)$ 表示以 $\texttt{)}$ 结尾且符合前缀和最小值限制的子序列。保持 $f(i)$ 不动，把这个 $f(i)$ 通过差分数组累加到区间 $[i+1,R_i]$ 上。注意这里注意 $R_i$ 是需要累加上的，正如上文提到的，这里是一个左闭右开的区间

这样，我们就在 $O(n)$ 的时间复杂度内解决了这个问题。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int MOD = 998244353, N = 1e6 + 5;
template<class...T> ll add(T...t){ll r=0;(((r+=t%MOD)>=MOD?r-=MOD:0),...);return(r%MOD+MOD)%MOD;}
template<class...T> ll mul(T...t){ll r=1;((r=r*(t%MOD)%MOD),...);return(r%MOD+MOD)%MOD;}
ll del(ll x,ll y){return add(x,MOD-y);}
ll qpow(ll a,ll b=MOD-2){ll r=1;for(;b;b>>=1,a=mul(a,a))if(b&1)r=mul(r,a);return r;}
ll fac[N + 5], Inv[N + 5];
void preprocess() {
    fac[0] = 1;
    for (int i = 1; i <= N; ++i) fac[i] = mul(fac[i-1], i);
    Inv[N] = qpow(fac[N]);
    for (int i = N - 1; ~i; --i) Inv[i] = mul(Inv[i + 1], i + 1);
}
ll C(ll n, ll m) { return m > n ? 0: mul(fac[n], Inv[m], Inv[n-m]); }
ll P(ll n, ll m) { return m > n ? 0: mul(fac[n], Inv[n-m]); }

void solve()
{
    int n;
    string s;
    cin >> n >> s;
    vector<int> a(n + 1);
    for (int i = 0; i < n; ++i) {
        a[i + 1] = (s[i] == '(' ? 1 : -1);
    }
    partial_sum(a.begin(), a.end(), a.begin());

    vector<int> R(n + 1);
    for (int i = 1, j = 1; i <= n; ++i) {
        gmax(j, i);
        while (j <= n && a[j] >= 2) {
            ++j;
        }
        R[i] = j;
    }

    int ans = 0;
    int sum = 0, diff = 0;
    vector<int> f(n + 2), g(n + 2);
    for (int i = 1; i <= n; ++i) {
        diff = add(diff, g[i]);
        f[i] = add(1, sum, diff);
        if (s[i - 1] == ')') {
            ans = add(ans, f[i]);
            sum = add(sum, f[i]);
        }
        else {
            ans = add(ans, qpow(2, i - 1));
            int l = i + 1;
            int r = min(n, R[i]);
            if (l <= r) {
                g[l] = add(g[l], f[i]);
                g[r + 1] = del(g[r + 1], f[i]);
            }
        }
    }

    cout << ans << "\n";
}

int main()
{
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int T = 1;
    cin >> T;
    preprocess();
    while (T--) {
        solve();
    }
}
```
