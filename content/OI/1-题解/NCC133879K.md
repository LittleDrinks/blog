## [NCC133879K. Decomposition Trees](https://ac.nowcoder.com/acm/contest/133879/K)

切入点是根据某个点到点分树根的距离做 DP
先钦定树根为 $1$
设 $f(u,d)$ 表示以 $u$ 为根的子树内做点分治，重构出一个点分树，且 $u$ 在点分树上的深度为 $d$（根为 $1$）的方案数
设 $g(u,d)$ 表示在上述场景中，$u$ 在点分树中深度 $\ge d$ 的方案数；$g$ 同时具备另一个含义，即在点分树上任选一段从树根出发长度为 $d$ 的路径的方案数
考虑转移
$$
f(u,d)=\sum_{i_1+i_2+\ldots+i_m=d-1}{\binom{d-1}{i_1, i_2, \ldots, i_m}}\left(\prod_{j=1}^mg(son_u[j],i_j)\right)
$$
这个式子表示从每个子树 $v$ 中选出一个长度为 $i_j$ 的路径，将这些节点重排成 $u$ 到点分树根的路径，用一个多重排列决定父子关系，再乘上 $g$ 表示选择这种路径的方案数
$$
g(u,d)=\sum_{s= d}^nf(u,s)
$$
上述的 $f,g$ 可以用树背包维护，时间复杂度 $O(n^2)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int N = 3005, MOD = 1e9 + 7;
ll fac[N], Inv[N];
ll add(ll x, ll y) { return (x + y) % MOD; }
ll del(ll x, ll y) { return (x - y + MOD) % MOD; }
ll mul(ll x, ll y) { return (x * y) % MOD; }
ll qpow(ll a, ll b = MOD - 2) {
    ll res = 1;
    for (; b; b >>= 1, a = mul(a, a)) {
        if (b & 1) {
            res = mul(res, a);
        }
    }
    return res;
}
ll C(int n, int k) { return k > n ? 0 : mul(mul(Inv[n-k], Inv[k]), fac[n]); }

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int n;
    cin >> n;

    fac[0] = Inv[0] = 1;
    for (int i = 1; i <= n; ++i) {
        fac[i] = mul(fac[i-1], i);
    }
    Inv[n] = qpow(fac[n]);
    for (int i = n-1; i; --i) {
        Inv[i] = mul(Inv[i+1], i+1);
    }

    vector G(n + 1, vector<int>());

    for (int i = 1; i < n; ++i) {
        int u, v;
        cin >> u >> v;
        G[u].push_back(v);
        G[v].push_back(u);
    }

    vector<int> siz(n + 1, 1);
    vector f(n + 1, vector<int>(2));
    vector g(n + 1, vector<int>(2));

    auto dfs = [&](auto &&dfs, int u, int fa) -> void {
        f[u][1] = 1;

        for (auto v: G[u]) {
            if (v == fa) {
                continue;
            }
            dfs(dfs, v, u);
            vector<int> h(siz[u] + siz[v] + 1);
            for (int i = 1; i <= siz[u]; ++i) {
                for (int j = 0; j <= siz[v]; ++j) {
                    ll x = mul(f[u][i], C(i+j-1, j));
                    x = mul(x, g[v][j]);
                    h[i+j] = add(h[i+j], x);
                }
            }
            siz[u] += siz[v];
            f[u] = move(h);
        }

        g[u].resize(siz[u] + 1);
        g[u][siz[u]] = f[u][siz[u]];
        for (int s = siz[u]-1; s >= 0; --s) {
            g[u][s] = add(g[u][s+1], f[u][s]);
        }
    };

    dfs(dfs, 1, 0);

    cout << g[1][0] << "\n";
}
```
