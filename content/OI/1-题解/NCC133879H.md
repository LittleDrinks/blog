## [NCC133879H. String](https://ac.nowcoder.com/acm/contest/133879/H)

操作肯定是将拼接后的字符串开头若干位的非 $\texttt{a}$ 字符替换为 $\texttt{a}$
最后答案字符串由三部分组成：

1. 将若干完整字符串替换为 $\texttt{a}$，由此构成的前缀 $\texttt{a}$
2. 某个单个的字符串，可能部分前缀被替换，这个字符串是前缀 $\texttt{a}$ 和后续字符串的分界线
3. 若干个没有修改的字符串拼接

在这种分割中，“未修改的子串”一定是以非 $\texttt{a}$ 字符开头的，我们维护一个 pair，第一维表示前缀 $\texttt{a}$ 的长度，第二维直接存储后缀，按第一维从大到小，第二维从小到大就可以比较答案

对于第三部分，一个经典的贪心是根据 $s_i+s_j<s_j+s_i$ 排序
据此，我们将所有字符串按照这个标准排序，然后 DP
设 $f(i,j,b)=\{|aaa\cdots a|,s_{rem}\}$，表示现在枚举到第 $i$ 个字符串，已经进行了 $j$ 次替换，分界字符串有没有定下来。我们对每个状态存储一个 pair 表示前缀 $\texttt{a}$ 的长度和剩余的字符串。
对于第 $i$ 个字符串，我们有三种操作方式：

1. 全都不替换，$f(i-1,j,b)+\{0,s_i\}\to f(i,j,b)$
2. 全都替换，$f(i-1,j-|s_i|,b)+\{|s_i|,\text{""} \}\to f(i,j,b)$
3. 将其作为分界线，此时可以部分替换，也可以不替换，枚举替换次数 $0\le h<\#_{\neq a}(s_i)$，设第 $h+1$ 个非 $\texttt{a}$ 字符出现的位置是 $p$，则 $f(i-1,j-h,0)+\{p-1,s[p:]\}\to f(i,j,1)$

这样 DP 做完，$f(n,i,1)$ 中存放的就是形如【前缀 $\texttt{a}$、分界字符串、未修改的字符串】这样的答案，即为所求。
一共 $O(nL)$ 个状态，每个状态转移取 min 时需要花 $O(L)$ 比较答案，时间复杂度 $O(nL^2)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

pair<int,string> operator+(const pair<int,string> &a, const pair<int,string>& b) {
    return { a.first+b.first, a.second+b.second };
}

void solve() {
    int n, L = 0;
    cin >> n;
    vector<string> s(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> s[i];
        L += s[i].size();
    }

    sort(s.begin() + 1, s.end(), [](string a, string b){
        return a + b < b + a;
    });

    vector f(n + 1, vector(L + 1, vector<pair<int,string>>(2)));

    for (int i = 1; i <= n; ++i) {

        int c = s[i].size() - count(s[i].begin(), s[i].end(), 'a');

        for (int j = 0; j <= L; ++j) {

            f[i][j][0] = f[i-1][j][0] + make_pair(0, s[i]);
            f[i][j][1] = f[i-1][j][1] + make_pair(0, s[i]);

            if (j >= c) {
                gmin(f[i][j][0], f[i-1][j-c][0] + make_pair(-s[i].size(), ""));
                gmin(f[i][j][1], f[i-1][j-c][1] + make_pair(-s[i].size(), ""));
            }

            int pos = -1;
            for (int k = 0, cnt = 0; k < c; ++k) {
                while (pos + 1 < s[i].size() && cnt < k + 1) {
                    cnt += (s[i][pos+1] != 'a');
                    ++pos;
                }
                if (j >= k) {
                    gmin(f[i][j][1], make_pair(
                        f[i-1][j-k][0].first-pos,
                        s[i].substr(pos)+f[i-1][j-k][0].second));
                }
            }
        }
    }

    for (int i = 0; i <= L; ++i) {
        auto ans = f[n][i][0];
        gmin(ans, f[n][i][1]);
        // cerr << ans << "\n";
        for (int i = 1; i <= -ans.first; ++i) {
            cout << "a";
        }
        cout << ans.second << "\n";
    }
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
