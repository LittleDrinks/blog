## [P3521 \[POI 2011\] ROT-Tree Rotations](https://www.luogu.com.cn/problem/P3521)

每个节点开一棵权值线段树，存放出现次数。
每次合并左儿子 $p$ 和右儿子 $q$ 的线段树。对于区间 $[l,r)$，线段树合并时会将 $[l,m)$ 和 $[m,r)$ 合并，此时计算左跨右的逆序数，不交换时为 $cnt[lson_p]\times cnt[rson_q]$，交换时为 $cnt[lson_q]\times cnt[rson_p]$。
然后递归两侧，继续在合并的同时计算。

[***AC 代码***](https://www.luogu.com.cn/record/297179104)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

#define m ((l+r)>>1)
struct segmentTree {
    vector<int> a, lson, rson;
    segmentTree(int cap = 0): a(1), lson(1), rson(1) {
        a.reserve(cap);
        lson.reserve(cap);
        rson.reserve(cap);
    }
    int newNode() {
        a.push_back({});
        lson.push_back({});
        rson.push_back({});
        return a.size()-1;
    }
    void pushup(int p) {
        a[p] = a[lson[p]] + a[rson[p]]; 
    }
    int modify(int p, int l, int r, int x, int v) {
        if (x+1 <= l || r <= x) return p;
        if (!p) p = newNode();
        if (l == r-1) {
            a[p] += v;
            return p;
        }
        lson[p] = modify(lson[p], l, m, x, v);
        rson[p] = modify(rson[p], m, r, x, v);
        pushup(p);
        return p;
    }
    int merge(int p, int q, ll &u, ll &v) {
        if (!p || !q) return p | q;
        a[p] += a[q];
        u += 1LL * a[lson[p]] * a[rson[q]];
        v += 1LL * a[lson[q]] * a[rson[p]];
        lson[p] = merge(lson[p], lson[q], u, v);
        rson[p] = merge(rson[p], rson[q], u, v);
        pushup(p);
        return p;
    }
};
#undef m

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int n;
    cin >> n;
    int tot = 1;
    vector<int> p(2*n), lson(2*n), rson(2*n), val(2*n);
    ll ans = 0;
    segmentTree t(n*20);
    vector<int> root(2*n);
    auto dfs1 = [&](auto &&dfs1, int u) -> int {
        cin >> val[u];
        if (!val[u]) {
            lson[u] = dfs1(dfs1, ++tot);
            rson[u] = dfs1(dfs1, ++tot);
            p[lson[u]] = u;
            p[rson[u]] = u;
            ll inv1=0, inv2=0;
            root[u] = t.merge(root[lson[u]], root[rson[u]], inv1, inv2);
            ans += min(inv1, inv2);

        } else {
            root[u] = t.modify(root[u], 1, n+1, val[u], 1);
        }
        return u;
    };
    dfs1(dfs1, tot);
    
    cout << ans << "\n";
}
```
