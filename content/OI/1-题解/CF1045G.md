## [CF1045G. AI robots](https://codeforces.com/problemset/problem/1045/G)

按照 $r$ 排序，从大到小做，每次只考虑 $r$ 小的和 $r$ 大的机器人对话，这样如果能对话，反过来一定可以
高维偏序问题，考虑 cdq 分治。
先按 $r$ 排序，考虑区间 $[l,m)$ 和 $[m,r)$，由于智商差是固定的 $k$，所以我们对左右两个子区间按照 $q$ 排序，利用单调性双指针一下，就可以求出左边某个机器人 $q_i$ 可以接受的智商区间了
我们用一棵树状数组，下标是坐标 $x$，存储这个位置有多少机器人，双指针的时候维护出智商区间对应的树状数组，然后对 $[x_i-R_i, x_i+R_i]$ 区间求和即可。
时间复杂度 $O(n\log^2n)$

[***AC 代码***](https://codeforces.com/contest/1045/submission/388240875)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

struct BIT {
    int n;
    vector<int> t;
    BIT(int n): n(n), t(n + 1) { }
    int lowbit(int x) {
        return x & -x;
    }
    void modify(int x, int d) {
        for (; x <= n; x += lowbit(x)) t[x] += d;
    }
    int query(int x) {
        int res = 0;
        for (; x; x -= lowbit(x)) res += t[x];
        return res;
    }
};

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int n, d;
    cin >> n >> d;
    ll ans = 0;
    vector<array<int,3>> p(n);
    vector<int> v;
    for (auto &[r, q, x]: p) {
        cin >> x >> r >> q;
        v.push_back(x);
        v.push_back(x+r);
        v.push_back(x-r);
    }

    sort(v.begin(), v.end());
    v.erase(unique(v.begin(), v.end()), v.end());
    auto gid = [&](int x) {
        return lower_bound(v.begin(), v.end(), x) - v.begin() + 1;
    };

    BIT t(v.size() + 1);
    sort(p.begin(), p.end());

    auto cdq = [&](auto &&cdq, int l, int r) -> void {
        if (l == r - 1) {
            return;
        }
        int m = (l + r) >> 1;
        cdq(cdq, l, m);
        cdq(cdq, m, r);
        int i = l, j = m, k = m;
        while (i < m) {
            while (j < r && p[j][1] <= p[i][1] + d) {
                t.modify(gid(p[j][2]), 1);
                ++j;
            }
            while (k < r && p[k][1] < p[i][1] - d) {
                t.modify(gid(p[k][2]), -1);
                ++k;
            }
            ans += t.query(gid(p[i][2]+p[i][0])) - t.query(gid(p[i][2]-p[i][0])-1);
            ++i;
        }
        for (int a = k; a < j; ++a) {
            t.modify(gid(p[a][2]), -1);
        }

        sort(p.begin()+l, p.begin()+r, [&](auto x, auto y){
            return x[1] < y[1];
        });
    };

    cdq(cdq, 0, n);
    cout << ans << "\n";
}
```
