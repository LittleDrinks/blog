## [CF2252F. Spectral Components](https://codeforces.com/contest/2252/problem/F)

如果只考虑一种颜色，可以考虑一个类似拓扑排序的贪心算法，每次从叶子向里开始删节点，每删一条边，最终的答案加上这条边下方的该颜色点的数量，每次贪心取最小即可。
建出虚树，跳过对同一个点的若干次删边过程，我们将其压缩为一次性从虚树上某个节点跳到父节点。
在虚树上跑这个算法，复杂度即为 $O(n\log n)$

[***AC 代码***](https://codeforces.com/contest/2252/submission/387127791)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int N = 2e5+5, I=20;
int n;
int c[N], k[N];
vector<int> G[N];
int dep[N], f[I+5][N], dfn[N], totdfn;

void dfs(int u, int fa) {
    dfn[u] = ++totdfn;
    f[0][u] = fa;
    dep[u] = dep[fa] + 1;
    for (int i = 1; i <= I; ++i) { 
        f[i][u] = f[i-1][ f[i-1][u] ];
    }
    for (auto v: G[u]) {
        if (v != fa) {
            dfs(v, u);
        }
    }
}

int LCA(int u, int v)
{   // 求 u,v 到其 LCA 路径上的最小值
    function<void(int&,int)> jump = [&](int &u, int i) {
        u = f[i][u];
    };
    if (dep[u] < dep[v]) { swap(u, v); }
    for (int i = I; i >= 0; --i) {
        if (dep[f[i][u]] >= dep[v]) { jump(u, i); }
    }
    if (u == v) { return u; }
    for (int i = I; i >= 0; --i) {
        if (f[i][u] != f[i][v]) {
            jump(u, i); jump(v, i);
        }
    }
    jump(u, 0); jump(v, 0);
    return u;
}

void solve() {
    cin >> n;
    vector node(n+1, vector<int>());
    for (int i = 1; i <= n; ++i) {
        cin >> c[i];
        node[c[i]].push_back(i);
    }
    for (int i = 1; i <= n; ++i) {
        cin >> k[i];
    }
    for (int i = 1; i < n; ++i) {
        int u, v;
        cin >> u >> v;
        G[u].push_back(v);
        G[v].push_back(u);
    }

    dfs(1, 0);

    auto work = [&](vector<int> h, int siz, int col) -> ll {

        auto cmp = [&](int x, int y) {
            return dfn[x] < dfn[y];
        };

        sort(h.begin(), h.end(), cmp);

        vector<int> a;
        for (int i = 0; i + 1 < (int)(h.size()); ++i) {
            a.push_back(h[i]);
            a.push_back(LCA(h[i], h[i+1]));
        }
        a.push_back(h.back());
        sort(a.begin(), a.end(), cmp);
        a.erase(unique(a.begin(), a.end()), a.end());
        int m = a.size();

        vector<int> v(a);
        sort(v.begin(), v.end());
        // cerr << "\t" << "nodes :: " << v << "\n";

        auto gid = [&](int x) {
            return lower_bound(v.begin(), v.end(), x) - v.begin() + 1;
        };
        
        int cnt = 1;
        vector T(m + 1, vector<pair<int,int>>());
        vector<int> deg(m + 1);
        
        auto add_edge = [&](int u, int v) {
            if (u == v) {
                return;
            }
            int w = abs(dep[u] - dep[v]);
            cnt += w;
            // cerr << "\tbuild " << u << " --- " << v << ", " << w << "\n";
            // cerr << "\t\t" << gid(u) << " " << gid(v) << "\n";
            T[gid(u)].emplace_back(gid(v), w);
            T[gid(v)].emplace_back(gid(u), w);
            ++deg[gid(u)];
            ++deg[gid(v)];
        };

        if (a.size() > 1) {
            add_edge(LCA(a[0], a[1]), a[0]);
        }
        for (int i = 0; i + 1 < int(a.size()); ++i) {
            int lc = LCA(a[i], a[i+1]);
            add_edge(lc, a[i+1]);
        }
        // cerr << "\tcnt = " << cnt << "\n";
        // cerr << "\tdeg :: " << deg << "\n";

        vector<int> wei(m + 1);
        for (int i = 0; i < m; ++i) {
            if (c[v[i]] == col) {
                wei[i+1]++;
            }
        }

        priority_queue<pair<int,int>, vector<pair<int,int>>, greater<>> q;
        for (int i = 1; i <= m; ++i) {
            if (deg[i] <= 1) {
                q.push({1, i});
                // cerr << "\tpush " << a[i-1] << "\n";
            }
        }

        vector<bool> vis(m + 1);
        ll ans = 0;
        while (!q.empty() && cnt > siz) {
            auto [w, u] = q.top();
            // cerr << "\tdel " << v[u-1] << "\n";
            q.pop();

            vis[u] = 1;

            for (auto [v, dis]: T[u]) {
                if (vis[v]) {
                    continue;
                }
                ll t = min(cnt-siz, dis);
                cnt -= t;
                dis -= t;
                ans += t * w;
                wei[v] += wei[u];
                if (--deg[v] == 1) {
                    q.push({wei[v], v});
                }
            }

            // cerr << "siz :: " << cnt << ", " << siz << "\n";
        }

        return ans;
    };
    for (int i = 1; i <= n; ++i) {
        if (node[i].empty()) {
            cout << "-1 ";
        } else {
            cout << work(node[i], k[i], i) << " ";
        }
    }
    cout << "\n";

    totdfn = 0;
    for (int i = 1; i <= n; ++i) {
        c[i] = k[i] = dep[i] = dfn[i] = 0;
        G[i].clear();
        for (int j = 0; j < I; ++j) {
            f[j][i] = 0;
        }
    }
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
