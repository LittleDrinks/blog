## [QOJ17344. Palindromic and Balanced](https://qoj.ac/contest/3729/problem/17344)

一个回文平衡字符串，显然最左边必须是 $\texttt{(}$，最右边必须是 $\texttt{)}$，并且长度为偶数，不然无法平衡
考虑如何生成一个回文平衡字符串，我们每次往最中间插入一对 $\texttt{((}$ 或 $\texttt{))}$，这样能符合回文的条件
而平衡的条件可以这样考虑，初始时前缀和后缀为 $\texttt{(}$ 和 $\texttt{)}$，平衡数分别 $+1$ 和 $-1$，每次插入相当于往前后缀上各自 $+1$ 或 $-1$，我们需要让前缀平衡数 $\ge0$，后缀平衡数 $\le 0$，相当于中间回文串的前半部分平衡数必须为 $\{-1,0,1\}$
考虑 DP，我们先在左右分别取一个 $\texttt{(}$ 和 $\texttt{)}$，然后取中间的字符串进行接下来的操作。
设 $f(i,j,x)$ 表示前 $i$ 个字符，已经取了 $j$ 个，并且平衡数为 $x$ 时，需要从右往左哪一个字符才能找到符合条件的回文子序列。可以通过需处理【下一个/上一个】【左/右】括号进行转移，详见代码。

[***AC 代码***](https://qoj.ac/submission/2342168)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]\n";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int N = 5007;
const int BIAS = 1;
int prl[N], prr[N], nxl[N], nxr[N];
int f[N][N][3];

void solve() {
    int n;
    string s;
    cin >> n >> s;

    bool fr = false;
    reverse(s.begin(), s.end());
    while (!s.empty() && s.back() != '(') { s.pop_back(); }
    if (!s.empty()) { s.pop_back(); fr = true; }

    bool bk = false;
    reverse(s.begin(), s.end());
    while (!s.empty() && s.back() != ')') { s.pop_back(); }
    if (!s.empty()) { s.pop_back(); bk = true; }
    
    if (!fr || !bk) {
        cout << "0\n";
        return;
    }

    if (s.empty()) {
        cout << "2\n";
        return;
    }

    n = s.size();
    assert(n > 0);
    for (int i = 1, lasl=0, lasr=0; i <= n + 1; ++i) {
        prl[i] = lasl;
        prr[i] = lasr;
        if (i > n) break; 
        if (s[i - 1] == '(') lasl = i;
        if (s[i - 1] == ')') lasr = i;
    }
    for (int i = n, lasl=n+1, lasr=n+1; i >= 0; --i) {
        nxl[i] = lasl;
        nxr[i] = lasr;
        if (i == 0) break;
        if (s[i - 1] == '(') lasl = i;
        if (s[i - 1] == ')') lasr = i;
    }

    f[0][0][0+BIAS] = n + 1;
    for (int i = 0; i <= n; ++i) {
        for (int j = 0; j <= i; ++j) {
            for (int x: {-1, 0, 1}) {
                gmax(f[i+1][j][x], f[i][j][x]);
                if (x + 1 <= 1) {  // left br
                    gmax(f[nxl[i]][j + 1][x + 1 + BIAS], prl[f[i][j][x + BIAS]]);
                }
                if (x - 1 >= -1) {  // right br
                    gmax(f[nxr[i]][j + 1][x - 1 + BIAS], prr[f[i][j][x + BIAS]]);
                }
            }
        }
    }
    int ans = 2;
    for (int i = 1; i <= n; ++i) {
        for (int j = 0; j <= i; ++j) {
            if (i < f[i][j][0+BIAS]) {
                gmax(ans, 2 + j * 2);
            }
        }
    }
    cout << ans << "\n";

    for (int i = 0; i <= n; ++i) {
        prl[i] = prr[i] = nxl[i] = nxr[i] = 0;
    }
    for (int i = 1; i <= n; ++i) {
        for (int j = 0; j <= i; ++j) {
            f[i][j][0] = f[i][j][1] = f[i][j][2] = 0;
        }
    }
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
