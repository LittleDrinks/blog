## [NCC133881J](https://ac.nowcoder.com/acm/contest/133881/J)

先考虑不带修怎么做
根据糖水不等式，我们可以先找到 $\dfrac{a_k}{b_k}$ 最大的 $k$，不妨设 $i\ne j$，$\dfrac{a_i}{b_i}\le \dfrac{a_j}{b_j}$，则
$$
\frac{a_i}{b_i}\le \frac{a_i+a_j}{b_i+b_j}\le \frac{a_j}{b_j}\le \frac{a_j+a_k}{b_j+b_k}\le \frac{a_k}{b_k}
$$
据此我们可以固定 $k$ 然后枚举 $j$ 计算目标函数
进一步优化，可以转化为
$$
\frac{a_k-(-a_j)}{b_k-(-b_j)}
$$
也就是固定 $k$，对剩余的点集坐标取反，计算 $k$ 与其中点斜率的最大值。由于坐标取反后一定满足 $k$ 在右上角第一象限，点集在左下角第三象限，不会有 corner case，所以就相当于直接过定点求凸包的切线，可以参考[这里](https://github.com/tarjen/tarjen_cp_template/blob/master/src/geometry/%E5%87%B8%E5%A4%9A%E8%BE%B9%E5%BD%A2.cpp)的 `extreme` 和 `tangent` 函数，可以做到 $O(\log n)$。

接下来考虑带修。把原来的 $n$ 个点分成 $B$ 个块，块长为 $S$，块内维护凸包，每次单点修改时重构凸包，散块和 $k$ 所在的块直接暴力，其余块求凸包切线，复杂度
$$O(q(\underbrace{\log S}_{重构凸包}+\underbrace{B\log S}_{凸包上二分}+\underbrace{S}_{散块直接暴力}))$$
取 $B=\sqrt{n}$ 可以做到 $O(n\sqrt{n}\log n)$ 的复杂度

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const db eps=1e-8;
const db PI=acos(-1.0);
int sgn(db x) { return (x > eps) - (x < -eps); }
int sgn(ll x) { return (x > 0) - (x < 0); }

using T = ll;
struct Point {
	T x, y;
    Point(T x=0, T y=0): x(x), y(y) {}
	friend istream& operator>> (istream& is, Point& p) { return is>>p.x>>p.y; }
	friend ostream& operator<< (ostream& os, const Point& p) { return os<<p.x<<" "<<p.y; }
	db ang() { return atan2(y, x); }
	bool operator== (const Point &p) const { return sgn(x-p.x)==0 && sgn(y-p.y)==0; }
	bool operator< (const Point &p) const { return x<p.x || (x==p.x && y<p.y); }
	Point operator+ (const Point &p) const { return Point(x+p.x, y+p.y); }
	Point operator- (const Point &p) const { return Point(x-p.x, y-p.y); }
	Point operator* (const T &k) const { return Point(k*x, k*y); }
	Point operator/ (const T &k) const { return Point(x/k, y/k); }
	T operator* (const Point &p) const { return x*p.x + y*p.y; }
	T operator^ (const Point &p) const { return x*p.y - y*p.x; }  // 叉乘时打括号
	T len2() { return (*this)*(*this); }
	db len() { return sqrtl(len2()); }  // 等价于 hypotl(x, y)
	Point rot(db ang) { return Point(x*cos(ang)-y*sin(ang), x*sin(ang)+y*cos(ang)); }
	Point trunc(db l) { return (*this) * (l/len()); }
};

// 求平面点集的凸包，conv[0]是最左下角的点，逆时针顺序
// Andrew 算法，时间复杂度 O(nlogn)
vector<Point> toConvex(vector<Point> po) {
	if (po.empty()) { return {}; }
	sort(po.begin(), po.end());
    vector<Point> conv;
	auto check = [&](Point u){
		Point p1=conv.back(), p2=conv[conv.size()-2];
		return sgn((p1-p2)^(u-p1))<=0;
	};
	for (const Point &u: po) {
		while (conv.size()>1 && check(u)) { conv.pop_back(); }
		conv.push_back(u);
	}
	size_t k = conv.size();
	po.pop_back();
    reverse(po.begin(),po.end());
	for (const Point &u: po) {
		while (conv.size()>k && check(u)) { conv.pop_back(); }
		conv.push_back(u);
	}
	conv.pop_back();
    return conv;
}
// 凸多边形关于某一方向的极点
// 求 p[i]，使得 cross(dir(p[i]), p[j]-p[i]) <= 0 对所有 j 均成立
// 复杂度 O(logn)
template<typename F>
size_t extreme(const vector<Point>& p, const F& dir) {
    int n = p.size();
    const auto check = [&](size_t i) {
        return (dir(p[i]) ^ (p[(i + 1) % n] - p[i])) >= 0;
    };
    const auto dir0 = dir(p[0]);
    const auto check0 = check(0);
    if (!check0 && check(n - 1)) return 0;
    const auto cmp = [&](const Point& v) -> bool {
        const size_t vi = &v - p.data();
        if (vi == 0) return true;
        const auto checkv = check(vi);
        const auto t = dir0 ^ (v - p[0]);
        if (vi == 1 && checkv == check0 && t == 0) return true;
        return checkv ^ (checkv == check0 && t <= 0);
    };
    return partition_point(p.begin(), p.end(), cmp) - p.begin();
}
// 过凸多边形外一点求凸多边形的切线，返回切点下标
// 复杂度 O(logn)
// 必须保证点在多边形外
pair<size_t, size_t> tangent(const vector<Point> &p, const Point &a) {
    const size_t i = extreme(p, [&](const Point &u){ return u - a; });
    const size_t j = extreme(p, [&](const Point &u){ return a - u; });
    return {i, j};
}

const ll inf = 2e9;
template<typename T>
struct Frac {
    T num;
    T den;
    void norm() {
        if (den < 0) {
            num = -num; den = -den;
        }
    }
    Frac(T num=-inf, T den=1): num(num), den(den) {
        norm();
    };
    friend ostream& operator<< (ostream& os, const Frac &rhs) {
        ll x = rhs.num;
        ll y = rhs.den;
        ll g = gcd(x, y);
        x /= g;
        y /= g;
        if (y < 0) {
            x=-x; y=-y;
        }
        return os << x << " " << y;
    }
    friend Frac operator+ (const Frac& lhs, const Frac& rhs) {
        return Frac(lhs.num*rhs.den + rhs.num*lhs.den, lhs.den*rhs.den);
    }
    friend Frac operator- (const Frac& lhs, const Frac &rhs) {
        return Frac(lhs.num*rhs.den - rhs.num*lhs.den, lhs.den*rhs.den);
    }
    friend bool operator< (const Frac& lhs, const Frac& rhs) {
        return lhs.num * rhs.den < rhs.num * lhs.den;
    }
};

const int N = 1e5 + 5;
int n, q, siz, B;
Point a[N];

// arr 1-index <-> bk 1-index
inline int bk(int i) { return i / siz + 1; }
inline int bkl(int b) { return max(1, (b - 1) * siz); }
inline int bkr(int b) { return min(b * siz - 1, n); }

#define lson (p << 1)
#define rson (p << 1 | 1)
#define m ((l + r) >> 1)
pair<Frac<ll>, int> t[N<<2];
void pushup(int p) {
    t[p] = max(t[lson], t[rson]);
}
void build(int p, int l, int r) {
    if (l == r-1) {
        t[p] = make_pair(Frac<ll>(a[l].x, a[l].y), l);
        return;
    }
    build(lson, l, m);
    build(rson, m, r);
    pushup(p);
}
pair<Frac<ll>,int> query(int p, int l, int r, int x, int y) {
    if (x <= l && r <= y) {
        return t[p];
    }
    if (y <= l || r <= x) {
        return make_pair(Frac<ll>(-inf, 1), -1);
    }
    pair<Frac<ll>, int> res;
    gmax(res, query(lson, l, m, x, y));
    gmax(res, query(rson, m, r, x, y));
    pushup(p);
    return res;
}
void modify(int p, int l, int r, int x) {
    if (l == r - 1) {
        t[p] = make_pair(Frac<ll>(a[l].x, a[l].y), l);
        return;
    }
    if (x+1 <= l || r <= x) {
        return;
    }
    modify(lson, l, m, x);
    modify(rson, m, r, x);
    pushup(p);
}
#undef lson
#undef rson
#undef m

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    cin >> n >> q;
    siz = sqrt(n) + 1;
    B = bk(n);
    for (int i = 1; i <= n; ++i) {
        cin >> a[i].x;
    }
    for (int i = 1; i <= n; ++i) {
        cin >> a[i].y;
    }

    vector<vector<Point>> conv(B+1);
    auto process = [&](int i) {
        int L=bkl(i), R=bkr(i);
        vector<Point> u;
        u.reserve(R-L+1);
        for (int i = L; i <= R; ++i) {
            u.emplace_back(-a[i].x, -a[i].y);
        }
        conv[i] = toConvex(move(u));
    };

    for (int i = 1; i <= B; ++i) {
        process(i);
    }

    build(1, 1, n+1);

    while (q--) {
        int o;
        cin >> o;
        if (o == 1) {
            int i, x;
            cin >> i >> x;
            a[i].x = x;
            process(bk(i));
            modify(1, 1, n+1, i);
        } else if (o == 2) {
            int i, y;
            cin >> i >> y;
            a[i].y = y;
            process(bk(i));
            modify(1, 1, n+1, i);
        } else {
            int l, r;
            cin >> l >> r;
            auto k = query(1, 1, n+1, l, r+1);
            Frac<ll> ans;
            // for (int i = l; i <= r; ++i) {
            //     if (i == k.second) continue;
            //     cerr << i << "\t" << Frac<ll>(k.first.num+a[i].x, k.first.den+a[i].y) << "\n";
            //     gmax(ans, Frac<ll>(k.first.num+a[i].x, k.first.den+a[i].y));
            // }
            if (bk(l) == bk(r)) {
                for (int i = l; i <= r; ++i) {
                    if (i == k.second) continue;
                    gmax(ans, Frac<ll>(k.first.num+a[i].x, k.first.den+a[i].y));
                }
            } else {
                for (int i = l; i <= bkr(bk(l)); ++i) {
                    if (i == k.second) continue;
                    gmax(ans, Frac<ll>(k.first.num+a[i].x, k.first.den+a[i].y));
                }
                for (int i = bkl(bk(r)); i <= r; ++i) {
                    if (i == k.second) continue;
                    gmax(ans, Frac<ll>(k.first.num+a[i].x, k.first.den+a[i].y));
                }
                if (bk(l) < bk(k.second) && bk(k.second) < bk(r)) {
                    for (int i = bkl(bk(k.second)); i <= bkr(bk(k.second)); ++i) {
                        if (i == k.second) continue;
                        gmax(ans, Frac<ll>(k.first.num+a[i].x, k.first.den+a[i].y));
                    }
                }
                for (int b = bk(l)+1; b < bk(r); ++b) {
                    if (b == bk(k.second)) continue;
                    auto id = tangent(conv[b], a[k.second]);
                    gmax(ans, Frac<ll>(k.first.num-conv[b][id.first].x, k.first.den-conv[b][id.first].y));
                    gmax(ans, Frac<ll>(k.first.num-conv[b][id.second].x, k.first.den-conv[b][id.second].y));
                }
            }

            cout << ans << "\n";
        }
    }
}
