## [CF2276E](https://codeforces.com/contest/2276/problem/E)

反过来数符合条件的 bad 子串数
设 $rl_i$ 表示左端点为 $i$ 时，第一次出现 `2025` 的右端点，$rr_i$ 表示第一次出现 `2026` 的右端点
对于位置 $i$，答案为：
$$\sum_{j\le i}\bigg| [rl_j,rr_j)\cap [i,n] \bigg|$$
这本质上是二维偏序，按 $i$ 从左到右做，每次激活 $[1,i]$，把对应的区间加到线段树上，然后求 $[i,n]$ 的区间和
然后考虑 $rl$ 和 $rr$ 怎么求
子序列 DP，从后往前做，$dp[0...3]$ 分别表示 `6`、`26`、`026`、`2026` 最近一次出现在什么时候
转移显然
时间复杂度 $O(n\log n)$

[***AC 代码***](https://codeforces.com/contest/2276/submission/396075178)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

struct BIT {
    int n;
    vector<ll> t;
    BIT(int n): n(n), t(n+1) {}
    int lowbit(int x) { return x & (-x); }
    void modify(int x, ll d) {
        for (; x <= n; x += lowbit(x)) t[x] += d;
    }
    ll query(int x) {
        ll res = 0;
        for (; x; x -= lowbit(x)) res += t[x];
        return res;
    }
};

void solve() {
    int n;
    cin >> n;
    vector<int> a(n+1, -1);
    for (int i = 1; i <= n; ++i) {
        char x;
        cin >> x;
        a[i] = (x - '0');
    }

    vector<int> rl(n+1), rr(n+1);
    vector<int> dp1(4, n+1), dp2(4, n+1);
    for (int i = n; i; --i) {
        if (a[i] == 5) {
            dp1[3] = i;
        } else if (a[i] == 6) {
            dp2[3] = i;
        } else if (a[i] == 2) {
            dp1[2] = dp1[3];
            dp2[2] = dp2[3];
            dp1[0] = dp1[1];
            dp2[0] = dp2[1];
        } else if (a[i] == 0) {
            dp1[1] = dp1[2];
            dp2[1] = dp2[2];
        }
        rl[i] = dp1[0];
        rr[i] = dp2[0];
    }

    BIT s1(n+1), s2(n+1);
    auto presum = [&](int x) -> ll {
        return (x+1)*s1.query(x)-s2.query(x);
    };
    vector<ll> ans(n+1);
    for (int i = 1; i <= n; ++i) {
        ans[i] = 1LL*i*(n-i+1);
        if (rl[i] < rr[i]) {
            s1.modify(rl[i], 1);
            s2.modify(rl[i], rl[i]);
            s1.modify(rr[i], -1);
            s2.modify(rr[i], -rr[i]);
        }
        ll bad = presum(n+1) - presum(i-1);
        ans[i] -= bad;
    }
    for (int i = 1; i <= n; ++i) {
        cout << ans[i] << " \n"[i == n];
    }
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
