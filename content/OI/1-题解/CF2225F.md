## [CF2225F. String Cutting](https://codeforces.com/contest/2225/problem/F)

显然恰好分成 $k$ 份是最优的。如果有超过 $k$ 份，我们一定可以选择前 $k-1$ 小的一个字符串 $A$，让他和旁边的 $B$ 粘合成为 $C$，显然有 $A<C$。我们观察比原先第 $k$ 小【更小的子串数量】，一开始是 $k-1$，合并后显然不会比原来更多。
对于一个子串 $s[l\ldots r]$，他能成为答案有四个条件：

1. 左侧 $s[1\ldots l-1]$ 能被分为至少一份
2. 右侧 $s[r+1\ldots n]$ 能被分为至少一份
3. 中间 $s[l\ldots r]$ 的长度不小于 $l$
4. 左右两侧切分出的每部分字典序都不超过 $s[l,\ldots r]$

前三个条件比较容易 check，我们可以很轻松地得到 $O(n^2)$ 个候选串，接下来有两个观察：

1. 由于 $s[l\ldots r]<s[l\ldots r+1]$，所以对于每一个左端点，我们只需要选择一个最长的子串作为候选
2. 最后的答案是候选串取 max，即使某个候选串（已满足前三个条件）不满足条件，左右侧那个比他大的部分一定会进入候选串集合，最终该子串会在取 max 的时候被淘汰

所以我们得到最终的算法：对于每个左端点 $i$，求出最长的那个满足前三个条件的子串，加入候选集合，最后取 max 即可。
接下来考虑如何实现【子串取 max】。一种做法是打擂台，每次使用哈希+二分在 $O(\log n)$ 的时间内求出两个子串的最长公共前缀，然后比较下一个字符即可。时间复杂度 $O(n\log n)$

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int N = 1e6 + 5, mod1 = 1e9+7, mod2 = 1e9+9;
using hs = pair<ll, ll>;
const hs base{114, 514};
hs P[N + 5];

hs operator+ (const hs &a, const hs &b) {
    return hs{ (a.first+b.first)%mod1, (a.second+b.second)%mod2};
};
hs operator- (const hs &a, const hs &b) {
    return hs{ (a.first-b.first+mod1)%mod1, (a.second-b.second+mod2)%mod2 };
}
hs operator* (const hs &a, const hs &b) {
    return hs{ (a.first*b.first)%mod1, (a.second*b.second)%mod2 };
}

void solve() {
    int n, l, k;
    string s;
    cin >> n >> l >> k >> s;
    
    if (1LL * n < 1LL * k * l) {
        cout << "NO\n";
        return;
    }

    if (k == 1) {
        cout << "YES\n";
        cout << s << "\n";
        return;
    }

    vector<hs> h(n + 1);
    for (int i = 1; i <= n; ++i) {
        h[i] = h[i - 1] * base + hs{s[i-1], s[i-1]};
    }
    
    auto get = [&](int l, int r) -> hs {
        return h[r] - h[l-1] * P[r-l+1];
    };

    auto cmp = [&](int l1, int r1, int l2, int r2) -> bool {
        // return true if s[l1...r1] < s[l2...r2]
        int len = min(r1-l1+1, r2-l2+1);
        int L = 0, R = len + 1;
        while (L + 1 < R) {
            int m = midpoint(L, R);
            if (get(l1, l1+m-1) == get(l2, l2+m-1)) {
                L = m;
            } else {
                R = m;
            }
        }
        char c1 = l1 + L > r1 ? 'a'-1: s[l1 + L - 1];
        char c2 = l2 + L > r2 ? 'a'-1: s[l2 + L - 1];
        return c1 < c2;
    };

    int ansl = 1, ansr = 1;
    for (int i = 1; i <= n; ++i) {
        int pre = (i - 1) / l;
        if (i > 1 && pre == 0) {
            continue;
        }
        int r = n - max(0, k - pre - 1) * l;
        if (r - i + 1 < l) {
            continue;
        }
        if (cmp(ansl, ansr, i, r)) {
            ansl = i;
            ansr = r;
        }
    }

    cout << "YES\n";
    cout << s.substr(ansl-1, ansr-ansl+1) << "\n";
}

int main()
{
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 

    P[0] = {1, 1};
    for (int i = 1; i <= N; ++i) {
        P[i] = P[i-1] * base;
    }

    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
