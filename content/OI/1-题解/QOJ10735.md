## [QOJ10735. Master of Both VII](https://qoj.ac/contest/2021/problem/10735)

如果已知每个点的度数，可以通过拓扑排序构造出一组三角剖分。接下来考虑如何求出度数。
固定顶点 $1$，对于 $i\in[3,n-2]$ 询问 $(1, i)$。
对于顶点 $2$ 和顶点 $n-1$，$query(1,3)$ 和 $query(1,n-2)$ 分别就是它们的度数。
对于剩余的顶点，我们统一考虑逆时针从 $1$ 开始遍历，将其度数分为三类：

- $1\to i$ 上的点，即 $[2,i)$ 与其相连的度数
- $i\to 1$ 上的点，即 $(i,n]$ 与其相连的度数
- $1$ 是否和 $i$ 直连

我们考虑逆时针从 $i$ 走到 $i+1$，两次询问答案的变化：

- 如果询问的答案变大了，说明 $i$ 会越过 $1\to i+1$ 这条线和 $(i,n]$ 相连，此时增加量就是第二部分的度数
- 如果询问的答案变小了，说明 $i+1$ 和 $[1,i)$ 上的点有连线，此时 $i$ 被包在里面，不会有连向 $i\to1$ 上的边

这样我们就可以求出 $i$ 向 $[i+2,n]$ 连接的边的数量。同理顺时针做可以求出 $i$ 向 $[2, i-2]$ 连接的边数。
因此，某一个点的度数应该由如下三部分组成：
$$
\begin{cases}
\max(0, query(i+1)-query(i)) \\
\max(0,query(i-1)-query(i)) \\
 [query(i)=0]
\end{cases}
$$
特别地，对于顶点 $3$，$max(0,query(i-1)-query(i))$ 这一项不存在，顶点 $n-2$ 同理。
这样我们就可以在 $n-3$ 次询问内拿到所有点的度数，然后拓扑排序即可。

[***AC 代码***](https://qoj.ac/submission/2165879)

```cpp
#include<bits/stdc++.h>
using namespace std;

vector<pair<int, int>> ans;
int query(int x, int y) {
    cout << "? " << x + 1 << " " << y + 1 << endl;
    int ans;
    cin >> ans;
    return ans;
}
void answer() {
    cout << "!";
    for (auto [x, y]: ans) {
        cout << " " << x + 1 << " " << y + 1;
    }
    cout << endl;
    int r;
    cin >> r;
    if (r == 0) exit(0);
}
void solve()
{
    int n;
    cin >> n;
    vector<int> q(n);
    vector<int> deg(n);
    for (int i = 2; i <= n - 2; ++i) {
        q[i] = query(0, i);
    }
    deg[1] = q[2];
    deg[n - 1] = q[n - 2];
    for (int i = 2; i <= n - 2; ++i) {
        deg[i] = (q[i] == 0);
        if (i + 1 <= n - 2) deg[i] += max(0, q[i + 1] - q[i]);
        if (i - 1 >= 2)     deg[i] += max(0, q[i - 1] - q[i]);
    }
    for (int i = 1; i < n; ++i) {
        deg[0] += deg[i];
    }
    deg[0] = 2 * (n - 3) - deg[0];
    for (int i = 0; i < n; ++i) {
        // cerr << deg[i] << " \n"[i + 1 == n];
    }

    vector<int> pre(n);
    vector<int> nxt(n);
    for (int i = 0; i < n; ++i) {
        pre[i] = (i - 1 + n) % n;
        nxt[i] = (i + 1 + n) % n;
    }

    queue<int> que;
    for (int i = 0; i < n; ++i) {
        if (deg[i] == 0) {
            que.push(i);
            // cerr << "push " << i+1 << endl;
        }
    }

    int cnt = 0;
    while (cnt < n - 3) {
        ++cnt;
        int u = que.front(); que.pop();
        int x = pre[u];
        int y = nxt[u];
        // cerr << u << " :: " << x << " - " << y << "\n";

        ans.emplace_back(x, y);
        nxt[x] = y;
        pre[y] = x;
        if (--deg[x] == 0) que.push(x);
        if (--deg[y] == 0) que.push(y);
    }

    answer();
    ans.clear();
}
int main()
{
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int T;
    cin >> T;
    while (T--) {
        solve();
    }
    return 0;
}
```
