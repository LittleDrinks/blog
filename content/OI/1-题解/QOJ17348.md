## [QOJ17348. Transform](https://qoj.ac/contest/3729/problem/17348)

先特判一些 corner case

- 当原序列为 $\{k\}$ 时，无需操作，答案为 $0$
- 当元素列 $mex=k$ 时，一步操作就可以得到 $k$
- 当原序列 $mex>k$ 时，$[1,k-1]$ 都出现过了，需要通过一步操作把 $\ge k$ 的都删了，再一步得到 $k$，于是答案为 $2$


接下来只讨论 $mex<k$ 的情况，此时答案至少为 $1$。此时 $\ge k$ 的那些数字不会影响操作，不再考虑。
手玩一下，发现从空序列得到 $\{k\}$ 的操作数是 $2^k$
> [!note] 
> 这可以通过数学归纳法证明，假设对 $1\sim n-1$ 成立，想要凑出 $n$，那就需要
> $$
> 1+\sum_{i=0}^{n-1}2^{i}=2^k
> $$

显然我们可以通过已经有的 $<k$ 的数字节约步数。
从大到小枚举，不妨设当前有 $cnt_x$ 个 $x$，我们需要 $need$ 个数，可以用 $take=\min(need,cnt_x)$ 个 $x$。
对于 $x-1$，当前的 $need$ 来自高位，应该原封不动地下传，同时现在还有 $need-take$ 个 $x$ 未凑出
所以对于 $x-1$，有：
$$
need_{x-1}=need_{x}+(need_{x}-take_{x})
$$
实现时，只要当 $need>n$，就可以视为 $need=\infty$，接下来无需精确计算。

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int MOD = 998244353;
template<class...T> ll add(T...t){ll r=0;(((r+=t%MOD)>=MOD?r-=MOD:0),...);return(r%MOD+MOD)%MOD;}
template<class...T> ll mul(T...t){ll r=1;((r=r*(t%MOD)%MOD),...);return(r%MOD+MOD)%MOD;}
ll del(ll x,ll y){return add(x,MOD-y);}
ll qpow(ll a,ll b=MOD-2){ll r=1;for(;b;b>>=1,a=mul(a,a))if(b&1)r=mul(r,a);return r;}
const int N = 5e5 + 5;
ll pw2[N];

void solve() {
    int n, k;
    cin >> n >> k;
    vector<int> s(n);
    set<int> excl;
    for (int i = 0; i <= n; ++i) {
        excl.insert(i);
    }
    for (int i = 0; i < n; ++i) {
        cin >> s[i];
        excl.erase(s[i]);
    }
    
    if (n == 1 && s[0] == k) {
        cout << "0\n";
        return;
    }
    
    int mex = *excl.begin();
    if (mex == k) {
        cout << "1\n";
        return;
    }
    if (mex > k) {
        cout << "2\n";
        return;
    }

    sort(s.begin(), s.end());
    s.erase(lower_bound(s.begin(), s.end(), k), s.end());

    map<int,int> a;
    for (auto x: s) {
        ++a[x];
    }

    int ans = qpow(2, k);
    int need = 1, last = k;
    bool exc = false;
    for (auto it = a.rbegin(); it != a.rend(); ++it) {
        auto [x, t] = *it;
        int step = last - x - 1;
        for (int i = 1; !exc && i <= step; ++i) {
            need <<= 1;
            if (need > n) {
                exc = true;
                break;
            }
        }
        int take = exc ? t : min(t, need);
        need += need - take;
        if (need > n) {
            exc = true;
        }
        ans = del(ans, mul(take, qpow(2, x)));
        last = x;
    }

    cout << ans << "\n";
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
