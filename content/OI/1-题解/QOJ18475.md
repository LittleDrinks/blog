## [QOJ18475. 星系观测计划](https://qoj.ac/contest/3789/problem/18475)

极限不能瞎求，要真的化简式子

> [!note] 结论一：本质在求期望
> 显然 $P(\theta)$ 的周期是 $2\pi$，于是可以将式子改写为带周期的形式
> $$
> \lim_{T\to+\infty}\frac{1}{T}\int_0^{T}P(\theta) \mathrm{d}\theta=\lim_{T\to+\infty}\frac{1}{2k\pi+t}\left(k\int_0^{2\pi}P(\theta) \mathrm{d}\theta+\int_0^{t}P(\theta)\mathrm{d}\theta\right)
> $$
> 两部分分开处理，得到
> $$
> \lim_{T\to+\infty}\left[\frac{1}{2\pi+\frac{t}{k}}\int_0^{2\pi}P(\theta) \mathrm{d}\theta+\frac{1}{T}\int_0^{t}P(\theta)\mathrm{d}\theta\right]
> $$
> 其中，$\frac{t}{k}, \frac{1}{T} \to0$，于是式子化简为
> $$
> \lim_{T\to+\infty}\frac{1}{2\pi}\int_0^{2\pi}P(\theta) \mathrm{d}\theta=\mathbb{E}[P(\theta)]
> $$
> 

我们观察 $P(\theta)$ 的式子，$\max-\min$ 可以理解为凸包在方向 $\theta$ 上的宽度，也就是：对 $n$ 个点形成的凸包，$\theta$ 方向上的两个极点对应的、垂直于方向 $\theta$ 的两条支撑线之间的距离，。
这个东西是可以直接计算的，但是太过麻烦。我们可以进一步化简。

> [!note] 结论二：根据期望线性性进一步化简
> 期望具有线性性，我们可以把 $\max-\min$ 拆解为凸包上的 $m$ 条边在 $\theta$ 方向上的投影。记旋转 $\theta$ 角度后某条边的实际极角为 $\theta+\varphi_i$。注意凸包转一圈会回到原点（注意两倍和绝对值），于是：
> $$
> 2\mathbb{E}(\Delta x)=\frac{1}{2\pi}\int_0^{2\pi}\sum_{i=1}^m |L_i\cos(\theta+\varphi_i)|\mathrm{d}\theta
> $$
> 有限积分和求和交换顺序，得到
> $$
> =\frac{1}{2\pi}\sum_{i=1}^mL_i\left(\int_0^{2\pi}|\cos(\theta+\varphi_i)|\mathrm{d}\theta\right)
> $$
> 完整周期上的积分不受相位平移影响，于是
> $$
> \int_0^{2\pi}|\cos(\theta+\varphi_i)|\mathrm{d}\theta=\int_0^{2\pi}|\cos\theta|\mathrm{d}\theta=4
> $$
> $y$ 方向上同理，记凸包周长为 $C$，于是
> $$
> \mathbb{E}[P(\theta)]=2\mathbb{E}(\Delta x)+2\mathbb{E}(\Delta y)=\frac{4C}{\pi}
> $$

简单来说就是，宽度的期望可以拆分为每条边的投影长度的期望相加。于是 $\dfrac{4C}{\pi}$ 即为答案，直接计算即可。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const db eps=1e-8;
const db PI=acos(-1.0);
const db inf = numeric_limits<db>::max();
int sgn(auto x) { return (x > eps) - (x < -eps); }

using T = ll;
struct Point {
	T x, y;
    Point(T x=0, T y=0): x(x), y(y) {}
	friend istream& operator>> (istream& is, Point& p) { return is>>p.x>>p.y; }
	friend ostream& operator<< (ostream& os, const Point& p) { return os<<p.x<<" "<<p.y; }
	db ang() { return atan2(y, x); }
	bool operator== (const Point &p) const { return sgn(x-p.x)==0 && sgn(y-p.y)==0; }
	bool operator< (const Point &p) const { return x<p.x || (x==p.x && y<p.y); }
	Point operator+ (const Point &p) const { return Point(x+p.x, y+p.y); }
	Point operator- (const Point &p) const { return Point(x-p.x, y-p.y); }
	Point operator* (const T &k) const { return Point(k*x, k*y); }
	Point operator/ (const T &k) const { return Point(x/k, y/k); }
	T operator* (const Point &p) const { return x*p.x + y*p.y; }
	T operator^ (const Point &p) const { return x*p.y - y*p.x; }  // 叉乘时打括号
	T len2() { return (*this)*(*this); }
	db len() { return sqrtl(len2()); }  // 等价于 hypotl(x, y)
	Point rot(db ang) { return Point(x*cos(ang)-y*sin(ang), x*sin(ang)+y*cos(ang)); }
	Point trunc(db l) { return (*this) * (l/len()); }
};

// 求平面点集的凸包，conv[0]是最左下角的点，逆时针顺序
// Andrew 算法，时间复杂度 O(nlogn)
vector<Point> toConvex(vector<Point> po) {
	if (po.empty()) { return {}; }
	sort(po.begin(), po.end());
    vector<Point> conv;
	auto check = [&](Point u){
		Point p1=conv.back(), p2=conv[conv.size()-2];
		return sgn((p1-p2)^(u-p1))<=0;
	};
	for (const Point &u: po) {
		while (conv.size()>1 && check(u)) { conv.pop_back(); }
		conv.push_back(u);
	}
	size_t k = conv.size();
	po.pop_back();
    reverse(po.begin(),po.end());
	for (const Point &u: po) {
		while (conv.size()>k && check(u)) { conv.pop_back(); }
		conv.push_back(u);
	}
	conv.pop_back();
    return conv;
}

db circ(const vector<Point> &po) {
    db s = 0;
    int n = po.size();
    for (int i = 0; i < n; ++i) {
        s += (po[i] - po[(i + 1) % n]).len();
    }
    return s;
}

void solve() {
    int n;
    cin >> n;
    vector<Point> p(n);
    for (int i = 0; i < n; ++i) {
        cin >> p[i];
    }
    p = toConvex(p);
    db c = circ(p);
    cout << 4 * c / PI << "\n";
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    cout << fixed << setprecision(12);

    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
