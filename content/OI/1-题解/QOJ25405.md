## [QOJ25405. 平衡问题（A Matter of Balance）](https://qoj.ac/contest/4706/problem/25405/statement/zh_cn)

把一个点在三角形内部，转化为三角形只有一条边与线段 $PQ$ 相交
以 $P$ 为原点，$PQ$ 为极轴，极角排序，或者说这相当于对 $A_i-P$ 极角排序
一个更优雅的写法是：对射线 $PQ$ 右侧半平面的那些点取反，也就是按照这个排序：
$$
\begin{cases}
A_i-P & \vec{PQ}\times \vec{PA_i}\ge0 \\
P-A_i& else
\end{cases}
$$
此时固定左侧的点 $A_i$，线段 $A_iA_j$ 与射线 $PQ$ 相交，就可以被转化为：排序后下标满足：
$$
r_i<r_j
$$
同理以 $Q$ 为中心，做类似的映射，可知左侧的点 $A_i$ 和右侧的点 $A_j$ 连线能与 $PQ$ 相交，当且仅当 $r_i<r_j$ 且 $s_i>s_j$，右侧点与左侧点满足条件可以转换为 $r_i>r_j$ 且 $s_i<s_j$
这是一个二维数点问题，可以用树状数组做，得到每个点的“度数”$d_i$，最终答案为
$$
\sum_{i\in L}d_i(|R|-d_i)+\sum_{i\in R}d_i(|L|-d_i)
$$

[***AC 代码***](https://qoj.ac/submission/3493431)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int sgn(ll x) { return (x>0)-(x<0); }

using T = ll;
struct Point {
    T x, y;
    Point(): x(0), y(0) { }
    Point(T x, T y): x(x), y(y) { }
    friend istream& operator>> (istream& is, Point &p) { return is >> p.x >> p.y; }
    int quad() const {
        if (y < 0)       { return 1; }
        else if (y > 0)  { return 3; }
        else if (x >= 0) { return 2; }
        else             { return 4; }
    }
    Point operator+ (const Point &p) const { return Point(x+p.x, y+p.y); }
    Point operator- (const Point &p) const { return Point(x-p.x, y-p.y); }
    T operator* (const Point &p) const { return x*p.x+y*p.y; }
    T operator^ (const Point &p) const { return x*p.y-y*p.x; }
};

struct Line {
    Point a, b, v;
    Line(Point a, Point b): a(a), b(b), v(b-a) { }
    int toLeft(Point p) const { return sgn(v^(p-a)); }
};

struct BIT {
    int n;
    vector<int> t;
    BIT(int n): n(n), t(n+1) { }
    int lowbit(int x) { return x&-x; }
    void modify(int x, int d) {
        for (; x <= n; x += lowbit(x)) { t[x] += d; }
    }
    int query(int x) {
        int res = 0;
        for (; x; x -= lowbit(x)) { res += t[x]; }
        return res;
    }
};

void solve() {
    int n;
    Point p, q;
    cin >> n >> p >> q;
    vector<Point> A(n);
    Line PQ(p, q);
    int lcnt=0, rcnt=0;
    vector<pair<Point,int>> r, s;
    for (int i = 0; i < n; ++i) {
        cin >> A[i];
        if (PQ.toLeft(A[i]) > 0) {
            ++lcnt;
            r.push_back({A[i]-p, i});
            s.push_back({A[i]-q, i});
        } else {
            ++rcnt;
            r.push_back({p-A[i], i});
            s.push_back({q-A[i], i});
        }
    }

    auto cmp = [&](pair<Point,int> x, pair<Point,int> y) {
        return (x.first^y.first) > 0;
    };
    sort(r.begin(), r.end(), cmp);
    sort(s.begin(), s.end(), cmp);
    
    vector<int> rv(n), sv(n);
    for (int i = 0; i < n; ++i) { rv[r[i].second] = i+1; }
    for (int i = 0; i < n; ++i) { sv[s[i].second] = i+1; }
    
    vector<int> d(n);
    
    BIT t1(n);
    for (auto [_, i]: r) {
        if (PQ.toLeft(A[i]) > 0) {
            t1.modify(sv[i], 1);
        } else {
            d[i] += t1.query(n)-t1.query(sv[i]);
        }
    }

    BIT t2(n);
    reverse(r.begin(), r.end());
    for (auto [_, i]: r) {
        if (PQ.toLeft(A[i]) > 0) {
            d[i] += t2.query(sv[i]);
        } else {
            t2.modify(sv[i], 1);
        }
    }

    ll ans = 0;
    for (int i = 0; i < n; ++i) {
        if (PQ.toLeft(A[i]) > 0) {
            ans += 1LL*d[i]*(rcnt-d[i]);
        } else {
            ans += 1LL*d[i]*(lcnt-d[i]);
        }
    }

    cout << ans << "\n";
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
