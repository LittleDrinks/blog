## [CF2210E. Binary Strings are Simple?](https://codeforces.com/contest/2210/problem/E)

> [!tip] Observation
> 每次询问的结果只和区间长度，以及区间内 $0$ 的数量 $num$ 有关

把 $1$ 移动到最后面，逆序数减少 $num$
把 $0$ 移动到最后面，逆序数增加 $len-num$
相当于在 $\bmod len$ 意义下每次 $\pm num$，答案为 $len / \gcd(len,num)$

> [!tip] Observation
> 询问长度为偶数的区间，设得到的答案为 $A$，$len / A$ 的奇偶性与 $num$ 的奇偶性一致

如果不考虑花费，直接询问 $(i,i+1)$ 即可
考虑调和级数，询问所有 $(1,2i)$ 和 $(2,2i+1)$，可以做到 $7n$ 多一点
问题在于每次询问的区间长度太小了。
不妨设 $n$ 为偶数，我们可以通过 $(1,n)$ 和 $(2i+1,n)$ 的答案反推 $(1,2i)$ 的答案，此时每次询问的区间长度都不小于 $n / 2$，一共需要询问大约 $n$ 次，可以做到 $2n$ 左右
如果 $n$ 为奇数，加上一次 $(n-1,n)$ 的询问即可

[***AC 代码***](https://codeforces.com/contest/2210/submission/387360803)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

const int debug = 0;
mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());
int Rand(int a, int b) {
    return uniform_int_distribution<int>(a, b)(rng);
}

string ans;

void solve() {
    int n;
    if (debug == 0) {
        cin >> n;
    } else {
        n = Rand(1, 10);
        ans = string(n, '0');
        for (int i = 0; i < n; ++i) {
            ans[i] += Rand(0, 1);
        }

        cerr << "**********\nans = " << ans << "\n";
        // n = 6;
        // ans = "011000";
    }

    auto query = [&](int l, int r) -> int {
        cout << "? " << l << " " << r << endl;
        if (debug == 0) {
            int x;
            cin >> x;
            return x;
        } else {
            string t = ans.substr(l-1, r-l+1);
            auto inv = [&](string t) {
                int cnt = 0;
                for (int i = 0; i < int(t.size()); ++i) {
                    for (int j = i+1; j < int(t.size()); ++j) {
                        cnt += (t[i] > t[j]);
                    }
                }
                return cnt;
            };
            set<int> S;
            for (int i = 0; i < int(t.size()); ++i) {
                t = t.substr(1) + t[0];
                S.insert(inv(t)%(r-l+1));
            }
            return S.size();
        }
    };

    map<pair<int,int>, int> cache;
    auto query_2 = [&](int l, int r) -> int {
        if (cache.count({l,r})) {
            return cache[{l,r}];
        }
        int v = query(l, r);
        return cache[{l,r}] = ((r-l+1)/v) % 2;
    };


    auto answer = [&](string p) -> int {
        cout << "! " << p << endl;
        if (debug == 0) {
            int x;
            cin >> x;
            return x;
        } else {
            return p == ans;
        }
    };

    auto qqq = [&](int l, int r) {
        int top = n;
        if (top % 2 == l % 2) {
            top--;
        }
        int a = r-l+1 >= top-r ? query_2(l, r) : query_2(l, top)^query_2(r+1, top);
        return a;
    };

    for (char fst = '0'; fst < '2'; ++fst) {
        string a = string(n, '0');
        a[0] = fst;

        for (int i = 2; i <= n; ++i) {
            int st = 1 + (i & 1);
            int s = 0;
            for (int j = st; j < i; ++j) {
                s += (a[j-1] - '0');
                s %= 2;
            }
            if (s != qqq(st, i)) {
                a[i-1] = '1';
            }
        }

        if (answer(a)) {
            return;
        }
    }

    assert(false);
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
