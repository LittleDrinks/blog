## [QOJ24846. 暴击装备](https://qoj.ac/contest/4685/problem/24846/statement/zh_cn)

第一感觉是折半搜索，相当于拿到了两个 $10^6$ 的集合 $A$ 和 $B$，把他们合并起来，求 $F$ 最大值
$$
F(i,j)=\min(P_{A_i}+P_{B_j}, 10^8)\cdot(D_{A_i}+D_{B_j})
$$
首先求一发帕累托前沿，如果 $S$ 内 $x_1\le x_2$ 并且 $y_1\le y_2$，那么 $(x_1,y_1)$ 一定是无用的

记决策点 $opt(j)$ 表示
$$
opt(j)=\arg\max_i F(i,j)
$$
> [!note]- 下证 $opt(j+1)\le opt(j)$
> 记 $h(\cdot)=\min(\cdot,10^8)$，取两个点满足 $P_{A_1}\le P_{A_2}, D_{A_1}\ge D_{A_2}$，计算
> $$
> \begin{align} \\
> \Delta(j)&=F(2,j)-F(1,j) \\
> &=h(P_{A_2}+P_{B_j})(D_{A_2}+D_{B_j})-h(P_{A_1}+P_{B_j})(D_{A_1}+D_{B_j}) \\
> \end{align}
> $$
> 左减右加 $h(P_{A_1}+P_{B_j})(D_{A_2}+D_{B_j})$，得到
> $$
> \begin{align}
> \Delta(j)=&(D_{A_2}-D_{A_1})h(P_{A_1}+P_{B_j}) \\
> &+(D_{A_2}+D_{B_j})[h(P_{A_2}+P_{B_j})-h(P_{A_1}+P_{B_j})]
> \end{align}
> $$
> 对于第一项，$D_{A_2}-D_{A_1}\le 0$，$P_{B_j}$ 单调增，于是整体单调减
> 对于第二项，$D_{A_2}+D_{B_j}\ge 0$，$h(P_{A_2}+P_{B_j})-h(P_{A_1}+P_{B_j})\le 0$，$P_{B_j}$ 和 $D_{B_j}$ 单调增，于是整体单调减
> 于是 $\Delta(j)$ 单调减，即
> $$\Delta(j+1)\le\Delta(j)$$
> 或者说
> $$
> F(1,j)+F(2,j+1)\le F(2,j)+F(1,j+1)
> $$
> （$1<2<j<j+1$，交叉小于包含，四边形不等式得证，于是决策点单调）
> 具体而言，反证法，假设 $opt(j)= i_1 < i_2 =opt(j+1)$，钦定相等时取较小下标，于是
> $$
> \begin{cases}
> F(i_1,j)\ge F(i_2,j) \\
> F(i_1,j+1)< F(i_2,j+1)
> \end{cases}
> $$
> 也就是 $\Delta(j+1)=F(i_2,j+1)-F(i_1,j+1)>0$，而 $\Delta(j)\le 0$，矛盾
> 

决策单调，分治，每次找到 $A$ 集合中点对应 $B$ 集合中的决策点，然后向下分治即可，每次 $A$ 集合的规模会缩小一半，复杂度 $O(n^3\log n^3)$

[***AC 代码***](https://qoj.ac/submission/3460071)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

int sgn(ll x) { return (x>0)-(x<0); }

using T = long long;
struct Point {
    T x, y;
    Point(): x(0), y(0) { }
    Point(T x, T y): x(x), y(y) { }
    friend istream& operator>> (istream& is, Point &p) { return is >> p.x >> p.y; }
    friend ostream& operator<< (ostream& os, const Point &p) { return os << "{" << p.x<<","<<p.y<<"}"; }
    Point operator+ (const Point &p) const { return Point(x+p.x, y+p.y); }
    Point operator- (const Point &p) const { return Point(x-p.x, y-p.y); }
    T operator* (const Point &p) const { return x*p.x+y*p.y; }
    T operator^ (const Point &p) const { return x*p.y-y*p.x; }
    bool operator< (const Point &p) const { return pair{x,y} < pair{p.x,p.y}; }
};

void pareto(vector<Point> &P) {
    sort(P.begin(), P.end());
    int k = P.size();
    ll mxY = -numeric_limits<ll>::max();
    for (int i = (int)P.size() - 1; i >= 0; --i) {
        if (P[i].y > mxY) {
            mxY = P[i].y;
            P[--k] = P[i];
        }
    }
    P.erase(P.begin(), P.begin() + k);
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int c[6];
    vector<Point> h[6];
    for (int i = 0; i < 6; ++i) {
        cin >> c[i];
        h[i].resize(c[i]);
    }
    for (int i = 0; i < 6; ++i) {
        for (int j = 0; j < c[i]; ++j) {
            cin >> h[i][j];
        }
    }
    vector<Point> P[2];
    for (int i = 0; i < c[0]; ++i) {
        for (int j = 0; j < c[1]; ++j) {
            for (int k = 0; k < c[2]; ++k) {
                P[0].push_back(h[0][i]+h[1][j]+h[2][k]);
            }
        }
    }
    for (int i = 0; i < c[3]; ++i) {
        for (int j = 0; j < c[4]; ++j) {
            for (int k = 0; k < c[5]; ++k) {
                P[1].push_back(h[3][i]+h[4][j]+h[5][k]);
            }
        }
    }

    pareto(P[0]);
    pareto(P[1]);

    ll ans = 0;
    auto solve = [&](auto &&solve, const vector<Point> &A, const vector<Point> &B, int l, int r, int x, int y) -> void {
        if (l > r) return;
        int m = ((l+r)>>1);
        ll mx = x, opt = min(A[m].x+B[x].x, (ll)1e8) * (A[m].y+B[x].y);
        for (int i = x; i <= y; ++i) {
            ll res = min(A[m].x+B[i].x, (ll)1e8) * (A[m].y+B[i].y);
            if (res > mx) {
                mx = res;
                opt = i;
            }
        }
        gmax(ans, mx);
        solve(solve, A, B, l, m-1, opt, y);
        solve(solve, A, B, m+1, r, x, opt);
    };

    solve(solve, P[0], P[1], 0, P[0].size()-1, 0, P[1].size()-1);
    // cout << ans << "\n";
    cout << fixed << setprecision(15) << (1+(db)ans/(db)1e16) << "\n";
}
```
