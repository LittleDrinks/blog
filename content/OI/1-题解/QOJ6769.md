## [QOJ6769. Monster Hunter](https://qoj.ac/contest/1292/problem/6769)

二分答案，这样就可以知道攻击力为 $3,2,1$ 的攻击分别可以用几次。
如果 $a_i$ 只有 $1,2$，那么可以在不浪费攻击的情况下贪心地放 $2$，然后放 $1$ 填上空位。
加上 $3$ 后的情况类似，但是不能直接按照这个思路贪心。一组会产生浪费的 hack 是
```
1
3
3 2 2
2
7 7
```
我们发现把 $3$ 放完之后，如果剩下的数中奇数比 $1$ 的数量多，就不得不把 $2$ 当 $1$ 用，产生很多浪费。所以我们先把 $3$ 放到不小于 $3$ 的奇数上面，然后凑成 $6$ 继续填到大整数上面，最后按原来的思路贪心放。

[***AC 代码***](https://qoj.ac/submission/2198604)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;
using i128 = __int128;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }

void solve()
{
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    int m;
    cin >> m;
    vector<ll> h(m);
    for (int i = 0; i < m; ++i) {
        cin >> h[i];
    }
    ll L = 0, R = 1E14 + 5;
    while (L + 1 < R)
    {
        // ll mid = 2;
        ll mid = (L + R) >> 1;
        bool ok = true;
        
        vector<ll> cnt(3);
        for (int i = 0; i < n; ++i) {
            cnt[a[i]-1] += mid / n;
            if (i < mid % n) {
                cnt[a[i]-1]++;
            }
        }
        
        vector<ll> hh(h);
        for (int i = 0; i < m; ++i) {
            if (hh[i] >= 3 && h[i] % 2 == 1 && cnt[2] > 0) {
                hh[i] -= 3;
                cnt[2]--;
            }
        }
        for (int i = 0; i < m; ++i) {
            if (hh[i] >= 6 && cnt[2] >= 2) {
                ll v = min(hh[i] / 6, cnt[2] / 2);
                hh[i] -= 6 * v;
                cnt[2] -= 2 * v;
            }
        }
        for (int i = 0; i < m; ++i) {
            if (hh[i] >= 3 && cnt[2] > 0) {
                hh[i] -= 3;
                cnt[2]--;
            }
        }
        for (int i = 0; i < m; ++i) {
            if (hh[i] == 2 && cnt[2] > 0) {
                hh[i] -= 2;
                cnt[2]--;
            }
        }
        for (int i = 0; i < m; ++i) {
            if (hh[i] == 1 && cnt[2] > 0) {
                hh[i] -= 1;
                cnt[2]--;
            }
        }
        ll s = accumulate(hh.begin(), hh.end(), 0LL);
        ll odd = count_if(hh.begin(), hh.end(), [&](ll x){ return x % 2 == 1; });

        if (cnt[0] + 2 * cnt[1] < s + max(0LL, odd - cnt[0])) {
            ok = false;
        }

        if (ok) {
            R = mid;
        } else {
            L = mid;
        }
    }
    cout << R << "\n";
}

signed main()
{
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0); 
    int T = 1;
    cin >> T;
    while (T--) {
        solve();
    }
}
```
