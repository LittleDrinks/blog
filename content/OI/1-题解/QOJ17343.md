## [QOJ17343. Operating Robot](https://qoj.ac/contest/3729/problem/17343)

只能往右上角走，没法往回走，因此总步数就等于起点到终点的曼哈顿距离。
设这个距离为 $t=x+y$，将已有的 $n$ 步分为两部分，先走 $\left\lfloor  \frac{t}{n} \right\rfloor$ 轮，每轮走完整的 $n$ 步，再走 $t\bmod n$ 步。
枚举字符串中有多少个 $\texttt{2}$ 实际上是 $\texttt{1}$，这样一来，$\texttt{01}$ 的数量都固定，进而前 $\left\lfloor  \frac{t}{n} \right\rfloor$ 轮固定，最后再根据两个方向上和终点的距离，确定最后一轮在前 $t\bmod n$ 个字符中需要放多少 $\texttt{01}$。
上述方法可以 $O(1)$ 检查是否合法，但是如何摆放仍然不确定。
由于要求字典序最小，我们发现答案一定是：在前 $t\bmod n$ 个字符中先填确定数量的 $\texttt{0}$，再填确定数量的 $\texttt{1}$，然后在剩下字符中先填 $\texttt{0}$ 后填 $\texttt{1}$。
只需要用一个二元组维护【前面填多少 $\texttt{0}$】、【总共填多少 $\texttt{0}$】，就可以表示答案，也可以很方便地在 $count_s(2)+1$ 个候选项中找到最大的那一个。

[***AC 代码***](https://qoj.ac/submission/2342099)

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

template<typename T1,typename T2> ostream& operator<<(ostream& os, const pair<T1,T2>& p) {
    return os<<"("<<p.first<<","<<p.second<<")";
}
template<typename T> ostream& operator<<(ostream& os, const vector<T>& v) {
    os<<"["; for(int i=0;i<v.size();++i) os<<(i?", ":"")<<v[i]; return os<<"]";
}
template<typename T> void gmax(T&x,T y){ x = max(x, y); }
template<typename T> void gmin(T&x,T y){ x = min(x, y); }


void solve() {
    int n;
    ll x, y;
    string s;
    cin >> n >> x >> y >> s;
    vector<int> a(n);
    vector<int> id;
    for (int i = 0; i < n; ++i) {
        a[i] = s[i] - '0';
        if (a[i] == 2) {
            id.push_back(i);
        }
    }

    ll t = (x + y) / n;
    ll p = (x + y) % n;
    ll cnt[3] = {0, 0, 0};
    cnt[0] = count(a.begin(), a.end(), 0);
    cnt[1] = count(a.begin(), a.end(), 1);
    cnt[2] = count(a.begin(), a.end(), 2);
    ll pre[3] = {0, 0, 0};
    pre[0] = count(a.begin(), a.begin() + p, 0);
    pre[1] = count(a.begin(), a.begin() + p, 1);
    pre[2] = count(a.begin(), a.begin() + p, 2);
    pair<ll,ll> ans = {-1, -1};

    for (ll z = 0; z <= cnt[2]; ++z) {
        ll xx = t * (cnt[0] + z) + pre[0];
        ll yy = t * (cnt[1] + cnt[2] - z) + pre[1];
        if (xx > x || yy > y) {
            continue;
        }
        if (x - xx > z || y - yy > cnt[2] - z) {
            continue;
        }
        ans = max(ans, make_pair(x-xx, z));
    }

    if (ans.first == -1) {
        cout << "-1\n";
    } else {
        for (int i = 0; i < ans.first; ++i) {
            a[id[i]] = 0;
        }
        for (int i = ans.first; i < pre[2]; ++i) {
            a[id[i]] = 1;
        }
        int rem = ans.second - ans.first;
        for (int i = pre[2]; i < pre[2] + rem; ++i) {
            a[id[i]] = 0;
        }
        for (int i = pre[2] + rem; i < cnt[2]; ++i) {
            a[id[i]] = 1;
        }
        for (int i = 0; i < n; ++i) {
            cout << a[i];
        }
        cout << "\n";
    }
    
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }
}
```
